namespace aspect

  namespace Adjoint

    /**
     * Semantic, rolling state of the reverse recurrence. The three entries of
     * every history are respectively the pending current, previous, and
     * previous-previous time levels for @p next_reverse_step. No cache
     * container or reconstructible forward field is serialized here.
     */
    struct AdjointRestartState
    {
      unsigned int next_reverse_step = 0;
      unsigned int cleanup_frontier = numbers::invalid_unsigned_int;
      unsigned int generation = 0;
      // Sparse replay state is deliberately metadata-only. The transient
      // replay window is reconstructed from this anchor and the schedule.
      unsigned int trajectory_anchor_spacing = 1;
      unsigned int required_sparse_anchor = numbers::invalid_unsigned_int;
      unsigned int trajectory_schedule_records = 0;
      std::uint64_t trajectory_schedule_fingerprint = 0;
      TrajectoryStorageMode trajectory_storage_mode =
        TrajectoryStorageMode::minimal;
      std::array<LinearAlgebra::BlockVector, 3> temperature_history;
      std::array<LinearAlgebra::BlockVector, 3> stokes_right_hand_side_history;
      std::vector<std::array<LinearAlgebra::BlockVector, 3>> composition_history;
      std::map<std::string, double> objective_values;
    };

    /** Result of one collective checkpoint transaction. */
    struct CheckpointWriteResult
    {
      bool committed = false;
      bool latest_published = false;
      std::size_t local_bytes = 0;
    };

    /**
     * Transactional, rank-local adjoint restart storage. A partially written
     * .tmp generation is never selected for restore. The format deliberately
     * supports only the same MPI rank count and DoF ownership that created it.
     */
    class RestartCheckpointManager
    {
      public:
        /** Create a checkpoint store next to the forward trajectory archive. */
        RestartCheckpointManager(const std::string &trajectory_directory,
                                 const MPI_Comm mpi_communicator);

        /** Return whether the collectively published latest pointer exists. */
        bool
        has_committed_checkpoint() const;

        /** Write, commit, and make @p state the latest restart generation. */
        CheckpointWriteResult
        write(
          AdjointRestartState &state,
          const Diagnostics::FailureInjection<Diagnostics::CheckpointFailurePoint> &injection = {});

        /** Restore the latest committed state into already initialized vectors. */
        std::size_t
        restore(AdjointRestartState &state) const;

        /** Remove restart generations and the pointer after normal completion. */
        void
        remove_temporary_data();

      private:
        std::string
        checkpoint_directory(const unsigned int generation) const;

        std::string
        temporary_checkpoint_directory(const unsigned int generation) const;

        static void
        write_vector(std::ostream &output,
                     const LinearAlgebra::BlockVector &vector);

        static void
        read_vector(std::istream &input,
                    LinearAlgebra::BlockVector &vector);

        std::size_t
        write_state(const std::string &path, const AdjointRestartState &state) const;

        std::size_t
        read_state(const std::string &path, AdjointRestartState &state) const;

        void
        write_metadata(const std::string &path,
                       const AdjointRestartState &state,
                       const unsigned int generation) const;

        void
        read_metadata(const std::string &path, AdjointRestartState &state) const;

        std::string
        rank_payload_filename() const;

        bool
        collective_success(const bool local_success,
                           const std::string &stage) const;

        void
        prune_old_generations(const unsigned int newest) const;

        std::string directory;
        MPI_Comm communicator;
        unsigned int rank;
        unsigned int n_processes;
    };
  }
}

    /**
     * Own the durable storage used by a time-dependent adjoint evaluation.
     *
     * The primary archive is the source of reverse replay. An optional dense
     * reference archive exists only while sparse-replay equivalence is being
     * checked. The restart manager is created next to the primary archive so
     * that a checkpoint always identifies the journal it resumes.
     *
     * Replay-window state remains owned by TimeDependentWorkflow because it
     * is transient and must never be serialized by this class.
     *
     * @ingroup Adjoint
     */
    template <int dim>
    class TrajectoryManager : public SimulatorAccess<dim>
    {
      public:
        /** Immutable-after-setup controls for storage, replay, and restart. */
        struct Configuration
        {
          TrajectoryStorageMode storage_mode = TrajectoryStorageMode::minimal;
          TrajectoryCleanupPolicy cleanup_policy = TrajectoryCleanupPolicy::safe_incremental;
          unsigned int records_per_segment = 4;
          unsigned int anchor_spacing = 1;
          unsigned int checkpoint_interval = 50;
        };

        /** Parse the production trajectory configuration. */
        void parse_parameters(ParameterHandler &prm);

        /** Return the parsed production trajectory configuration. */
        const Configuration &configuration() const;

        /** Record one accepted forward state and its outgoing schedule data. */
        void record_forward_snapshot(Manager<dim> &manager,
                                     const double outgoing_time_step,
                                     const bool has_outgoing_step);

        /** Mark the final accepted state and materialize its sparse anchor. */
        void mark_forward_snapshot_terminal(Manager<dim> &manager);

        /** Destroy all in-memory handles without deleting durable files. */
        void reset();

        /**
         * Discard transient reconstruction windows and reset replay counters.
         * Durable trajectory and restart storage remain open.
         */
        void reset_replay_state();

        /** Create or open the primary accepted-step archive. */
        void create_archive(const TrajectoryStorageMode mode,
                            const std::string &output_directory,
                            const bool open_existing,
                            const unsigned int records_per_segment);

        /** Create the optional dense archive used by sparse-replay checks. */
        void create_reference_archive(const TrajectoryStorageMode mode,
                                      const std::string &output_directory,
                                      const unsigned int records_per_segment);

        bool has_archive() const;
        bool has_reference_archive() const;
        bool has_restart_checkpoint_manager() const;

        TrajectoryStore &archive();
        const TrajectoryStore &archive() const;
        TrajectoryStore &reference_archive();
        const TrajectoryStore &reference_archive() const;

        /** Create reverse-checkpoint storage next to the primary archive. */
        void create_restart_checkpoint_manager();

        RestartCheckpointManager &restart_checkpoint();
        const RestartCheckpointManager &restart_checkpoint() const;

      private:
        friend class Manager<dim>;
        friend class TimeDependentWorkflow<dim>;

        struct ReconstructedTrajectoryState
        {
          unsigned int record = numbers::invalid_unsigned_int;
          LinearAlgebra::BlockVector state;
        };

        struct ForwardReplayWindow
        {
          unsigned int anchor = numbers::invalid_unsigned_int;
          unsigned int end = numbers::invalid_unsigned_int;
          std::deque<ReconstructedTrajectoryState> states;
        };

        /** Return the inclusive sparse interval containing @p record. */
        std::pair<unsigned int, unsigned int>
        sparse_interval_for_record(const unsigned int record) const;

        /** Return the anchor needed to execute the given reverse step. */
        unsigned int required_anchor_for_reverse_step(
          const unsigned int reverse_step) const;

        /** Store the durable trajectory identity in a reverse checkpoint. */
        void set_restart_trajectory_identity(AdjointRestartState &state) const;

        /** Validate a reverse checkpoint against the currently open archive. */
        void validate_sparse_restart_state(const AdjointRestartState &state) const;

        /** Materialize the three-level state package for one sparse anchor. */
        void store_sparse_anchor_package(Manager<dim> &manager,
                                         const unsigned int anchor);

        std::unique_ptr<TrajectoryStore> primary_archive;
        std::unique_ptr<TrajectoryStore> dense_reference_archive;
        std::unique_ptr<RestartCheckpointManager> restart_manager;
        std::deque<ReconstructedTrajectoryState> reverse_state_cache;
        std::deque<ForwardReplayWindow> forward_replay_windows;
        std::deque<ReconstructedTrajectoryState> terminal_anchor_history;
        std::vector<unsigned int> sparse_anchor_steps;
        unsigned int stokes_recovery_solves = 0;
        unsigned int sparse_replay_transition_count = 0;
        unsigned int sparse_restart_replay_transition_count = 0;
        unsigned int sparse_anchor_load_count = 0;
        unsigned int sparse_interval_replay_count = 0;
        unsigned int sparse_peak_window_state_count = 0;
        double sparse_max_state_relative_difference = 0.0;
        double sparse_max_velocity_relative_difference = 0.0;
        double sparse_max_pressure_relative_difference = 0.0;
        Configuration parsed_configuration;
    };
  }

    struct ReplayStepMetadata
    {
      /** Index of the archived physical state q_n. */
      unsigned int step = numbers::invalid_unsigned_int;

      /** Physical time associated with q_n. */
      double time = 0.0;

      /**
       * Time step that produced q_n from q_(n-1). This is the simulator's
       * current time_step while q_n is archived.
       */
      double time_step = 0.0;

      /** Time step preceding @p time_step, needed by the BDF2 history. */
      double old_time_step = 0.0;

      /**
       * The accepted step size that advances q_n to q_(n+1). This field is
       * deliberately distinct from @p time_step: it is selected by the
       * forward time-step manager after q_n has been solved.
       */
      double outgoing_time_step = 0.0;

      /** Whether q_n has a subsequently solved accepted transition. */
      bool has_outgoing_step = false;
    };

    struct TrajectoryStorageStatistics
    {
      std::size_t metadata_bytes = 0;
      std::size_t temperature_bytes = 0;
      std::size_t composition_bytes = 0;
      std::size_t velocity_bytes = 0;
      std::size_t pressure_bytes = 0;

      std::size_t total_bytes() const
      {
        return metadata_bytes + temperature_bytes + composition_bytes
               + velocity_bytes + pressure_bytes;
      }
    };

    struct TrajectoryCleanupResult
    {
      bool complete = false;
      unsigned int records_deleted = 0;
      unsigned int payload_files_deleted = 0;
      TrajectoryStorageStatistics reclaimed;
    };

    /**
     * Fixed-mesh, same-layout trajectory archive. Each logical record is one
     * concatenated, rank-local payload. Ranks write non-overlapping ranges of

     * the small manifest. Bytes not named by that manifest are never valid
     * records. This is intentionally not a repartition-capable checkpoint.
     */
    class TrajectoryStore
    {
      public:
        TrajectoryStore(const TrajectoryStorageMode mode,
                        const std::string &output_directory,
                        const MPI_Comm mpi_communicator,
                        const bool open_existing = false,
                        const unsigned int requested_records_per_segment = 4);

        ~TrajectoryStore();

        TrajectoryStore(const TrajectoryStore &) = delete;
        TrajectoryStore &operator=(const TrajectoryStore &) = delete;

        TrajectoryStorageMode mode() const;
        const std::string &archive_directory() const;
        unsigned int size() const;
        bool has_record(const unsigned int record) const;
        unsigned int first_available_record() const;
        unsigned int last_available_record() const;
        unsigned int payload_record_count() const;

        /** A stable identity for the dense accepted-step journal. */
        std::uint64_t schedule_fingerprint() const;

        /** Sum manifest and retained shared segment bytes. */
        std::size_t shared_archive_bytes() const;

        /** Number of shared payload segments named by the manifest. */
        unsigned int shared_segment_count() const;

        TrajectoryCleanupResult
        reclaim_sparse_payloads_after(
          const unsigned int last_to_keep,
          const Diagnostics::FailureInjection<Diagnostics::CleanupFailurePoint> &injection = {});

        /** Install a typed append failure from a test driver. */
        void
        set_append_failure_injection_for_testing(
          const Diagnostics::FailureInjection<Diagnostics::AppendFailurePoint> &injection);

        TrajectoryCleanupResult
        reclaim_records_after(
          const unsigned int first_to_remove,
          const Diagnostics::FailureInjection<Diagnostics::CleanupFailurePoint> &injection = {});

        void remove_archive();
        const ReplayStepMetadata &metadata(const unsigned int record) const;
        void mark_last_record_terminal();
        void append_schedule(const ReplayStepMetadata &step_metadata);

        void store_payload(const unsigned int record,
                           const LinearAlgebra::BlockVector &solution,
                           const unsigned int temperature_block,
                           const std::vector<unsigned int> &composition_blocks,

                  const unsigned int temperature_block,
                  const std::vector<unsigned int> &composition_blocks,
                  const unsigned int velocity_block,
                  const unsigned int pressure_block) const;

        TrajectoryStorageStatistics storage_statistics() const;
        std::size_t memory_consumption() const;

      private:
        class Implementation;
        std::unique_ptr<Implementation> implementation;
    };

    RestartCheckpointManager::RestartCheckpointManager(
      const std::string &trajectory_directory,
      const MPI_Comm mpi_communicator)
      :
      directory(trajectory_directory + "adjoint_restart/"),
      communicator(mpi_communicator),
      rank(Utilities::MPI::this_mpi_process(mpi_communicator)),
      n_processes(Utilities::MPI::n_mpi_processes(mpi_communicator))
    {}



    bool
    RestartCheckpointManager::has_committed_checkpoint() const
    {
      // A generation directory alone is not a commit record. Only the
      // atomically replaced "latest" pointer makes a checkpoint visible to
      // restore(), so abandoned temporary or complete-but-unpublished
      // generations are deliberately ignored.
      unsigned int has_checkpoint = 0;
      bool local_success = true;
      if (rank == 0)
        {
          std::error_code error;
          has_checkpoint = std::filesystem::exists(directory + "latest", error) ? 1 : 0;
          local_success = !error;
        }
      collective_success(local_success, "inspect committed checkpoint pointer");
      return Utilities::MPI::broadcast(communicator, has_checkpoint, 0) == 1;
    }



    CheckpointWriteResult
    RestartCheckpointManager::write(
      AdjointRestartState &state,
      const Diagnostics::FailureInjection<Diagnostics::CheckpointFailurePoint> &injection)
    {
      const unsigned int generation = state.generation + 1;
      const std::string checkpoint = checkpoint_directory(generation);
      const std::string temporary = temporary_checkpoint_directory(generation);
      const auto inject_here = [this, &injection]
                               (const Diagnostics::CheckpointFailurePoint point)
      {
        return injection.point == point && injection.applies_to_rank(rank);
      };

      // The on-disk transaction is
      //
      //   create generation.tmp
      //   -> write all rank payloads
      //   -> write shared metadata
      //   -> rename generation.tmp to generation
      //   -> atomically replace latest.
      //
      // Every stage is made collective before advancing. Consequently no rank
      // can publish a generation for which another rank failed to write its
      // payload. The temporary directory is intentionally harmless after a
      // crash because restore follows only the atomically published pointer.
      bool local_success = true;
      if (rank == 0)
        try
          {
            std::error_code error;
            std::filesystem::create_directories(directory, error);
            if (error)
              local_success = false;
            std::filesystem::remove_all(temporary, error);
            if (error)
              local_success = false;
            std::filesystem::create_directory(temporary, error);
            if (error)
              local_success = false;
          }
        catch (const std::exception &)
          {
            local_success = false;
          }
      if (!collective_success(local_success, "create temporary checkpoint generation"))
        return {};
      if (!collective_success(
            !inject_here(Diagnostics::CheckpointFailurePoint::after_temporary_directory_creation),
            "inject checkpoint failure after temporary directory creation"))
        return {};

      std::size_t bytes = 0;
      local_success = true;
      try
        {
          // Each rank writes only its owned pieces of the fixed three-level
          // reverse history. The metadata is withheld until all payload files
          // have closed successfully.
          bytes = write_state(temporary, state);
        }
      catch (const std::exception &)
        {
          local_success = false;
        }
      local_success = local_success
                      && !inject_here(Diagnostics::CheckpointFailurePoint::after_rank_payload_write);
      if (!collective_success(local_success, "write rank-local checkpoint payloads"))
        return {false, false, bytes};
      if (!collective_success(
            !inject_here(Diagnostics::CheckpointFailurePoint::before_metadata_publication),
            "inject checkpoint failure before metadata publication"))
        return {false, false, bytes};

      local_success = true;
      if (rank == 0)
        try
          {
            // Shared metadata binds the vector payloads to the next reverse
            // step, cleanup frontier, sparse anchor, and trajectory schedule
            // identity. Publishing a payload without these invariants would
            // not be sufficient for a reproducible resume.
            write_metadata(temporary, state, generation);
          }
        catch (const std::exception &)
          {
            local_success = false;
          }
      if (!collective_success(local_success, "write checkpoint metadata"))
        return {false, false, bytes};
      if (!collective_success(
            !inject_here(Diagnostics::CheckpointFailurePoint::after_metadata_publication),
            "inject checkpoint failure before directory commit"))
        return {false, false, bytes};

      local_success = true;
      if (rank == 0)
        try
          {
            std::error_code error;
            std::filesystem::remove_all(checkpoint, error);
            if (error)
              local_success = false;
            if (local_success)
              {
                // This rename makes the generation internally complete, but
                // it is still unreachable by restore() until "latest" is
                // replaced below. A crash in between merely leaves garbage
                // that can be reclaimed later.
                std::filesystem::rename(temporary, checkpoint, error);
                if (error)
                  local_success = false;
              }
          }
        catch (const std::exception &)
          {
            local_success = false;
          }
      if (!collective_success(local_success, "commit checkpoint generation"))
        return {false, false, bytes};
      if (!collective_success(
            !inject_here(Diagnostics::CheckpointFailurePoint::before_latest_publication),
            "inject checkpoint failure before latest publication"))
        return {false, false, bytes};

      local_success = true;
      if (rank == 0)
        try
          {
            // Replacing latest is the external commit point. The temporary
            // pointer is fully flushed/closed before rename, so readers select
            // either the preceding generation or this complete generation.
            const std::string latest_tmp = directory + "latest.tmp";
            std::ofstream latest(latest_tmp, std::ios::trunc);
            latest << generation << '\n';
            latest.close();
            if (!latest)
              local_success = false;
            if (local_success)
              {
                std::error_code error;
                std::filesystem::rename(latest_tmp, directory + "latest", error);
                if (error)
                  local_success = false;
              }
          }
        catch (const std::exception &)
          {
            local_success = false;
          }
      if (!collective_success(local_success, "publish committed checkpoint generation"))
        return {false, false, bytes};
      if (!collective_success(
            !inject_here(Diagnostics::CheckpointFailurePoint::after_latest_publication),
            "inject checkpoint failure after latest publication"))
        return {false, true, bytes};

      state.generation = generation;
      return {true, true, bytes};
    }



    std::size_t
    RestartCheckpointManager::restore(AdjointRestartState &state) const
    {
      // Resolve the commit pointer first and never scan for the numerically
      // largest directory. This preserves the transaction semantics when a
      // crash leaves a newer, complete-looking but unpublished generation.
      unsigned int generation = 0;
      bool local_success = true;
      if (rank == 0)
        try
          {
            std::ifstream latest(directory + "latest");
            latest >> generation;
            if (!latest || generation == 0)
              local_success = false;
          }
        catch (const std::exception &)
          {
            local_success = false;
          }
      collective_success(local_success, "read committed checkpoint pointer");
      generation = Utilities::MPI::broadcast(communicator, generation, 0);
      const std::string checkpoint = checkpoint_directory(generation);

      // Shared metadata is restored before rank payloads so the caller's
      // vectors and trajectory identity can be validated against the selected
      // generation. Every rank must agree on this phase before local reads.
      local_success = true;
      try
        {
          read_metadata(checkpoint, state);
          if (state.generation != generation)
            local_success = false;
        }
      catch (const std::exception &)
        {
          local_success = false;
        }
      collective_success(local_success, "restore shared checkpoint metadata");

      std::size_t bytes = 0;
      local_success = true;
      try
        {
          bytes = read_state(checkpoint, state);
        }
      catch (const std::exception &)
        {
          local_success = false;
        }
      collective_success(local_success, "restore rank-local checkpoint payloads");
      return bytes;
    }



    void
    RestartCheckpointManager::remove_temporary_data()
    {
      bool local_success = true;
      if (rank == 0)
        {
          std::error_code error;
          std::filesystem::remove_all(directory, error);
          local_success = !error;
        }
      collective_success(local_success, "remove adjoint restart data");
    }

    void
    RestartCheckpointManager::write_vector(std::ostream &output,
                                           const LinearAlgebra::BlockVector &vector)
    {
      // Persist global indices together with values for every locally owned
      // entry. The index makes same-size-but-different MPI partitions fail
      // loudly during restore instead of silently permuting an adjoint state;
      // ghosts are excluded because they are not owned data.
      const std::uint64_t blocks = vector.n_blocks();
      output.write(reinterpret_cast<const char *>(&blocks), sizeof(blocks));
      for (unsigned int b = 0; b < vector.n_blocks(); ++b)
        {
          const std::uint64_t count = vector.block(b).locally_owned_elements().n_elements();
          output.write(reinterpret_cast<const char *>(&count), sizeof(count));
          for (const auto index : vector.block(b).locally_owned_elements())
            {
              const std::uint64_t stored_index = index;
              const double value = vector.block(b)(index);
              output.write(reinterpret_cast<const char *>(&stored_index), sizeof(stored_index));
              output.write(reinterpret_cast<const char *>(&value), sizeof(value));
            }
        }
      AssertThrow(output, ExcMessage("Writing an adjoint restart vector failed."));
    }



    void
    RestartCheckpointManager::read_vector(std::istream &input,
                                          LinearAlgebra::BlockVector &vector)
    {
      std::uint64_t blocks = 0;
      input.read(reinterpret_cast<char *>(&blocks), sizeof(blocks));
      AssertThrow(input && blocks == vector.n_blocks(),
                  ExcMessage("Adjoint restart vector does not match the active DoF layout."));
      for (unsigned int b = 0; b < vector.n_blocks(); ++b)
        {
          std::uint64_t count = 0;
          input.read(reinterpret_cast<char *>(&count), sizeof(count));
          AssertThrow(input && count == vector.block(b).locally_owned_elements().n_elements(),
                      ExcMessage("Adjoint restart vector does not match the active MPI partitioning."));
          // IndexSet iteration order must match the writer. Checking every
          // stored global index turns partition compatibility into an exact
          // invariant rather than relying only on the local element count.
          for (const auto index : vector.block(b).locally_owned_elements())
            {
              std::uint64_t stored_index = 0;
              double value = 0;
              input.read(reinterpret_cast<char *>(&stored_index), sizeof(stored_index));
              input.read(reinterpret_cast<char *>(&value), sizeof(value));
              AssertThrow(input && stored_index == index,
                          ExcMessage("Adjoint restart vector does not match the active MPI ownership."));
              vector.block(b)(index) = value;
            }
          vector.block(b).compress(VectorOperation::insert);
        }
    }



    std::size_t
    RestartCheckpointManager::write_state(const std::string &path,
                                          const AdjointRestartState &state) const
    {
      const std::string filename = path + rank_payload_filename();
      std::ofstream output(filename, std::ios::binary | std::ios::trunc);
      AssertThrow(output, ExcMessage("Could not write rank-local adjoint restart payload <"
                                     + filename + ">."));
      const std::uint64_t magic = 0x4153504543544152ULL;
      output.write(reinterpret_cast<const char *>(&magic), sizeof(magic));
      const auto write = [&output](const LinearAlgebra::BlockVector &vector)
      {
        write_vector(output, vector);
      };
      // Reverse BDF propagation has exactly three slots: current, previous,
      // and previous-previous. Temperature and Stokes RHS histories are stored
      // level-major; each composition field carries the same three-level
      // layout. Restore reads this schema in precisely the same order.
      for (unsigned int level = 0; level < 3; ++level)
        {
          write(state.temperature_history[level]);
          write(state.stokes_right_hand_side_history[level]);
        }
      for (const auto &composition : state.composition_history)
        for (unsigned int level = 0; level < 3; ++level)
          write(composition[level]);
      output.close();
      AssertThrow(output, ExcMessage("Writing rank-local adjoint restart payload <"
                                     + filename + "> failed."));
      return static_cast<std::size_t>(std::filesystem::file_size(filename));
    }



    std::size_t
    RestartCheckpointManager::read_state(const std::string &path,
                                         AdjointRestartState &state) const
    {
      const std::string filename = path + rank_payload_filename();
      std::ifstream input(filename, std::ios::binary | std::ios::ate);
      AssertThrow(input, ExcMessage("Could not read rank-local adjoint restart payload <"
                                    + filename + ">."));
      const std::size_t bytes = static_cast<std::size_t>(input.tellg());
      input.seekg(0);
      std::uint64_t magic = 0;
      input.read(reinterpret_cast<char *>(&magic), sizeof(magic));
      AssertThrow(input && magic == 0x4153504543544152ULL,
                  ExcMessage("Adjoint restart payload has an invalid format."));
      const auto read = [&input](LinearAlgebra::BlockVector &vector)
      {
        read_vector(input, vector);
      };
      for (unsigned int level = 0; level < 3; ++level)
        {
          read(state.temperature_history[level]);
          read(state.stokes_right_hand_side_history[level]);
        }
      for (auto &composition : state.composition_history)
        for (unsigned int level = 0; level < 3; ++level)
          read(composition[level]);
      return bytes;
    }



    void
    RestartCheckpointManager::write_metadata(const std::string &path,
                                             const AdjointRestartState &state,
                                             const unsigned int generation) const
    {
      std::ofstream output(path + "state.txt", std::ios::trunc);
      AssertThrow(output, ExcMessage("Could not write adjoint restart metadata."));
      output.precision(std::numeric_limits<double>::max_digits10);
      output << "ASPECT_ADJOINT_RESTART_V3\n"
             << generation << ' ' << state.next_reverse_step << ' '
             << state.cleanup_frontier << ' '
             << (state.trajectory_storage_mode == TrajectoryStorageMode::minimal ? 0 : 1)
             << ' ' << state.composition_history.size() << ' ' << n_processes << ' '
             << state.trajectory_anchor_spacing << ' '
             << state.required_sparse_anchor << ' '
             << state.trajectory_schedule_records << ' '
             << state.trajectory_schedule_fingerprint << '\n'
             << state.objective_values.size() << '\n';

      // Objective values are scalar accumulated results, not reconstructible
      // from the three adjoint history slots. Store their names with quoting
      // so whitespace in registered objective names round-trips exactly.
      for (const auto &entry : state.objective_values)
        output << std::quoted(entry.first) << '\t' << entry.second << '\n';
      output.close();
      AssertThrow(output, ExcMessage("Writing adjoint restart metadata failed."));
    }



    void
    RestartCheckpointManager::read_metadata(const std::string &path,
                                            AdjointRestartState &state) const
    {
      unsigned int composition_fields = 0;
      unsigned int storage_mode = 0;
      unsigned int stored_processes = 0;
      std::ifstream input(path + "state.txt");
      std::string magic;
      std::size_t objectives = 0;
      input >> magic >> state.generation >> state.next_reverse_step
      >> state.cleanup_frontier >> storage_mode >> composition_fields
      >> stored_processes >> state.trajectory_anchor_spacing
      >> state.required_sparse_anchor >> state.trajectory_schedule_records
      >> state.trajectory_schedule_fingerprint >> objectives;
      AssertThrow(input && magic == "ASPECT_ADJOINT_RESTART_V3"
                  && composition_fields == state.composition_history.size()
                  && stored_processes == n_processes,
                  ExcMessage("Adjoint restart metadata is incompatible with this MPI run."));
      AssertThrow(storage_mode == 0 || storage_mode == 1,
                  ExcMessage("Adjoint restart metadata has an invalid storage mode."));
      state.trajectory_storage_mode = storage_mode == 0
                                      ? TrajectoryStorageMode::minimal
                                      : TrajectoryStorageMode::stokes_cached;
      state.objective_values.clear();
      for (std::size_t i = 0; i < objectives; ++i)
        {
          std::string name;
          double value = 0;
          input >> std::quoted(name) >> value;
          AssertThrow(input,
                      ExcMessage("Adjoint restart objective metadata ended unexpectedly."));
          state.objective_values.emplace(name, value);
        }
    }



    std::string
    RestartCheckpointManager::rank_payload_filename() const
    {
      return "rank_" + Utilities::int_to_string(rank, 5) + ".bin";
    }



    bool
    RestartCheckpointManager::collective_success(const bool local_success,
                                                 const std::string &stage) const
    {
      const unsigned int global_success = Utilities::MPI::min(local_success ? 1U : 0U,
                                                              communicator);
      const int barrier_status = MPI_Barrier(communicator);
      (void)stage;
      return global_success == 1 && barrier_status == MPI_SUCCESS;
    }



    void
    RestartCheckpointManager::prune_old_generations(const unsigned int newest) const
    {
      if (rank != 0 || newest <= 2)
        return;
      std::error_code error;
      std::filesystem::remove_all(checkpoint_directory(newest - 2), error);
      AssertThrow(!error, ExcMessage("Could not prune obsolete adjoint checkpoint: "
                                     + error.message()));
    }
  }

    template <int dim>
    void
    Manager<dim>::write_initial_temperature_gradient() const
    {
      if (time_dependent_workflow.initial_temperature_gradient.size() == 0)
        return;

      const unsigned int rank =
        Utilities::MPI::this_mpi_process(this->get_mpi_communicator());
      const unsigned int temperature_block_index =
        this->introspection().block_indices.temperature;

      const std::string filename =
        this->get_output_directory()
        + "initial_temperature_gradient_rank_"
        + Utilities::int_to_string(rank, 5)
        + ".txt";

      std::ofstream out(filename);
      out << "# global_dof_index\tgradient";
      for (unsigned int d = 0; d < dim; ++d)
        out << "\tx" << d;
      out << '\n';
      out << std::setprecision(16);

      (void)temperature_block_index;
      const IndexSet locally_owned_dofs =
        this->get_dof_handler().locally_owned_dofs();
      const std::vector<Point<dim>> unit_support_points =
        Utilities::get_unit_support_points(*this);
      const Quadrature<dim> support_quadrature(unit_support_points);
      FEValues<dim> fe_values(this->get_mapping(),
                              this->get_fe(),
                              support_quadrature,
                              update_quadrature_points);
      std::vector<types::global_dof_index> local_dof_indices(this->get_fe().dofs_per_cell);
      std::set<types::global_dof_index> written_dofs;

      for (const auto &cell : this->get_dof_handler().active_cell_iterators())
        if (cell->is_locally_owned())
          {
            fe_values.reinit(cell);
            cell->get_dof_indices(local_dof_indices);

            for (unsigned int i = 0; i < this->get_fe().dofs_per_cell; ++i)
              {
                const unsigned int component =
                  this->get_fe().system_to_component_index(i).first;
                if (this->introspection().component_masks.temperature[component] == false)
                  continue;

                const types::global_dof_index global_index = local_dof_indices[i];
                if (locally_owned_dofs.is_element(global_index) == false
                    || written_dofs.insert(global_index).second == false)
                  continue;

                out << global_index << '\t'
                    << time_dependent_workflow.initial_temperature_gradient(global_index);
                for (unsigned int d = 0; d < dim; ++d)
                  out << '\t' << fe_values.quadrature_point(i)[d];
                out << '\n';
              }
          }
    }



    template <int dim>
    void
    Manager<dim>::write_initial_composition_gradients() const
    {
      if (time_dependent_workflow.initial_composition_gradients.empty())
        return;

      const unsigned int rank =
        Utilities::MPI::this_mpi_process(this->get_mpi_communicator());
      const IndexSet locally_owned_dofs =
        this->get_dof_handler().locally_owned_dofs();
      const std::vector<Point<dim>> unit_support_points =
        Utilities::get_unit_support_points(*this);
      const Quadrature<dim> support_quadrature(unit_support_points);
      FEValues<dim> fe_values(this->get_mapping(),
                              this->get_fe(),
                              support_quadrature,
                              update_quadrature_points);
      std::vector<types::global_dof_index> local_dof_indices(this->get_fe().dofs_per_cell);

      for (unsigned int c = 0; c < time_dependent_workflow.initial_composition_gradients.size(); ++c)
        {
          const AdvectionField composition_field = AdvectionField::composition(c);
          const unsigned int composition_component =
            composition_field.component_index(this->introspection());
          const std::string filename =
            this->get_output_directory()
            + "initial_composition_gradient_"
            + this->introspection().name_for_compositional_index(c)
            + "_rank_"
            + Utilities::int_to_string(rank, 5)
            + ".txt";

          std::ofstream out(filename);
          out << "# global_dof_index\tgradient";
          for (unsigned int d = 0; d < dim; ++d)
            out << "\tx" << d;
          out << '\n';
          out << std::setprecision(16);

          std::set<types::global_dof_index> written_dofs;
          for (const auto &cell : this->get_dof_handler().active_cell_iterators())
            if (cell->is_locally_owned())
              {
                fe_values.reinit(cell);
                cell->get_dof_indices(local_dof_indices);

                for (unsigned int i = 0; i < this->get_fe().dofs_per_cell; ++i)
                  {
                    const unsigned int component =
                      this->get_fe().system_to_component_index(i).first;
                    if (component != composition_component)
                      continue;

                    const types::global_dof_index global_index = local_dof_indices[i];
                    if (locally_owned_dofs.is_element(global_index) == false
                        || written_dofs.insert(global_index).second == false)
                      continue;

                    out << global_index << '\t'
                        << time_dependent_workflow.initial_composition_gradients[c](global_index);
                    for (unsigned int d = 0; d < dim; ++d)
                      out << '\t' << fe_values.quadrature_point(i)[d];
                    out << '\n';
                  }
              }
        }
    }



    template <int dim>
    void
    Manager<dim>::add_stokes_field_transpose(const AdvectionField &advection_field,
                                             const ForwardStateView<dim> &snapshot,
                                             const LinearAlgebra::BlockVector &stokes_adjoint_rhs,
                                             LinearAlgebra::BlockVector &field_adjoint)
    {
      // At time n the quasi-static Stokes equation S_n(u_n,p_n;q_n)=0 depends
      // on the advected field q_n through rho and eta. Given a Stokes RHS seed
      // accumulated from later advection steps, solve
      //
      //   (partial S_n/partial[u,p])^T psi_n = bar([u,p]_n)
      //
      // and then add -(partial S_n/partial q_n)^T psi_n to bar(q_n). The
      // zero-velocity test avoids rebuilding an operator for an empty edge in
      // the reverse graph.
      const unsigned int velocity_block_index =
        this->introspection().block_indices.velocities;

      if (stokes_adjoint_rhs.block(velocity_block_index).l2_norm()
          <= std::numeric_limits<double>::min())
        return;

      ScopedSimulatorState<dim> simulator_state(*simulator);

      simulator->solution = snapshot.solution.get();
      simulator->old_solution = snapshot.old_solution.get();
      simulator->old_old_solution = snapshot.old_old_solution.get();
      simulator->current_linearization_point = snapshot.stokes_linearization_point.get();
      simulator->time = snapshot.time;
      simulator->time_step = snapshot.time_step;
      simulator->old_time_step = snapshot.old_time_step;
      simulator->timestep_number = snapshot.timestep_number;
      simulator->last_pressure_normalization_adjustment =
        snapshot.pressure_normalization_adjustment;

      // Matrix-free Stokes preconditioners must be built with the primal
      // constraint topology: homogeneous adjoint constraints leave some
      // eliminated mass-matrix diagonal entries at zero. This is also the
      // topology used by the uninterrupted forward path whose preconditioner
      // was previously reused by the adjoint solve.
      simulator->current_constraints.copy_from(snapshot.constraints);
      // This scalar is intentionally runtime-only. The normal forward Stokes
      // path computes it immediately before every assembly; a cold adjoint
      // resume skips that path and therefore retains the constructor's NaN.
      // Recompute it from this restored primal snapshot, rather than storing
      // any solver or matrix-free state in the checkpoint.
      simulator->pressure_scaling = simulator->compute_pressure_scaling_factor();
      simulator->rebuild_stokes_matrix = true;
      simulator->rebuild_stokes_preconditioner = true;
      simulator->assemble_stokes_system();

      // Preconditioner data may depend on constrained block diagonals and the
      // pressure scaling but not on the adjoint RHS. Build it once from the
      // restored primal topology before replacing constraints by their
      // homogeneous adjoint form.
      if (simulator->stokes_matrix_free)
        simulator->stokes_matrix_free->build_preconditioner();
      else
        simulator->build_stokes_preconditioner();

      BoundaryConditions<dim> boundary_conditions;
      boundary_conditions.build_stokes_adjoint_constraints(snapshot,
                                                           simulator->current_constraints);
      // Assemble the adjoint matrix/RHS under homogeneous constraints, but
      // retain the already initialized primal-topology preconditioner.
      simulator->rebuild_stokes_matrix = true;
      simulator->rebuild_stokes_preconditioner = false;
      simulator->assemble_stokes_system();

      LinearAlgebra::BlockVector stokes_adjoint_solution;
      solve_stokes_adjoint_from_rhs(stokes_adjoint_rhs,
                                    stokes_adjoint_solution);

      // The implicit-state solve has removed the [u,p] dependence. What
      // remains is the material-input VJP at fixed primal velocity, evaluated
      // for the selected temperature or composition component.
      add_stokes_rhs_field_transpose(advection_field,
                                     snapshot,
                                     stokes_adjoint_solution,
                                     field_adjoint);

    }



    template <int dim>
    void
    Manager<dim>::add_stokes_rhs_field_transpose(const AdvectionField &advection_field,
                                                 const ForwardStateView<dim> &snapshot,
                                                 const LinearAlgebra::BlockVector &stokes_test_vector,
                                                 LinearAlgebra::BlockVector &field_adjoint)
    {
      // For the incompressible weak residual
      //
      //   S(v)=integral [2 eta(q) eps(u):eps(v)-rho(q) g.v- p div(v)] dx,
      //
      // contraction with the Stokes adjoint psi seeds material outputs as
      //
      //   bar(rho)= +(g.psi),
      //   bar(eta)= -2 eps(u):eps(psi),
      //
      // under the b-Ax sign convention used by the reverse sweep. The
      // material-model transpose then maps these seeds to q.
      const unsigned int velocity_block_index =
        this->introspection().block_indices.velocities;

      if (stokes_test_vector.block(velocity_block_index).l2_norm()
          <= std::numeric_limits<double>::min())
        return;

      // This is an FE read of the Stokes adjoint. The solve exports an
      // owned-only vector so it can be accumulated safely; materialize its
      // locally-relevant view only at this read boundary.
      LinearAlgebra::BlockVector stokes_test_vector_read(
        this->introspection().index_sets.system_partitioning,
        this->introspection().index_sets.system_relevant_partitioning,
        this->get_mpi_communicator());
      stokes_test_vector_read = stokes_test_vector;

      ScopedSimulatorState<dim> simulator_state(*simulator);

      simulator->solution = snapshot.solution.get();
      simulator->old_solution = snapshot.old_solution.get();
      simulator->old_old_solution = snapshot.old_old_solution.get();
      simulator->current_linearization_point = snapshot.stokes_linearization_point.get();
      simulator->time = snapshot.time;
      simulator->time_step = snapshot.time_step;
      simulator->old_time_step = snapshot.old_time_step;
      simulator->timestep_number = snapshot.timestep_number;
      simulator->last_pressure_normalization_adjustment =
        snapshot.pressure_normalization_adjustment;

      const Quadrature<dim> &quadrature_formula =
        this->introspection().quadratures.velocities;
      const unsigned int dofs_per_cell = this->get_fe().dofs_per_cell;
      const FEValuesExtractors::Scalar field_extractor =
        advection_field.scalar_extractor(this->introspection());
      const unsigned int field_component =
        advection_field.component_index(this->introspection());

      FEValues<dim> fe_values(this->get_mapping(),
                              this->get_fe(),
                              quadrature_formula,
                              update_values |
                              update_gradients |
                              update_quadrature_points |
                              update_JxW_values);

      std::vector<Tensor<1,dim>> psi_velocity_values(quadrature_formula.size());
      std::vector<SymmetricTensor<2,dim>> forward_velocity_symmetric_gradients(quadrature_formula.size());
      std::vector<SymmetricTensor<2,dim>> psi_velocity_symmetric_gradients(quadrature_formula.size());
      Vector<double> local_field_adjoint(dofs_per_cell);
      std::vector<types::global_dof_index> local_dof_indices(dofs_per_cell);

      for (const auto &cell : this->get_dof_handler().active_cell_iterators())
        if (cell->is_locally_owned())
          {
            fe_values.reinit(cell);
            fe_values[this->introspection().extractors.velocities]
            .get_function_symmetric_gradients(simulator->solution,
                                              forward_velocity_symmetric_gradients);
            fe_values[this->introspection().extractors.velocities]
            .get_function_values(stokes_test_vector_read,
                                 psi_velocity_values);
            fe_values[this->introspection().extractors.velocities]
            .get_function_symmetric_gradients(stokes_test_vector_read,
                                              psi_velocity_symmetric_gradients);

            local_field_adjoint = 0.0;
            cell->get_dof_indices(local_dof_indices);

            MaterialModel::MaterialModelInputs<dim> in(fe_values,
                                                       cell,
                                                       this->introspection(),
                                                       simulator->current_linearization_point);
            MaterialModel::MaterialModelOutputs<dim> out(fe_values.n_quadrature_points,
                                                         this->n_compositional_fields());
            in.requested_properties =
              static_cast<MaterialModel::MaterialProperties::Property>(
                static_cast<int>(MaterialModel::MaterialProperties::equation_of_state_properties)
                | static_cast<int>(MaterialModel::MaterialProperties::viscosity));
            this->get_material_model().evaluate(in, out);

            MaterialModelOutputAdjoints<dim> bar_out(fe_values.n_quadrature_points);
            MaterialModelInputAdjoints<dim> bar_in(fe_values.n_quadrature_points,
                                                   this->n_compositional_fields());

            for (unsigned int q = 0; q < fe_values.n_quadrature_points; ++q)
              {
                // Include JxW in the material-output seed so the following VJP
                // is already a quadrature contribution. This is the exact
                // transpose of material outputs being sampled inside the weak
                // Stokes assembly, not a pointwise continuum derivative.
                const Tensor<1,dim> gravity =
                  this->get_gravity_model().gravity_vector(fe_values.quadrature_point(q));
                bar_out.densities[q] =
                  (gravity * psi_velocity_values[q]) * fe_values.JxW(q);
                bar_out.viscosities[q] =
                  -2.0
                  * (forward_velocity_symmetric_gradients[q]
                     * psi_velocity_symmetric_gradients[q])
                  * fe_values.JxW(q);
              }

            const MaterialModelAdjoint<dim> material_model_adjoint(*simulator);
            material_model_adjoint.apply_derivative_transpose(in,
                                                              out,
                                                              bar_out,
                                                              bar_in);

            for (const double pressure_adjoint : bar_in.pressure)
              AssertThrow(pressure_adjoint == 0.0,
                          ExcMessage("The Stokes material adjoint generated a non-zero pressure input "
                                     "adjoint, but the v1 Stokes-material scatter supports only "
                                     "temperature and composition states."));

            // Scatter the selected material input seed with the scalar FE
            // evaluation transpose: bar(q_i)+=bar(q(x_q)) phi_i(x_q).
            // Material dependence on other advected fields is handled by
            // separate calls, so each call selects exactly one component.
            for (unsigned int q = 0; q < fe_values.n_quadrature_points; ++q)
              {
                const double seed =
                  advection_field.is_temperature()
                  ? bar_in.temperature[q]
                  : bar_in.composition[q][advection_field.compositional_variable];
                for (unsigned int i = 0; i < dofs_per_cell; ++i)
                  if (this->get_fe().system_to_component_index(i).first == field_component)
                    local_field_adjoint(i) +=
                      seed
                      * fe_values[field_extractor].value(i, q);
              }

            this->get_current_constraints().distribute_local_to_global(local_field_adjoint,
                                                                       local_dof_indices,
                                                                       field_adjoint);
          }

      field_adjoint.compress(VectorOperation::add);

    }

  }

    template <int dim>
    void
    TimeDependentWorkflow<dim>::run(Manager<dim> &manager)
    {
      // The accepted forward trajectory defines residuals
      //   R_n(q_n,q_{n-1},q_{n-2},u_n)=0,  n=1,...,N,
      // plus a quasi-static Stokes residual S_n(u_n,T_n,C_n)=0. Starting from
      // terminal/running objective seeds, this workflow applies the discrete
      // Jacobian transpose in descending n. No continuous adjoint equation is
      // re-discretized here: every action reverses an implemented forward
      // algebraic dependency at its recorded time level.
      if (!manager.time_dependent_temperature_workflow_enabled())
        return;

      manager.validate_time_dependent_temperature_setup();

      const bool resuming_reverse = !manager.trajectory_manager.has_archive()
                                    && manager.has_pending_adjoint_restart();
      const bool restart_enabled = !manager.suppress_evaluation_side_effects;
      const bool lifecycle_cleanup_enabled = true;
      if (resuming_reverse)
        // Opening an archive reconstructs durable schedule/payload metadata;
        // replay windows remain transient and are rebuilt only when demanded
        // by a reverse step.
        manager.trajectory_manager.create_archive(manager.trajectory_manager.configuration().storage_mode,
                                                  manager.get_output_directory(),
                                                  true,
                                                  manager.trajectory_manager.configuration().records_per_segment);

      if (resuming_reverse && manager.trajectory_manager.configuration().anchor_spacing > 1)
        {
          manager.trajectory_manager.sparse_anchor_steps.clear();
          for (unsigned int record = 0; record < manager.trajectory_manager.archive().size(); ++record)
            if (manager.trajectory_manager.archive().has_record(record)
                && (record % manager.trajectory_manager.configuration().anchor_spacing == 0
                    || record + 1 == manager.trajectory_manager.archive().size()))
              manager.trajectory_manager.sparse_anchor_steps.push_back(record);
        }

      if (resuming_reverse)
        {
          // The normal forward loop reaches start_timestep() after a generic
          // restart. The adjoint-resume path deliberately skips that loop, so
          // reproduce the runtime-only initialization that checkpoint I/O does
          // not serialize: current constraints, system storage, and the GMG
          // hierarchy/level constraints.
          manager.report_mpi_adjoint_memory_checkpoint(
            "before initialize_current_linearization_point");
          manager.simulator->initialize_current_linearization_point();
          manager.report_mpi_adjoint_memory_checkpoint(
            "after initialize_current_linearization_point");
          manager.report_mpi_adjoint_memory_checkpoint("before compute_current_constraints");
          manager.simulator->compute_current_constraints();
          manager.report_mpi_adjoint_memory_checkpoint("after compute_current_constraints");
          if (manager.simulator->rebuild_sparsity_and_matrices)
            {
              manager.simulator->setup_system_matrix(
                manager.introspection().index_sets.system_partitioning);
              manager.simulator->setup_system_preconditioner(
                manager.introspection().index_sets.system_partitioning);
              manager.simulator->rebuild_sparsity_and_matrices = false;
            }
          if (manager.simulator->stokes_matrix_free)
            manager.simulator->stokes_matrix_free->setup_dofs();
          manager.simulator->rebuild_stokes_matrix = true;
          manager.simulator->rebuild_stokes_preconditioner = true;
          manager.get_pcout()
              << "   Adjoint cold-restart Stokes runtime reconstruction complete: "
              << "timestep = " << manager.simulator->timestep_number
              << ", time = " << manager.simulator->time
              << ", next reverse step pending." << std::endl;
        }

      AssertThrow(manager.trajectory_manager.has_archive() && manager.trajectory_manager.archive().size() > 0,
                  ExcMessage("The time-dependent temperature adjoint did not record any "
                             "accepted forward time steps. Check the end time and time-stepping setup."));

      manager.report_trajectory_statistics("forward archive complete");
      if (!resuming_reverse && manager.run_trajectory_stokes_recovery_check)
        manager.run_trajectory_stokes_recovery_diagnostic();
      if (!resuming_reverse && manager.run_trajectory_replay_check)
        manager.run_trajectory_replay_diagnostic();
      manager.trajectory_manager.reset_replay_state();
      // Reverse operations temporarily install historical states, constraints,
      // time stamps, and rebuild flags in Simulator. The guard makes the whole
      // sweep transactional with respect to the caller's forward state,
      // including exceptions from solvers, diagnostics, or checkpoint I/O.
      ScopedSimulatorState<dim> simulator_state(*manager.simulator);

      // This is the authoritative reverse state. Level zero is the seed at
      // next_reverse_step; levels one and two are its two pending predecessors.
      // The old N-sized history arrays are deliberately not retained.
      AdjointRestartState reverse_state;
      reverse_state.trajectory_storage_mode = manager.trajectory_manager.configuration().storage_mode;
      reverse_state.composition_history.resize(manager.introspection().n_compositional_fields);
      const auto initialize_vector = [&manager](LinearAlgebra::BlockVector &vector)
      {
        // Reverse histories are accumulation targets for constraint
        // distribution and transpose assembly. Keep them owned-only and
        // writable; temporary ghosted views belong solely to replay states
        // used for FE evaluation.
        vector.reinit(manager.introspection().index_sets.system_partitioning,
                      manager.get_mpi_communicator());
        vector = 0.0;
      };
      for (auto &vector : reverse_state.temperature_history)
        initialize_vector(vector);
      for (auto &vector : reverse_state.stokes_right_hand_side_history)
        initialize_vector(vector);
      for (auto &field_history : reverse_state.composition_history)
        for (auto &vector : field_history)
          initialize_vector(vector);

      auto &temperature_adjoints = reverse_state.temperature_history;
      auto &composition_adjoints = reverse_state.composition_history;
      auto &stokes_adjoint_right_hand_sides =
        reverse_state.stokes_right_hand_side_history;
      if (restart_enabled)
        manager.trajectory_manager.create_restart_checkpoint_manager();

      AdvectionAdjointCalculator<dim> advection_adjoint_calculator(*manager.simulator);
      Diagnostics::AdvectionOperatorChecks<dim> advection_operator_checks(
        advection_adjoint_calculator);
      advection_adjoint_calculator.set_run_preconditioner_comparison(
        manager.run_advection_preconditioner_comparison);
      if (manager.advection_preconditioner == "none")
        advection_adjoint_calculator.set_preconditioner(
          AdvectionAdjointCalculator<dim>::Preconditioner::none);
      else if (manager.advection_preconditioner == "forward ilu experimental")
        advection_adjoint_calculator.set_preconditioner(
          AdvectionAdjointCalculator<dim>::Preconditioner::forward_ilu_experimental);
      else
        {
          AssertThrow(manager.advection_preconditioner == "transpose ilu",
                      ExcMessage("Unknown adjoint advection preconditioner."));
          advection_adjoint_calculator.set_preconditioner(
            AdvectionAdjointCalculator<dim>::Preconditioner::transpose_ilu);
        }

      if (!resuming_reverse)
        {
          manager.report_mpi_adjoint_memory_checkpoint(
            "fresh reverse: initialize_current_linearization_point not invoked");
          // Evaluate terminal objective derivatives at the final accepted forward
          // state. A checkpoint at N is the state immediately before reverse step N.
          const ForwardStateView<dim> final_snapshot =
            manager.reconstruct_forward_snapshot(manager.trajectory_manager.archive().size() - 1);
          manager.simulator->solution = final_snapshot.solution.get();
          manager.simulator->old_solution = final_snapshot.old_solution.get();
          manager.simulator->old_old_solution = final_snapshot.old_old_solution.get();
          manager.simulator->current_linearization_point = final_snapshot.stokes_linearization_point.get();
          manager.simulator->time = final_snapshot.time;
          manager.simulator->time_step = final_snapshot.time_step;
          manager.simulator->old_time_step = final_snapshot.old_time_step;
          manager.simulator->timestep_number = final_snapshot.timestep_number;
          manager.simulator->last_pressure_normalization_adjustment =
            final_snapshot.pressure_normalization_adjustment;

          manager.report_mpi_adjoint_memory_checkpoint("before terminal objective assembly",
                                                       &final_snapshot.solution.get());
          reverse_state.objective_values =
            manager.objective_manager.evaluate_and_assemble_terminal_temperature_right_hand_side(temperature_adjoints[0]);
          // For J=Phi(q_N)+sum_n L_n(q_n), initialize lambda_N with
          // dPhi/dq_N. Running Stokes objectives were accumulated during the
          // forward pass because their scalar values require the accepted
          // physical state at each n; their derivatives are injected below at
          // the matching reverse step.
          for (const auto &value : this->running_stokes_objective_values)
            reverse_state.objective_values[value.first] += value.second;
          manager.report_mpi_adjoint_memory_checkpoint("after terminal temperature objective assembly",
                                                       &temperature_adjoints[0]);
          for (unsigned int c = 0; c < manager.introspection().n_compositional_fields; ++c)
            manager.objective_manager.evaluate_and_assemble_terminal_advection_field_right_hand_side(
              AdvectionField::composition(c),
              composition_adjoints[c][0]);

          if (manager.run_material_model_adjoint_vjp_diagnostic)
            manager.run_material_model_adjoint_vjp_test(final_snapshot);
          if (manager.run_direct_stokes_viscosity_derivative_diagnostic)
            manager.run_direct_stokes_viscosity_derivative_test(final_snapshot);

          if (manager.run_composition_stokes_rhs_transpose_diagnostic)
            {
              manager.run_composition_stokes_rhs_transpose_test(final_snapshot);
              if (final_snapshot.timestep_number <= 1)
                advection_operator_checks.velocity_transpose(
                  AdvectionField::temperature(),
                  final_snapshot,
                  temperature_adjoints[0],
                  manager.composition_stokes_rhs_transpose_diagnostic_epsilon);
              else
                manager.get_pcout() << "   Skipping temperature-advection velocity transpose diagnostic "
                                    << "on timestep "
                                    << final_snapshot.timestep_number
                                    << "; the current diagnostic is a one-step "
                                    << "u-star check and does not compare the BDF2 "
                                    << "history split." << std::endl;
            }
          if (manager.run_transposed_advection_constraint_diagnostic)
            {
              advection_operator_checks.transposed_constraints(
                AdvectionField::temperature(),
                final_snapshot,
                temperature_adjoints[0]);

              for (unsigned int c = 0; c < manager.introspection().n_compositional_fields; ++c)
                advection_operator_checks.transposed_constraints(
                  AdvectionField::composition(c),
                  final_snapshot,
                  composition_adjoints[c][0]);
            }
          if (manager.run_artificial_viscosity_coefficient_seed_diagnostic)
            {
              AssertThrow(manager.artificial_viscosity_coefficient_seed_diagnostic_epsilon > 0.0,
                          ExcMessage("Adjoint/artificial viscosity coefficient seed test epsilon must be positive."));

              advection_operator_checks.artificial_viscosity_coefficient_seed(
                AdvectionField::temperature(),
                final_snapshot,
                temperature_adjoints[0],
                manager.artificial_viscosity_coefficient_seed_diagnostic_epsilon);

              for (unsigned int c = 0; c < manager.introspection().n_compositional_fields; ++c)
                advection_operator_checks.artificial_viscosity_coefficient_seed(
                  AdvectionField::composition(c),
                  final_snapshot,
                  composition_adjoints[c][0],
                  manager.artificial_viscosity_coefficient_seed_diagnostic_epsilon);
            }
          if (manager.run_artificial_viscosity_estimator_transpose_diagnostic)
            {
              AssertThrow(manager.artificial_viscosity_estimator_transpose_diagnostic_epsilon > 0.0,
                          ExcMessage("Adjoint/artificial viscosity estimator transpose test epsilon must be positive."));

              advection_operator_checks.artificial_viscosity_estimator_transpose(
                AdvectionField::temperature(),
                final_snapshot,
                manager.artificial_viscosity_estimator_transpose_diagnostic_epsilon);

              for (unsigned int c = 0; c < manager.introspection().n_compositional_fields; ++c)
                advection_operator_checks.artificial_viscosity_estimator_transpose(
                  AdvectionField::composition(c),
                  final_snapshot,
                  manager.artificial_viscosity_estimator_transpose_diagnostic_epsilon);
            }
          if (manager.run_local_entropy_residual_transpose_diagnostic)
            {
              AssertThrow(manager.local_entropy_residual_transpose_diagnostic_epsilon > 0.0,
                          ExcMessage("Adjoint/local entropy residual transpose test epsilon must be positive."));

              advection_operator_checks.local_entropy_residual_transpose(
                AdvectionField::temperature(),
                final_snapshot,
                manager.local_entropy_residual_transpose_diagnostic_epsilon);

              for (unsigned int c = 0; c < manager.introspection().n_compositional_fields; ++c)
                advection_operator_checks.local_entropy_residual_transpose(
                  AdvectionField::composition(c),
                  final_snapshot,
                  manager.local_entropy_residual_transpose_diagnostic_epsilon);
            }
          if (manager.run_global_entropy_denominator_scalar_diagnostic)
            {
              AssertThrow(manager.global_entropy_denominator_scalar_diagnostic_epsilon > 0.0,
                          ExcMessage("Adjoint/global entropy denominator scalar test epsilon must be positive."));

              advection_operator_checks.global_entropy_denominator_scalar(
                AdvectionField::temperature(),
                final_snapshot,
                manager.global_entropy_denominator_scalar_diagnostic_epsilon);

              for (unsigned int c = 0; c < manager.introspection().n_compositional_fields; ++c)
                advection_operator_checks.global_entropy_denominator_scalar(
                  AdvectionField::composition(c),
                  final_snapshot,
                  manager.global_entropy_denominator_scalar_diagnostic_epsilon);
            }
          if (manager.run_global_entropy_variation_transpose_diagnostic)
            {
              AssertThrow(manager.global_entropy_variation_transpose_diagnostic_epsilon > 0.0,
                          ExcMessage("Adjoint/global entropy variation transpose test epsilon must be positive."));

              advection_operator_checks.global_entropy_variation_transpose(
                AdvectionField::temperature(),
                final_snapshot,
                manager.global_entropy_variation_transpose_diagnostic_epsilon);

              for (unsigned int c = 0; c < manager.introspection().n_compositional_fields; ++c)
                advection_operator_checks.global_entropy_variation_transpose(
                  AdvectionField::composition(c),
                  final_snapshot,
                  manager.global_entropy_variation_transpose_diagnostic_epsilon);
            }

          reverse_state.next_reverse_step = manager.trajectory_manager.archive().size();
          // A checkpoint stores the semantic three-level adjoint window before
          // executing step N. Publishing this initial generation makes a
          // restart independent of whether the process fails before its first
          // transpose solve.
          manager.trajectory_manager.set_restart_trajectory_identity(reverse_state);
          if (restart_enabled)
            {
              TimerOutput::Scope checkpoint_timer(manager.get_computing_timer(),
                                                  "Adjoint restart checkpoint write");
              const CheckpointWriteResult checkpoint_write =
                manager.trajectory_manager.restart_checkpoint().write(reverse_state);
              if (!checkpoint_write.committed)
                {
                  manager.get_pcout()
                      << "   Adjoint checkpoint transaction failed collectively before "
                      << "its first reverse step; "
                      << (checkpoint_write.latest_published
                          ? "the new generation was already published."
                          : "no new generation was published.")
                      << std::endl;
                  if (Utilities::MPI::n_mpi_processes(manager.get_mpi_communicator()) > 1)
                    throw QuietException();
                  AssertThrow(false,
                              ExcMessage("Adjoint checkpoint transaction failed before "
                                         "the first reverse step."));
                }
              manager.get_pcout() << "   Adjoint restart checkpoint committed before reverse step "
                                  << reverse_state.next_reverse_step << " ("
                                  << checkpoint_write.local_bytes << " rank-local bytes)." << std::endl;
            }
        }
      else
        {
          // Restore lambda_n, lambda_{n-1}, lambda_{n-2} and the corresponding
          // Stokes RHS histories exactly as they were committed. Forward replay
          // caches are intentionally absent from the checkpoint and are
          // reconstructed from the durable anchor named by its identity data.
          TimerOutput::Scope restore_timer(manager.get_computing_timer(),
                                           "Adjoint restart checkpoint restore");
          const std::size_t restore_bytes =
            manager.trajectory_manager.restart_checkpoint().restore(reverse_state);
          AssertThrow(reverse_state.trajectory_storage_mode == manager.trajectory_manager.configuration().storage_mode,
                      ExcMessage("The committed adjoint restart checkpoint has a "
                                 "different trajectory storage mode."));
          AssertThrow(reverse_state.next_reverse_step <= manager.trajectory_manager.archive().size(),
                      ExcMessage("Adjoint restart next_reverse_step is outside the "
                                 "trajectory archive."));
          manager.trajectory_manager.validate_sparse_restart_state(reverse_state);
          manager.get_pcout() << "   Restored adjoint restart checkpoint before reverse step "
                              << reverse_state.next_reverse_step << " ("
                              << restore_bytes << " rank-local bytes)." << std::endl;
        }

      // A committed safe-incremental frontier is authoritative even if the
      // previous process died before or during removal. Finish the idempotent
      // cleanup before rebuilding the disposable sparse replay window.
      if (resuming_reverse
          && manager.trajectory_manager.configuration().cleanup_policy == TrajectoryCleanupPolicy::safe_incremental
          && reverse_state.cleanup_frontier != numbers::invalid_unsigned_int)
        {
          AssertThrow(reverse_state.cleanup_frontier <= manager.trajectory_manager.archive().size(),
                      ExcMessage("Adjoint restart cleanup frontier is outside the "
                                 "trajectory archive."));
          manager.cleanup_trajectory_after_checkpoint(reverse_state);
        }

      if (resuming_reverse && manager.trajectory_manager.configuration().anchor_spacing > 1
          && reverse_state.next_reverse_step > 0)
        {
          const unsigned int anchor =
            manager.trajectory_manager.required_anchor_for_reverse_step(reverse_state.next_reverse_step);
          manager.get_pcout() << "ASPECT_ADJOINT_SPARSE_RESTART next_reverse_step="
                              << reverse_state.next_reverse_step
                              << " required_lower_anchor=" << anchor
                              << " action=replay_disposable_interval" << std::endl;
          const unsigned int replayed_before = manager.trajectory_manager.sparse_replay_transition_count;
          manager.materialize_forward_replay_window(reverse_state.next_reverse_step - 1);
          manager.trajectory_manager.sparse_restart_replay_transition_count +=
            manager.trajectory_manager.sparse_replay_transition_count - replayed_before;
        }

      manager.objective_results.clear();
      for (const auto &objective_value : reverse_state.objective_values)
        {
          auto result = std::make_unique<ObjectiveResult>();
          result->objective_name = objective_value.first;
          result->value = objective_value.second;
          result->right_hand_side.reinit(
            manager.introspection().index_sets.system_partitioning,
            manager.get_mpi_communicator());
          result->right_hand_side = 0.0;
          manager.objective_results.push_back(std::move(result));

          manager.get_pcout() << "   Time-dependent terminal objective '"
                              << objective_value.first
                              << "' = "
                              << std::setprecision(16)
                              << objective_value.second
                              << std::setprecision(6)
                              << std::endl;
        }

      TimerOutput::Scope reverse_timer(manager.get_computing_timer(),
                                       "Adjoint trajectory reverse sweep");
      while (reverse_state.next_reverse_step > 0)
        {
          // At loop entry, history slot 0 contains all seeds of q_n. Applying
          // (dR_n/dq_{n-k})^T accumulates into slots k for k=1,2. The Stokes
          // RHS history follows the same convention because velocity at an
          // older level is eliminated only after all later residuals that use
          // it have contributed.
          const unsigned int reverse_index = reverse_state.next_reverse_step;
          const unsigned int snapshot_index = reverse_index - 1;
          const ForwardStateView<dim> snapshot =
            manager.reconstruct_forward_snapshot(snapshot_index);

          if (!this->running_stokes_objective_values.empty())
            // Add dL_n/du_n to the current Stokes adjoint RHS before eliminating
            // u_n. Objective dispatch is keyed by the recorded timestep, not
            // by loop position, which is important after restart and replay.
            manager.objective_manager.assemble_running_stokes_adjoint_right_hand_side(
              snapshot.timestep_number, stokes_adjoint_right_hand_sides[0]);

          // Deliberately reproduce the runtime portion of the adjoint-resume
          // path from a clean matrix-free state. The following first Stokes
          // transpose call performs the same primal-topology assembly and
          // build_preconditioner() call as an actual cold resume. Keep this
          // narrow experiment separate from the normal resume path: it tells
          // us whether GMG can rebuild at all (A), or whether resume lacks
          // another dependency (B).
          const bool run_cold_rebuild_diagnostic_here =
            manager.run_matrix_free_cold_rebuild_diagnostic
            && reverse_index == manager.trajectory_manager.archive().size();
          if (run_cold_rebuild_diagnostic_here)
            {
              AssertThrow(manager.simulator->stokes_matrix_free != nullptr,
                          ExcMessage("The matrix-free cold rebuild diagnostic "
                                     "requires the block GMG Stokes solver."));
              manager.get_pcout()
                  << "   Adjoint matrix-free cold-rebuild diagnostic: resetting "
                  << "GMG runtime state before reverse step " << reverse_index
                  << "." << std::endl;
              manager.simulator->stokes_matrix_free->setup_dofs();
              manager.simulator->rebuild_stokes_matrix = true;
              manager.simulator->rebuild_stokes_preconditioner = true;
            }

          // reverse_index names the adjoint seed at the current state q_n.
          // snapshot_index names the forward residual that advanced the model
          // from older states to q_n. Applying the transpose of that residual
          // moves sensitivities into q_{n-1} and possibly q_{n-2}.
          if (snapshot.timestep_number == 0)
            {
              // The first recorded state has no previous accepted advection
              // residual to invert. Only the Stokes dependence can still feed
              // back to field variables, and the remaining terminal/current
              // seed is simply part of the initial-condition gradient.
              manager.add_stokes_field_transpose(AdvectionField::temperature(),
                                                 snapshot,
                                                 stokes_adjoint_right_hand_sides[0],
                                                 temperature_adjoints[1]);
              temperature_adjoints[1] += temperature_adjoints[0];

              for (unsigned int c = 0; c < manager.introspection().n_compositional_fields; ++c)
                {
                  manager.add_stokes_field_transpose(AdvectionField::composition(c),
                                                     snapshot,
                                                     stokes_adjoint_right_hand_sides[0],
                                                     composition_adjoints[c][1]);
                  composition_adjoints[c][1] += composition_adjoints[c][0];
                }
            }
          else
            {
              // A Stokes RHS already accumulated at q_n affects temperature and
              // composition through the Stokes operator assembled at q_n. Apply
              // this before propagating the advection residual so that all seeds
              // at q_n are present in temperature_adjoints[reverse_index] and
              // composition_adjoints[*][reverse_index].
              if (reverse_index == manager.trajectory_manager.archive().size())
                manager.report_mpi_adjoint_memory_checkpoint(
                  "before first reverse Stokes setup", &snapshot.solution.get());
              manager.add_stokes_field_transpose(AdvectionField::temperature(),
                                                 snapshot,
                                                 stokes_adjoint_right_hand_sides[0],
                                                 temperature_adjoints[0]);
              if (reverse_index == manager.trajectory_manager.archive().size())
                manager.report_mpi_adjoint_memory_checkpoint(
                  "after first reverse Stokes setup", &snapshot.solution.get());
              if (run_cold_rebuild_diagnostic_here)
                manager.get_pcout()
                    << "   Adjoint matrix-free cold-rebuild diagnostic: first "
                    << "reverse Stokes preconditioner rebuild completed." << std::endl;
              for (unsigned int c = 0; c < manager.introspection().n_compositional_fields; ++c)
                manager.add_stokes_field_transpose(AdvectionField::composition(c),
                                                   snapshot,
                                                   stokes_adjoint_right_hand_sides[0],
                                                   composition_adjoints[c][0]);

              std::vector<LinearAlgebra::BlockVector> previous_composition_adjoint_slice;
              std::vector<LinearAlgebra::BlockVector> previous_previous_composition_adjoint_slice;
              previous_composition_adjoint_slice.reserve(manager.introspection().n_compositional_fields);
              previous_previous_composition_adjoint_slice.reserve(manager.introspection().n_compositional_fields);
              // The temperature material-prefactor transpose can modify all
              // composition adjoints. Build per-time-level slices so the helper can
              // treat them as a compact vector over composition fields, then copy
              // them back into the time-major storage below.
              for (unsigned int c = 0; c < manager.introspection().n_compositional_fields; ++c)
                {
                  previous_composition_adjoint_slice.push_back(composition_adjoints[c][1]);
                  if (snapshot_index > 0)
                    previous_previous_composition_adjoint_slice.push_back(composition_adjoints[c][2]);
                }

              advection_adjoint_calculator.propagate_temperature_adjoint(
                snapshot,
                temperature_adjoints[0],
                temperature_adjoints[1],
                snapshot_index > 0 ? &temperature_adjoints[2] : nullptr,
                stokes_adjoint_right_hand_sides[1],
                snapshot_index > 0 ? &stokes_adjoint_right_hand_sides[2] : nullptr,
                manager.introspection().n_compositional_fields > 0 ? &previous_composition_adjoint_slice : nullptr,
                (manager.introspection().n_compositional_fields > 0 && snapshot_index > 0)
                ? &previous_previous_composition_adjoint_slice
                : nullptr);

              // Algebraically this call applies the temperature row of
              // R_n^T. For BDF2 it distributes mass-term sensitivity with the
              // recorded variable-step coefficients to both q_{n-1} and
              // q_{n-2}; it also reverses advection velocity, diffusion,
              // material prefactors, and the active artificial-viscosity map.

              // Scatter the modified slices back into the persistent adjoint
              // history after propagate_temperature_adjoint() has added BDF,
              // velocity, artificial-viscosity, and material-prefactor transpose
              // contributions.
              for (unsigned int c = 0; c < manager.introspection().n_compositional_fields; ++c)
                {
                  composition_adjoints[c][1] = previous_composition_adjoint_slice[c];
                  if (snapshot_index > 0)
                    composition_adjoints[c][2] = previous_previous_composition_adjoint_slice[c];
                }

              for (unsigned int c = 0; c < manager.introspection().n_compositional_fields; ++c)
                advection_adjoint_calculator.propagate_advection_field_adjoint(
                  AdvectionField::composition(c),
                  snapshot,
                  composition_adjoints[c][0],
                  composition_adjoints[c][1],
                  snapshot_index > 0 ? &composition_adjoints[c][2] : nullptr,
                  stokes_adjoint_right_hand_sides[1],
                  snapshot_index > 0 ? &stokes_adjoint_right_hand_sides[2] : nullptr);
            }

          // Commit the state immediately before the next step. Moving the
          // three-level window happens before the checkpoint so its semantic
          // next_reverse_step is exact and unambiguous.
          --reverse_state.next_reverse_step;
          if (reverse_state.next_reverse_step > 0)
            {
              // Rotate the fixed semantic window after all contributions from
              // R_n have landed:
              //   [bar q_n,bar q_{n-1},bar q_{n-2}]
              //       <- [bar q_{n-1},bar q_{n-2},0].
              // Assignment rather than deque growth keeps restart storage and
              // working memory independent of the number of time steps.
              temperature_adjoints[0] = temperature_adjoints[1];
              temperature_adjoints[1] = temperature_adjoints[2];
              temperature_adjoints[2] = 0.0;
              stokes_adjoint_right_hand_sides[0] = stokes_adjoint_right_hand_sides[1];
              stokes_adjoint_right_hand_sides[1] = stokes_adjoint_right_hand_sides[2];
              stokes_adjoint_right_hand_sides[2] = 0.0;
              for (auto &field_history : composition_adjoints)
                {
                  field_history[0] = field_history[1];
                  field_history[1] = field_history[2];
                  field_history[2] = 0.0;
                }
            }

          if (restart_enabled && reverse_state.next_reverse_step > 0
              && reverse_state.next_reverse_step % manager.trajectory_manager.configuration().checkpoint_interval == 0)
            {
              // The transaction order is checkpoint -> optional cleanup. Thus
              // the durable checkpoint always names every anchor it needs. If
              // cleanup fails midway, resume repeats the idempotent cleanup
              // from cleanup_frontier without losing a recoverable state.
              manager.inject_lifecycle_failure(Diagnostics::LifecycleFailurePoint::before_checkpoint_commit,
                                               reverse_state.next_reverse_step);
              if (manager.trajectory_manager.configuration().cleanup_policy
                  == TrajectoryCleanupPolicy::safe_incremental)
                // Persist the target before deleting anything. A resume can
                // therefore repeat a partially completed deletion idempotently.
                reverse_state.cleanup_frontier = reverse_state.next_reverse_step;
              manager.trajectory_manager.set_restart_trajectory_identity(reverse_state);
              TimerOutput::Scope checkpoint_timer(manager.get_computing_timer(),
                                                  "Adjoint restart checkpoint write");
              const bool inject_checkpoint_transaction =
                reverse_state.next_reverse_step
                == manager.checkpoint_transaction_failure_next_reverse_step;
              const CheckpointWriteResult checkpoint_write =
                manager.trajectory_manager.restart_checkpoint().write(
                  reverse_state,
              {
                inject_checkpoint_transaction
                ? manager.checkpoint_transaction_failure_injection
                : Diagnostics::CheckpointFailurePoint::none,
                manager.checkpoint_transaction_failure_rank
              });
              if (!checkpoint_write.committed)
                {
                  manager.get_pcout()
                      << "   Adjoint checkpoint transaction failed collectively before "
                      << "reverse step " << reverse_state.next_reverse_step
                      << "; "
                      << (checkpoint_write.latest_published
                          ? "the new generation was already published."
                          : "no new generation was published.")
                      << std::endl;
                  if (Utilities::MPI::n_mpi_processes(manager.get_mpi_communicator()) > 1)
                    throw QuietException();
                  AssertThrow(false,
                              ExcMessage("Adjoint checkpoint transaction failed before "
                                         "reverse step "
                                         + Utilities::int_to_string(
                                           reverse_state.next_reverse_step)
                                         + "."));
                }
              manager.get_pcout() << "   Adjoint restart checkpoint committed before reverse step "
                                  << reverse_state.next_reverse_step << "." << std::endl;
              manager.inject_lifecycle_failure(
                Diagnostics::LifecycleFailurePoint::after_checkpoint_commit_before_cleanup,
                reverse_state.next_reverse_step);
              if (manager.trajectory_manager.configuration().cleanup_policy
                  == TrajectoryCleanupPolicy::safe_incremental)
                manager.cleanup_trajectory_after_checkpoint(reverse_state);
              manager.inject_lifecycle_failure(Diagnostics::LifecycleFailurePoint::after_cleanup,
                                               reverse_state.next_reverse_step);
              if (reverse_state.next_reverse_step == manager.stop_after_adjoint_checkpoint)
                {
                  manager.get_pcout() << "   Controlled stop after committed adjoint checkpoint "
                                      << "before reverse step "
                                      << reverse_state.next_reverse_step
                                      << ". Restart with 'Stop after adjoint checkpoint = -1' "
                                      << "to continue." << std::endl;
                  // This is an expected M2 control-flow exit, not a scientific
                  // error. Throwing a deal.II assertion independently on every
                  // MPI rank causes its verbose exception formatter to grow
                  // unboundedly on macOS. QuietException gives all ranks the
                  // same non-zero termination path after the checkpoint has
                  // already been published.
                  if (Utilities::MPI::n_mpi_processes(manager.get_mpi_communicator()) > 1)
                    throw QuietException();
                  AssertThrow(false,
                              ExcMessage("Controlled stop after committed adjoint checkpoint "
                                         "before reverse step "
                                         + Utilities::int_to_string(reverse_state.next_reverse_step)
                                         + ". Restart with 'Stop after adjoint checkpoint = -1' "
                                         "to continue."));
                }
            }
        }

      // Any Stokes RHS left at the initial time level has no earlier
      // advection step to absorb it, so solve/scatter it directly into the
      // initial field gradients.
      if (stokes_adjoint_right_hand_sides[1].l2_norm()
          > std::numeric_limits<double>::min())
        {
          manager.add_stokes_field_transpose(AdvectionField::temperature(),
                                             manager.reconstruct_forward_snapshot(0),
                                             stokes_adjoint_right_hand_sides[1],
                                             temperature_adjoints[1]);
          for (unsigned int c = 0; c < manager.introspection().n_compositional_fields; ++c)
            manager.add_stokes_field_transpose(AdvectionField::composition(c),
                                               manager.reconstruct_forward_snapshot(0),
                                               stokes_adjoint_right_hand_sides[1],
                                               composition_adjoints[c][1]);
        }

      this->initial_temperature_gradient = temperature_adjoints[1];
      // After step zero there is no earlier residual. Slot 1 is therefore the
      // covector with respect to the initial field itself, dJ/dT_0 (and
      // analogously dJ/dC_0), in the full constrained FE space.
      this->initial_composition_gradients.clear();
      this->initial_composition_gradients.reserve(composition_adjoints.size());
      for (const auto &field_adjoints : composition_adjoints)
        this->initial_composition_gradients.push_back(field_adjoints[1]);

      manager.get_pcout() << "   Time-dependent temperature adjoint recorded "
                          << manager.trajectory_manager.archive().size()
                          << " accepted forward trajectory records." << std::endl;
      if (manager.trajectory_manager.configuration().anchor_spacing > 1)
        manager.get_pcout() << "ASPECT_ADJOINT_SPARSE_REPLAY_SUMMARY anchors_spacing="
                            << manager.trajectory_manager.configuration().anchor_spacing
                            << " schedule_count=" << manager.trajectory_manager.archive().size()
                            << " anchor_count=" << manager.trajectory_manager.sparse_anchor_steps.size()
                            << " anchor_payload_state_count="
                            << manager.trajectory_manager.archive().payload_record_count()
                            << " anchor_load_count=" << manager.trajectory_manager.sparse_anchor_load_count
                            << " interval_replay_count=" << manager.trajectory_manager.sparse_interval_replay_count
                            << " replayed_transition_count=" << manager.trajectory_manager.sparse_replay_transition_count
                            << " restart_induced_extra_replay_transitions="
                            << manager.trajectory_manager.sparse_restart_replay_transition_count
                            << " peak_window_state_count=" << manager.trajectory_manager.sparse_peak_window_state_count
                            << " max_state_relative_difference=" << manager.trajectory_manager.sparse_max_state_relative_difference
                            << " max_velocity_relative_difference=" << manager.trajectory_manager.sparse_max_velocity_relative_difference
                            << " max_gauge_pressure_relative_difference="
                            << manager.trajectory_manager.sparse_max_pressure_relative_difference << std::endl;
      manager.get_pcout() << "   Completed reverse temperature-advection transpose sweep "
                          << "and wrote the initial-temperature gradient." << std::endl;

      if (!manager.suppress_evaluation_side_effects)
        {
          manager.write_initial_temperature_gradient();
          manager.write_initial_composition_gradients();
        }

      manager.report_trajectory_statistics("reverse complete");

      // A normal return means all final scientific outputs above are durable.
      // Do not place this in a destructor: an exception must leave both the
      // archive and any committed reverse checkpoint available for diagnosis
      // or restart. Incremental reclamation is performed only after a future
      // committed reverse checkpoint; at completion all remaining records are
      // unconditionally obsolete.
      if (lifecycle_cleanup_enabled
          && manager.trajectory_manager.configuration().cleanup_policy != TrajectoryCleanupPolicy::keep)
        {
          TimerOutput::Scope cleanup_timer(manager.get_computing_timer(),
                                           "Adjoint trajectory cleanup");
          const TrajectoryStorageStatistics storage_before_cleanup =
            manager.trajectory_manager.archive().storage_statistics();
          manager.trajectory_manager.archive().remove_archive();
          manager.get_pcout() << "   Adjoint trajectory cleanup reclaimed "
                              << Utilities::MPI::sum(
                                static_cast<unsigned long long>(storage_before_cleanup.total_bytes()),
                                manager.get_mpi_communicator())
                              << " bytes after successful reverse completion."
                              << std::endl;
        }

      // simulator_state restores the forward state even if the reverse sweep,
      // checkpoint transaction, or cleanup path throws.
    }

  }

    template <int dim>
    void
    TrajectoryManager<dim>::parse_parameters(ParameterHandler &prm)
    {
      prm.enter_subsection("Adjoint");
      prm.enter_subsection("Time dependent");
      {
        const std::string storage_mode = prm.get("Trajectory storage mode");
        AssertThrow(storage_mode == "minimal" || storage_mode == "stokes cached",
                    ExcMessage("Unknown adjoint trajectory storage mode."));
        parsed_configuration.storage_mode = storage_mode == "minimal"
                                            ? TrajectoryStorageMode::minimal
                                            : TrajectoryStorageMode::stokes_cached;

        const std::string cleanup_policy = prm.get("Trajectory cleanup policy");
        AssertThrow(cleanup_policy == "safe incremental" || cleanup_policy == "on success"
                    || cleanup_policy == "keep",
                    ExcMessage("Unknown adjoint trajectory cleanup policy."));
        parsed_configuration.cleanup_policy =
          cleanup_policy == "safe incremental" ? TrajectoryCleanupPolicy::safe_incremental
          : cleanup_policy == "on success" ? TrajectoryCleanupPolicy::on_success
          : TrajectoryCleanupPolicy::keep;

        parsed_configuration.records_per_segment =
          prm.get_integer("Trajectory records per segment");
        parsed_configuration.anchor_spacing = prm.get_integer("Trajectory anchor spacing");
        parsed_configuration.checkpoint_interval =
          prm.get_integer("Adjoint checkpoint interval");
      }
      prm.leave_subsection();
      prm.leave_subsection();
    }



    template <int dim>
    const typename TrajectoryManager<dim>::Configuration &
    TrajectoryManager<dim>::configuration() const
    {
      return parsed_configuration;
    }



    template <int dim>
    void
    TrajectoryManager<dim>::record_forward_snapshot(Manager<dim> &manager,
                                                    const double outgoing_time_step,
                                                    const bool has_outgoing_step)
    {
      // Recording is attached to accepted forward states only. A logical
      // record contains the complete time-grid metadata; its physical payload
      // may be dense or may be one member of a sparse three-state anchor
      // package used to seed deterministic replay.
      if (!manager.time_dependent_temperature_workflow_enabled())
        return;

      manager.validate_time_dependent_temperature_setup();

      if (!has_archive())
        {
          // Archive creation also starts a new objective accumulation epoch.
          // The optional dense reference is diagnostic-only and never serves
          // the production reverse sweep or restart path.
          create_archive(parsed_configuration.storage_mode,
                         manager.get_output_directory(),
                         false,
                         parsed_configuration.records_per_segment);
          sparse_anchor_steps.clear();
          if (parsed_configuration.anchor_spacing > 1
              && manager.run_sparse_trajectory_replay_check)
            create_reference_archive(
              TrajectoryStorageMode::stokes_cached,
              manager.get_output_directory() + "sparse_replay_reference/",
              parsed_configuration.records_per_segment);
          manager.time_dependent_workflow.running_stokes_objective_values.clear();
        }

      archive().set_append_failure_injection_for_testing(
      {
        manager.simulator->timestep_number == manager.trajectory_append_failure_step
        ? manager.trajectory_append_failure_injection
        : Diagnostics::AppendFailurePoint::none,
        manager.trajectory_append_failure_rank
      });

      TimerOutput::Scope write_timer(manager.get_computing_timer(),
                                     "Adjoint trajectory write");
      const ReplayStepMetadata metadata
      {
        manager.simulator->timestep_number,
                          manager.simulator->time,
                          manager.simulator->time_step,
                          manager.simulator->old_time_step,
                          outgoing_time_step,
                          has_outgoing_step
      };
      // Running Stokes objectives are quadrature-in-time contributions already
      // associated with this accepted step by the objective plugin. Store only
      // their scalar sum here; their state seeds are reconstructed again at
      // the corresponding reverse step.
      const std::map<std::string, double> running_values =
        manager.objective_manager.evaluate_running_stokes_objectives(
          manager.simulator->timestep_number);
      for (const auto &value : running_values)
        manager.time_dependent_workflow.running_stokes_objective_values[value.first]
        += value.second;

      if (parsed_configuration.anchor_spacing == 1)
        // Dense storage commits one payload per logical state.
        archive().store(metadata,
                        manager.simulator->solution,
                        manager.introspection().block_indices.temperature,
                        manager.introspection().block_indices.compositional_fields,
                        manager.introspection().block_indices.velocities,
                        manager.introspection().block_indices.pressure);
      else
        {
          // Sparse storage always commits the dense schedule but only persists
          // payloads at anchor packages. The three cached states
          // [q_a,q_{a-1},q_{a-2}] are sufficient to restart BDF2 replay from
          // anchor a without reaching into a preceding interval.
          archive().append_schedule(metadata);
          terminal_anchor_history.clear();
          terminal_anchor_history.push_back({metadata.step, manager.simulator->solution});
          if (metadata.step > 0)
            terminal_anchor_history.push_back({metadata.step - 1, manager.simulator->old_solution});
          if (metadata.step > 1)
            terminal_anchor_history.push_back({metadata.step - 2, manager.simulator->old_old_solution});
          if (metadata.step == 0 || metadata.step % parsed_configuration.anchor_spacing == 0)
            store_sparse_anchor_package(manager, metadata.step);
          if (has_reference_archive())
            reference_archive().store(metadata,
                                      manager.simulator->solution,
                                      manager.introspection().block_indices.temperature,
                                      manager.introspection().block_indices.compositional_fields,
                                      manager.introspection().block_indices.velocities,
                                      manager.introspection().block_indices.pressure);
        }
    }



    template <int dim>
    void
    TrajectoryManager<dim>::mark_forward_snapshot_terminal(Manager<dim> &manager)
    {
      if (!manager.time_dependent_temperature_workflow_enabled())
        return;

      AssertThrow(has_archive(), ExcInternalError());
      // The last state may not fall on the periodic anchor grid. Persist its
      // history package explicitly before declaring that its schedule record
      // has no outgoing step; otherwise the first reverse window could lack a
      // replay starting point.
      if (parsed_configuration.anchor_spacing > 1)
        store_sparse_anchor_package(manager, archive().size() - 1);
      archive().mark_last_record_terminal();
      if (has_reference_archive())
        reference_archive().mark_last_record_terminal();
    }



    template <int dim>
    void
    TrajectoryManager<dim>::store_sparse_anchor_package(Manager<dim> &manager,
                                                        const unsigned int anchor)
    {
      AssertThrow(has_archive(), ExcInternalError());
      AssertThrow(!terminal_anchor_history.empty()
                  && terminal_anchor_history.front().record == anchor,
                  ExcMessage("Sparse anchor history no longer matches its logical state."));
      // Store every missing member of the BDF history package idempotently.
      // Neighboring anchors may share records, and restart can repeat this
      // method after an interrupted append without duplicating payloads.
      const bool new_anchor = !archive().has_record(anchor);
      for (const ReconstructedTrajectoryState &state : terminal_anchor_history)
        if (!archive().has_record(state.record))
          archive().store_payload(
            state.record,
            state.state,
            manager.introspection().block_indices.temperature,
            manager.introspection().block_indices.compositional_fields,
            manager.introspection().block_indices.velocities,
            manager.introspection().block_indices.pressure);

      if (new_anchor)
        {
          sparse_anchor_steps.push_back(anchor);
          manager.get_pcout() << "ASPECT_ADJOINT_SPARSE_ANCHOR step=" << anchor
                              << " package_begin=" << terminal_anchor_history.back().record
                              << " package_end=" << terminal_anchor_history.front().record
                              << " stored_physical_states="
                              << archive().payload_record_count() << std::endl;
        }
    }



    template <int dim>
    void
    TrajectoryManager<dim>::reset()
    {
      restart_manager.reset();
      dense_reference_archive.reset();
      primary_archive.reset();
      reset_replay_state();
      terminal_anchor_history.clear();
      sparse_anchor_steps.clear();
    }



    template <int dim>
    void
    TrajectoryManager<dim>::reset_replay_state()
    {
      // These caches and counters describe one reverse execution only. The
      // durable archive and parsed configuration survive separately; clearing
      // them here would conflate transient replay memory with trajectory
      // ownership.
      reverse_state_cache.clear();
      forward_replay_windows.clear();
      stokes_recovery_solves = 0;
      sparse_replay_transition_count = 0;
      sparse_restart_replay_transition_count = 0;
      sparse_anchor_load_count = 0;
      sparse_interval_replay_count = 0;
      sparse_peak_window_state_count = 0;
      sparse_max_state_relative_difference = 0.0;
      sparse_max_velocity_relative_difference = 0.0;
      sparse_max_pressure_relative_difference = 0.0;
    }



    template <int dim>
    void
    TrajectoryManager<dim>::create_archive(const TrajectoryStorageMode mode,
                                           const std::string &output_directory,
                                           const bool open_existing,
                                           const unsigned int records_per_segment)
    {
      // Opening/creating a primary archive invalidates any checkpoint manager
      // bound to the previous archive directory. Recreate it explicitly only
      // after the trajectory identity is known.
      primary_archive = std::make_unique<TrajectoryStore>(
                          mode, output_directory, this->get_mpi_communicator(), open_existing,
                          records_per_segment);
      restart_manager.reset();
    }



    template <int dim>
    void
    TrajectoryManager<dim>::create_reference_archive(
      const TrajectoryStorageMode mode,
      const std::string &output_directory,
      const unsigned int records_per_segment)
    {
      dense_reference_archive = std::make_unique<TrajectoryStore>(
                                  mode, output_directory, this->get_mpi_communicator(), false,
                                  records_per_segment);
    }



    template <int dim>
    bool
    TrajectoryManager<dim>::has_archive() const
    {
      return primary_archive != nullptr;
    }



    template <int dim>
    bool
    TrajectoryManager<dim>::has_reference_archive() const
    {
      return dense_reference_archive != nullptr;
    }



    template <int dim>
    bool
    TrajectoryManager<dim>::has_restart_checkpoint_manager() const
    {
      return restart_manager != nullptr;
    }



    template <int dim>
    TrajectoryStore &
    TrajectoryManager<dim>::archive()
    {

    template <int dim>
    std::pair<unsigned int, unsigned int>
    TrajectoryManager<dim>::sparse_interval_for_record(const unsigned int record) const
    {
      AssertThrow(has_archive() && archive().size() > 0, ExcInternalError());
      AssertIndexRange(record, archive().size());
      // Partition logical records into half-open anchor bins [a,a+s), clipped
      // at the terminal record. The returned end is inclusive because reverse
      // replay materializes concrete state indices rather than step edges.
      const unsigned int anchor = (record / parsed_configuration.anchor_spacing)
                                  * parsed_configuration.anchor_spacing;
      const unsigned int end = std::min(anchor + parsed_configuration.anchor_spacing,
                                        archive().size() - 1);
      AssertThrow(anchor <= record && record <= end,
                  ExcMessage("Sparse replay interval does not contain its record."));
      return {anchor, end};
    }



    template <int dim>
    unsigned int
    TrajectoryManager<dim>::required_anchor_for_reverse_step(
      const unsigned int reverse_step) const
    {
      AssertThrow(has_archive() && reverse_step > 0 && reverse_step <= archive().size(),
                  ExcMessage("A reverse step has no corresponding sparse replay interval."));

      // Reverse step r transposes the forward edge q_(r-1)->q_r and therefore
      // consumes q_(r-1). Map that state, not q_r, to an interval. As a result
      // an exact anchor-boundary reverse step belongs to the preceding
      // interval, preventing cleanup from deleting the anchor it still needs.
      return sparse_interval_for_record(reverse_step - 1).first;
    }



    template <int dim>
    void
    TrajectoryManager<dim>::set_restart_trajectory_identity(
      AdjointRestartState &state) const
    {
      AssertThrow(has_archive() && archive().size() > 0, ExcInternalError());
      // Bind the checkpoint to both sparse-layout policy and the exact dense
      // accepted-step schedule. required_sparse_anchor is a durable cleanup
      // frontier: payload reclamation may delete newer intervals only after a
      // checkpoint naming the next required anchor has committed.
      state.trajectory_anchor_spacing = parsed_configuration.anchor_spacing;
      state.trajectory_schedule_records = archive().size();
      state.trajectory_schedule_fingerprint = archive().schedule_fingerprint();
      state.required_sparse_anchor = parsed_configuration.anchor_spacing > 1
                                     && state.next_reverse_step > 0
                                     ? required_anchor_for_reverse_step(state.next_reverse_step)
                                     : numbers::invalid_unsigned_int;
    }



    template <int dim>
    void
    TrajectoryManager<dim>::validate_sparse_restart_state(
      const AdjointRestartState &state) const
    {
      // Reject a numerically plausible but structurally different replay.
      // Time-grid coefficients, record count, and anchor spacing jointly
      // define the discrete reverse operator and must all match.
      AssertThrow(state.trajectory_anchor_spacing == parsed_configuration.anchor_spacing
                  && state.trajectory_schedule_records == archive().size()
                  && state.trajectory_schedule_fingerprint == archive().schedule_fingerprint(),
                  ExcMessage("The committed adjoint restart checkpoint refers to a different "
                             "accepted-step schedule or sparse-anchor layout."));

      if (parsed_configuration.anchor_spacing > 1 && state.next_reverse_step > 0)
        {
          // Validate both the checkpoint's claimed frontier and the payload
          // needed to realize it. Boundary assertions protect the subtle
          // r-1 interval rule against later off-by-one regressions.
          const unsigned int required =
            required_anchor_for_reverse_step(state.next_reverse_step);
          AssertThrow(state.required_sparse_anchor == required,
                      ExcMessage("The committed sparse restart anchor does not match the "
                                 "authoritative reverse-step interval mapping."));
          AssertThrow(archive().has_record(required),
                      ExcMessage("The committed sparse restart anchor package is missing."));
          AssertThrow(required_anchor_for_reverse_step(1) == 0, ExcInternalError());
          if (archive().size() >= parsed_configuration.anchor_spacing)
            AssertThrow(required_anchor_for_reverse_step(parsed_configuration.anchor_spacing) == 0,
                        ExcMessage("An anchor-boundary reverse step selected the wrong interval."));
        }
    }
  }

    class TrajectoryStore::Implementation
    {
      public:
        Implementation(const TrajectoryStorageMode mode,
                       const std::string &output_directory,
                       const MPI_Comm mpi_communicator,
                       const bool open_existing = false,
                       const unsigned int requested_records_per_segment = 4)
          :
          storage_mode(mode),
          directory(output_directory + "adjoint_trajectory/"),
          communicator(mpi_communicator),
          rank(Utilities::MPI::this_mpi_process(mpi_communicator)),
          n_processes(Utilities::MPI::n_mpi_processes(mpi_communicator)),
          records_per_segment(requested_records_per_segment)
        {
          AssertThrow(records_per_segment > 0, ExcInternalError());
          if (open_existing)
            open_existing_archive();
          else
            initialize_fresh_archive();
        }

        TrajectoryStorageMode mode() const
        {
          return storage_mode;
        }
        const std::string &archive_directory() const
        {
          return directory;
        }
        unsigned int size() const
        {
          return metadata_table.size();
        }
        bool has_record(const unsigned int record) const
        {
          return record < metadata_table.size()
                 && !record_locations[record].ranks.empty();
        }
        unsigned int first_available_record() const
        {
          return first_available;
        }
        unsigned int last_available_record() const
        {
          return last_available;
        }
        unsigned int payload_record_count() const
        {
          unsigned int result = 0;
          for (const RecordLocation &location : record_locations)
            result += !location.ranks.empty();
          return result;
        }

        /** A stable identity for the dense accepted-step journal. */
        std::uint64_t schedule_fingerprint() const
        {
          // FNV-1a over the logical accepted-step journal detects a restart
          // against a trajectory with different time levels. Hash the IEEE
          // bit patterns, not formatted decimal text, so equality means the
          // replay coefficients are bitwise identical to those checkpointed.
          std::uint64_t hash = 1469598103934665603ULL;
          hash_append(hash, metadata_table.size());
          hash_append(hash, records_per_segment);
          hash_append(hash, storage_mode == TrajectoryStorageMode::minimal ? 0 : 1);
          for (const ReplayStepMetadata &metadata : metadata_table)
            {
              hash_append(hash, metadata.step);
              hash_double(hash, metadata.time);
              hash_double(hash, metadata.time_step);
              hash_double(hash, metadata.old_time_step);
              hash_double(hash, metadata.outgoing_time_step);
              hash_append(hash, metadata.has_outgoing_step ? 1 : 0);
            }
          return hash;
        }

        /** Sum manifest and retained shared segment bytes (reported on rank 0). */
        std::size_t shared_archive_bytes() const
        {
          std::size_t bytes = 0;
          if (rank == 0)
            {
              std::error_code error;
              bytes = static_cast<std::size_t>(
                        std::filesystem::file_size(directory + manifest_filename, error));
              AssertThrow(!error, ExcMessage("Could not inspect trajectory manifest."));
              for (unsigned int segment = 0; segment <= last_segment(); ++segment)
                if (std::filesystem::exists(segment_filename(segment), error))
                  {
                    AssertThrow(!error, ExcMessage("Could not inspect trajectory segment."));
                    bytes += static_cast<std::size_t>(
                               std::filesystem::file_size(segment_filename(segment), error));
                    AssertThrow(!error, ExcMessage("Could not inspect trajectory segment."));
                  }
                else
                  AssertThrow(!error, ExcMessage("Could not inspect trajectory segment."));
            }
          return Utilities::MPI::broadcast(communicator, bytes, 0);
        }

        /** Number of shared payload segments still named by the manifest. */
        unsigned int shared_segment_count() const
        {
          unsigned int count = 0;
          if (rank == 0)
            for (unsigned int segment = 0; segment <= last_segment(); ++segment)
              for (const RecordLocation &location : record_locations)
                if (location.segment == segment && !location.ranks.empty())
                  {
                    ++count;
                    break;
                  }
          return Utilities::MPI::broadcast(communicator, count, 0);
        }

        /**
         * Drop only sparse physical payloads after @p last_to_keep. The dense
         * accepted-step schedule deliberately remains intact. A missing
         * segment before manifest publication is recoverable: a restart
         * repeats this idempotent transaction using its durable frontier.
         */
        TrajectoryCleanupResult
        reclaim_sparse_payloads_after(
          const unsigned int last_to_keep,
          const Diagnostics::FailureInjection<Diagnostics::CleanupFailurePoint> &injection = {})
        {
          TrajectoryCleanupResult result;
          const auto inject_here = [this, &injection]
                                   (const Diagnostics::CleanupFailurePoint point)
          {
            return injection.point == point && injection.applies_to_rank(rank);
          };
          if (!collective_cleanup_success(
                !inject_here(Diagnostics::CleanupFailurePoint::before_cleanup)))
            return result;

          // The cleanup frontier is a restart invariant: records <=
          // last_to_keep may include the sparse anchor required to replay the
          // next reverse step. First remove only segments that contain no
          // such payload. Because segments are append-only, deleting a whole
          // obsolete file cannot damage a retained record.
          //
          // the durable restart-safe anchor. If interrupted here, the old
          // manifest may still name obsolete records, but it never loses a
          // valid restart anchor and the checkpoint frontier repeats cleanup.
          if (!delete_sparse_obsolete_segments(last_to_keep, result))
            return result;
          if (!collective_cleanup_success(
                !inject_here(Diagnostics::CleanupFailurePoint::after_segment_deletion)))
            return result;
          if (!collective_cleanup_success(
                !inject_here(Diagnostics::CleanupFailurePoint::after_rank_payload_deletion)))
            return result;

          for (unsigned int record = last_to_keep + 1; record < record_locations.size(); ++record)
            if (!record_locations[record].ranks.empty())
              {
                ++result.records_deleted;
                add_statistics(result.reclaimed, payload_bytes[record]);
                record_locations[record].ranks.clear();
                payload_bytes[record] = {};
              }
          if (!collective_cleanup_success(
                !inject_here(Diagnostics::CleanupFailurePoint::before_collective_success))
              || !collective_cleanup_success(
                !inject_here(Diagnostics::CleanupFailurePoint::before_manifest_update)))
            return result;

          // Clearing the in-memory locations is not durable until the new
          // manifest is published. The atomic rename below is the commit
          // point: after a crash readers see either the old manifest or the
          // new one, never a partially serialized record table.
          bool local_success = true;
          if (rank == 0)
            try
              {
                write_manifest_atomically();
              }
            catch (const std::exception &)
              {
                local_success = false;
              }
          if (!collective_cleanup_success(local_success))
            return result;
          if (!collective_cleanup_success(
                !inject_here(Diagnostics::CleanupFailurePoint::after_manifest_update)))
            return result;
          result.complete = true;
          return result;
        }

        /** Test-only hook for validating an uncommitted shared append. */
        void set_append_failure_injection_for_testing(
          const Diagnostics::FailureInjection<Diagnostics::AppendFailurePoint> &injection)
        {
          append_failure_injection = injection;
        }

        TrajectoryCleanupResult
        reclaim_records_after(
          const unsigned int first_to_remove,
          const Diagnostics::FailureInjection<Diagnostics::CleanupFailurePoint> &injection = {})
        {
          TrajectoryCleanupResult result;
          const auto inject_here = [this, &injection]
                                   (const Diagnostics::CleanupFailurePoint point)
          {
            return injection.point == point && injection.applies_to_rank(rank);
          };
          if (metadata_table.empty() || first_available == numbers::invalid_unsigned_int
              || first_to_remove > last_available)
            {
              result.complete = delete_fully_reclaimed_segments(result);
              return result;
            }

          const unsigned int begin = std::max(first_to_remove, first_available);
          if (!collective_cleanup_success(
                !inject_here(Diagnostics::CleanupFailurePoint::before_cleanup))
              || !collective_cleanup_success(
                !inject_here(Diagnostics::CleanupFailurePoint::after_rank_payload_deletion))
              || !collective_cleanup_success(
                !inject_here(Diagnostics::CleanupFailurePoint::before_collective_success))
              || !collective_cleanup_success(
                !inject_here(Diagnostics::CleanupFailurePoint::before_manifest_update)))
            return result;

          for (unsigned int record = begin; record <= last_available; ++record)
            {
              ++result.records_deleted;
              add_statistics(result.reclaimed, payload_bytes[record]);
            }
          const unsigned int old_last = last_available;
          last_available = begin == 0 ? numbers::invalid_unsigned_int : begin - 1;
          bool local_success = true;
          if (rank == 0)
            try
              {
                write_manifest_atomically();
              }
            catch (const std::exception &)
              {
                local_success = false;
              }
          if (!collective_cleanup_success(local_success))
            {
              last_available = old_last;
              return result;
            }
          if (!collective_cleanup_success(
                !inject_here(Diagnostics::CleanupFailurePoint::after_manifest_update)))
            return result;
          result.complete = delete_fully_reclaimed_segments(result);
          return result;
        }

        void remove_archive()
        {
          bool local_success = true;
          if (rank == 0)
            {
              std::error_code error;
              std::filesystem::remove_all(directory, error);
              local_success = !error;
            }
          AssertThrow(collective_cleanup_success(local_success),
                      ExcMessage("Could not remove the adjoint trajectory archive collectively."));
          metadata_table.clear();
          record_locations.clear();
          payload_bytes.clear();
          first_available = 0;
          last_available = numbers::invalid_unsigned_int;
        }

        const ReplayStepMetadata &metadata(const unsigned int record) const
        {
          AssertIndexRange(record, metadata_table.size());
          return metadata_table[record];
        }

        /** Mark the final stored state as having no solved successor. */
        void mark_last_record_terminal()
        {
          AssertThrow(!metadata_table.empty(), ExcInternalError());
          // A stored state normally describes the step leaving it. The final
          // accepted state has no successor, so explicitly remove that edge;
          // reverse replay must not manufacture an extra dt beyond the
          // terminal objective time.
          metadata_table.back().has_outgoing_step = false;
          metadata_table.back().outgoing_time_step = 0.0;
          bool local_success = true;
          if (rank == 0)
            try
              {
                write_manifest_atomically();
              }
            catch (const std::exception &)
              {
                local_success = false;
              }
          collective_success(local_success, "publish terminal trajectory schedule metadata");
        }

        /** Append one dense accepted-step schedule entry without a payload. */
        void append_schedule(const ReplayStepMetadata &step_metadata)
        {
          AssertThrow(step_metadata.step == metadata_table.size(),
                      ExcMessage("Adjoint trajectory records must be contiguous."));
          // The dense schedule is the authoritative time grid and exists even
          // when sparse storage omits this state's vector payload. Contiguous
          // step numbers make record index == time-step index, avoiding a
          // second mapping during reverse traversal.
          metadata_table.push_back(step_metadata);
          record_locations.emplace_back();
          payload_bytes.emplace_back();
          if (metadata_table.size() == 1)
            first_available = step_metadata.step;
          last_available = step_metadata.step;
          bool local_success = true;
          if (rank == 0)
            try
              {
                write_manifest_atomically();
              }
            catch (const std::exception &)
              {
                local_success = false;
              }
          collective_success(local_success, "publish dense trajectory schedule metadata");
        }

        /** Store the physical payload for an existing logical schedule entry. */
        void store_payload(const unsigned int record,
                           const LinearAlgebra::BlockVector &solution,
                           const unsigned int temperature_block,
                           const std::vector<unsigned int> &composition_blocks,
                           const unsigned int velocity_block,
                           const unsigned int pressure_block)
        {
          AssertIndexRange(record, metadata_table.size());
          AssertThrow(!has_record(record),
                      ExcMessage("A physical payload was already committed for this trajectory state."));
          // Append is a two-phase transaction. First every rank serializes
          // its locally owned blocks and collectively writes an unpublished
          // byte range. Only after the shared file is synchronized does rank
          // zero atomically publish the new RecordLocation in the manifest.
          // An orphaned byte range is therefore harmless after a failed
          // append: has_record() is determined solely by committed metadata.
          std::vector<char> local_payload;
          TrajectoryStorageStatistics local_bytes;
          bool local_success = true;
          try
            {
              local_bytes.temperature_bytes =
                append_vector(local_payload, solution.block(temperature_block));
              for (const unsigned int block : composition_blocks)
                local_bytes.composition_bytes += append_vector(local_payload, solution.block(block));
              if (storage_mode == TrajectoryStorageMode::stokes_cached)
                {
                  local_bytes.velocity_bytes = append_vector(local_payload, solution.block(velocity_block));
                  local_bytes.pressure_bytes = append_vector(local_payload, solution.block(pressure_block));
                }
            }
          catch (const std::exception &)
            {
              local_success = false;
            }
          collective_success(local_success, "serialize rank-local trajectory payloads");
          abort_append_for_testing(Diagnostics::AppendFailurePoint::after_local_serialization);

          const RecordLocation location = allocate_record_location(
                                            record, local_payload.size(),
                                            partition_fingerprint(solution, temperature_block, composition_blocks,
                                                                  velocity_block, pressure_block), local_bytes);
          local_success = write_shared_payload(location, local_payload);
          collective_success(local_success, "write and synchronize shared trajectory payload");
          abort_append_for_testing(
            Diagnostics::AppendFailurePoint::after_collective_payload_write);
          abort_append_for_testing(Diagnostics::AppendFailurePoint::after_payload_write);

          // This mutation is tentative until write_manifest_atomically()
          // succeeds collectively. Fault-injection after payload write tests
          // that reopening the archive still ignores these unpublished bytes.
          record_locations[record] = location;
          payload_bytes[record] = local_bytes;
          local_success = true;
          if (rank == 0)
            try
              {
                write_manifest_atomically();
              }
            catch (const std::exception &)
              {
                local_success = false;
              }
          collective_success(local_success, "publish shared trajectory record metadata");
          abort_append_for_testing(Diagnostics::AppendFailurePoint::after_manifest_publication);
        }

        void store(const ReplayStepMetadata &step_metadata,
                   const LinearAlgebra::BlockVector &solution,
                   const unsigned int temperature_block,
                   const std::vector<unsigned int> &composition_blocks,
                   const unsigned int velocity_block,
                   const unsigned int pressure_block)
        {
          append_schedule(step_metadata);
          store_payload(step_metadata.step, solution, temperature_block,
                        composition_blocks, velocity_block, pressure_block);
        }

        void load(const unsigned int record,
                  LinearAlgebra::BlockVector &solution,
                  const unsigned int temperature_block,
                  const std::vector<unsigned int> &composition_blocks,
                  const unsigned int velocity_block,
                  const unsigned int pressure_block) const
        {
          AssertThrow(has_record(record),
                      ExcMessage("Adjoint trajectory record is not available in the committed archive."));
          // The manifest fixes both the shared byte range and the per-rank
          // partition fingerprint. A restart is intentionally same-rank and
          // same-partition: values are restored directly into their original
          // locally owned global indices without repartitioning.
          const RecordLocation &location = record_locations[record];
          AssertThrow(location.ranks.size() == n_processes,
                      ExcMessage("Trajectory record has an incompatible MPI rank count."));
          const RankPayload &local = location.ranks[rank];
          collective_success(local.fingerprint == partition_fingerprint(solution, temperature_block,
                                                                        composition_blocks, velocity_block,
                                                                        pressure_block),
                             "validate the active same-rank DoF partition");
          AssertThrow(local.size <= std::numeric_limits<std::size_t>::max(),
                      ExcMessage("Trajectory payload is too large for this process."));
          std::vector<char> payload(static_cast<std::size_t>(local.size));
          collective_success(read_shared_payload(location, payload), "read shared trajectory payload");
          const char *cursor = payload.data();
          const char *const end = cursor + payload.size();
          read_vector(cursor, end, solution.block(temperature_block));
          for (const unsigned int block : composition_blocks)
            read_vector(cursor, end, solution.block(block));
          if (storage_mode == TrajectoryStorageMode::stokes_cached)
            {
              read_vector(cursor, end, solution.block(velocity_block));
              read_vector(cursor, end, solution.block(pressure_block));
            }
          AssertThrow(cursor == end, ExcMessage("Trajectory payload has an invalid schema."));
        }

        TrajectoryStorageStatistics storage_statistics() const
        {
          TrajectoryStorageStatistics result;
          if (first_available != numbers::invalid_unsigned_int)
            for (unsigned int record = first_available; record <= last_available; ++record)
              add_statistics(result, payload_bytes[record]);
          if (rank == 0)
            {
              std::error_code error;
              result.metadata_bytes = static_cast<std::size_t>(
                                        std::filesystem::file_size(directory + manifest_filename, error));
              AssertThrow(!error, ExcMessage("Could not inspect trajectory manifest."));
            }
          return result;
        }

        std::size_t memory_consumption() const
        {
          std::size_t result = sizeof(*this) + directory.capacity();
          result += metadata_table.capacity() * sizeof(ReplayStepMetadata);
          result += payload_bytes.capacity() * sizeof(TrajectoryStorageStatistics);
          for (const RecordLocation &location : record_locations)
            result += sizeof(location) + location.ranks.capacity() * sizeof(RankPayload);
          return result;
        }

      private:
        struct RankPayload
        {
          std::uint64_t offset = 0, size = 0, fingerprint = 0;
          std::uint64_t temperature_bytes = 0, composition_bytes = 0;
          std::uint64_t velocity_bytes = 0, pressure_bytes = 0;
        };
        struct RecordLocation
        {
          unsigned int segment = 0;
          std::uint64_t record_offset = 0, total_size = 0;
          std::vector<RankPayload> ranks;
        };
        static constexpr const char *manifest_filename = "manifest";
        static constexpr unsigned int schema_version = 4;
        // Small enough that ordinary short trajectory cases cross a boundary.

        static void add_statistics(TrajectoryStorageStatistics &into,
                                   const TrajectoryStorageStatistics &from)
        {
          into.temperature_bytes += from.temperature_bytes;
          into.composition_bytes += from.composition_bytes;
          into.velocity_bytes += from.velocity_bytes;
          into.pressure_bytes += from.pressure_bytes;
        }
        static void append_bytes(std::vector<char> &output, const void *data,
                                 const std::size_t size)
        {
          AssertThrow(size <= std::numeric_limits<std::size_t>::max() - output.size(),
                      ExcMessage("Trajectory payload size overflows local memory."));
          const char *begin = static_cast<const char *>(data);
          output.insert(output.end(), begin, begin + size);
        }
        template <typename T>
        static void append_scalar(std::vector<char> &output, const T &value)
        {
          append_bytes(output, &value, sizeof(value));
        }
        static std::size_t append_vector(std::vector<char> &output,
                                         const LinearAlgebra::Vector &vector)
        {
          // Store only locally owned entries in IndexSet iteration order.
          // Ghost values are reconstructed by the caller after loading and
          // would otherwise be duplicated across MPI payloads.
          const std::size_t beginning = output.size();
          const std::uint64_t count = vector.locally_owned_elements().n_elements();
          append_scalar(output, count);
          for (const auto index : vector.locally_owned_elements())
            append_scalar(output, vector(index));
          return output.size() - beginning;
        }
        template <typename T>
        static T read_scalar(const char *&cursor, const char *const end)
        {
          AssertThrow(static_cast<std::uint64_t>(end - cursor) >= sizeof(T),
                      ExcMessage("Trajectory payload ended unexpectedly."));
          T value;
          std::memcpy(&value, cursor, sizeof(T));
          cursor += sizeof(T);
          return value;
        }
        static void read_vector(const char *&cursor, const char *const end,
                                LinearAlgebra::Vector &vector)
        {
          const std::uint64_t count = read_scalar<std::uint64_t>(cursor, end);
          AssertThrow(count == vector.locally_owned_elements().n_elements(),
                      ExcMessage("Trajectory payload does not match active MPI partitioning."));
          for (const auto index : vector.locally_owned_elements())
            vector(index) = read_scalar<double>(cursor, end);
          vector.compress(VectorOperation::insert);
        }
        static void hash_append(std::uint64_t &hash, const std::uint64_t value)
        {
          for (unsigned int byte = 0; byte < sizeof(value); ++byte)
            {
              hash ^= (value >> (8 * byte)) & 0xffU;
              hash *= 1099511628211ULL;
            }
        }
        static void hash_double(std::uint64_t &hash, const double value)
        {
          std::uint64_t bits = 0;
          static_assert(sizeof(bits) == sizeof(value));
          std::memcpy(&bits, &value, sizeof(bits));
          hash_append(hash, bits);
        }
        static void hash_vector_layout(std::uint64_t &hash, const LinearAlgebra::Vector &vector)
        {
          hash_append(hash, vector.locally_owned_elements().n_elements());
          for (const auto index : vector.locally_owned_elements())
            hash_append(hash, static_cast<std::uint64_t>(index));
        }
        std::uint64_t partition_fingerprint(const LinearAlgebra::BlockVector &solution,
                                            const unsigned int temperature_block,
                                            const std::vector<unsigned int> &composition_blocks,
                                            const unsigned int velocity_block,
                                            const unsigned int pressure_block) const
        {
          // Hash block identities and every locally owned global index. Equal
          // vector lengths alone are insufficient: a differently partitioned
          // mesh could have the same local size but assign different DoFs.
          std::uint64_t hash = 1469598103934665603ULL;
          hash_append(hash, storage_mode == TrajectoryStorageMode::minimal);
          hash_append(hash, temperature_block);
          hash_vector_layout(hash, solution.block(temperature_block));
          hash_append(hash, composition_blocks.size());
          for (const unsigned int block : composition_blocks)
            {
              hash_append(hash, block);
              hash_vector_layout(hash, solution.block(block));
            }
          if (storage_mode == TrajectoryStorageMode::stokes_cached)
            {
              hash_append(hash, velocity_block);
              hash_vector_layout(hash, solution.block(velocity_block));
              hash_append(hash, pressure_block);
              hash_vector_layout(hash, solution.block(pressure_block));
            }
          return hash;
        }

        RecordLocation allocate_record_location(const unsigned int record,
                                                const std::uint64_t local_size,
                                                const std::uint64_t local_fingerprint,
                                                const TrajectoryStorageStatistics &bytes) const
        {
          // Rank payloads for one logical record occupy one contiguous shared
          // range. Exscan computes this rank's prefix, while Allgather builds
          // the complete table that is later written to the manifest. Thus
          // every rank independently derives the same total record size and
          // can participate in the collective MPI-IO call.
          std::uint64_t prefix = 0;
          int status = MPI_Exscan(&local_size, &prefix, 1, MPI_UINT64_T, MPI_SUM, communicator);
          if (rank == 0) prefix = 0;
          collective_success(status == MPI_SUCCESS, "determine shared trajectory rank offsets");
          RecordLocation result;
          result.segment = record / records_per_segment;
          result.record_offset = end_offset_for_segment(result.segment);
          result.ranks.resize(n_processes);
          std::vector<std::uint64_t> sizes(n_processes), fingerprints(n_processes);
          std::vector<std::uint64_t> temperature(n_processes), composition(n_processes);
          std::vector<std::uint64_t> velocity(n_processes), pressure(n_processes);
          const auto remember_status = [&status](const int current_status)
          {
            if (current_status != MPI_SUCCESS)
              status = current_status;
          };
          status = MPI_SUCCESS;
          remember_status(MPI_Allgather(&local_size, 1, MPI_UINT64_T,
                                        sizes.data(), 1, MPI_UINT64_T, communicator));
          remember_status(MPI_Allgather(&local_fingerprint, 1, MPI_UINT64_T,
                                        fingerprints.data(), 1, MPI_UINT64_T, communicator));
          const std::uint64_t local_fields[4] = {bytes.temperature_bytes, bytes.composition_bytes,
                                                 bytes.velocity_bytes, bytes.pressure_bytes
                                                };
          std::uint64_t *fields[4] = {temperature.data(), composition.data(), velocity.data(), pressure.data()};
          for (unsigned int i = 0; i < 4; ++i)
            remember_status(MPI_Allgather(&local_fields[i], 1, MPI_UINT64_T,
                                          fields[i], 1, MPI_UINT64_T, communicator));
          collective_success(status == MPI_SUCCESS, "exchange shared trajectory record metadata");
          std::uint64_t offset = 0;
          for (unsigned int p = 0; p < n_processes; ++p)
            {
              AssertThrow(sizes[p] <= std::numeric_limits<std::uint64_t>::max() - offset,
                          ExcMessage("Trajectory record exceeds MPI file offset range."));
              RankPayload &payload = result.ranks[p];
              payload.offset = offset;
              payload.size = sizes[p];
              payload.fingerprint = fingerprints[p];
              payload.temperature_bytes = temperature[p];
              payload.composition_bytes = composition[p];
              payload.velocity_bytes = velocity[p];
              payload.pressure_bytes = pressure[p];
              offset += sizes[p];
            }
          AssertThrow(result.ranks[rank].offset == prefix,
                      ExcMessage("MPI_Exscan offset does not match record metadata."));
          result.total_size = offset;
          return result;
        }

        bool write_shared_payload(const RecordLocation &location,
                                  const std::vector<char> &payload) const
        {
          // Collective MPI-IO writes each rank's bytes at
          // record_offset+rank_offset. File_sync and collective close precede
          // manifest publication, establishing the payload-before-metadata
          // durability order required by store_payload().
          MPI_File file = MPI_FILE_NULL;
          const std::string filename = segment_filename(location.segment);
          bool success = MPI_File_open(communicator, const_cast<char *>(filename.c_str()),
                                       MPI_MODE_CREATE | MPI_MODE_WRONLY, MPI_INFO_NULL, &file) == MPI_SUCCESS;
          if (!collective_cleanup_success(success)) return false;
          MPI_Offset existing_size = 0;
          success = MPI_File_get_size(file, &existing_size) == MPI_SUCCESS;
          const MPI_Offset required_size = static_cast<MPI_Offset>(
                                             location.record_offset + location.total_size);
          if (success && existing_size < required_size)
            success = MPI_File_set_size(file, required_size) == MPI_SUCCESS;
          if (!collective_cleanup_success(success))
            {
              MPI_File_close(&file);
              return false;
            }
          const RankPayload &local = location.ranks[rank];
          const void *buffer = payload.empty() ? nullptr : payload.data();
          success = dealii::Utilities::MPI::LargeCount::File_write_at_all_c(
                      file, static_cast<MPI_Offset>(location.record_offset + local.offset), buffer,
                      static_cast<MPI_Count>(payload.size()), MPI_BYTE, MPI_STATUS_IGNORE) == MPI_SUCCESS;
          const bool sync_success = MPI_File_sync(file) == MPI_SUCCESS;
          const bool close_success = MPI_File_close(&file) == MPI_SUCCESS;
          return collective_cleanup_success(success && sync_success && close_success);
        }
        bool read_shared_payload(const RecordLocation &location,
                                 std::vector<char> &payload) const
        {
          MPI_File file = MPI_FILE_NULL;
          const std::string filename = segment_filename(location.segment);
          bool success = MPI_File_open(communicator, const_cast<char *>(filename.c_str()),
                                       MPI_MODE_RDONLY, MPI_INFO_NULL, &file) == MPI_SUCCESS;
          if (!collective_cleanup_success(success)) return false;
          const RankPayload &local = location.ranks[rank];
          void *buffer = payload.empty() ? nullptr : payload.data();
          success = dealii::Utilities::MPI::LargeCount::File_read_at_all_c(
                      file, static_cast<MPI_Offset>(location.record_offset + local.offset), buffer,
                      static_cast<MPI_Count>(payload.size()), MPI_BYTE, MPI_STATUS_IGNORE) == MPI_SUCCESS;
          const bool close_success = MPI_File_close(&file) == MPI_SUCCESS;
          return collective_cleanup_success(success && close_success);
        }
        bool append_failure_here(const Diagnostics::AppendFailurePoint point) const
        {
          return append_failure_injection.point == point
                 && append_failure_injection.applies_to_rank(rank);
        }
        void abort_append_for_testing(const Diagnostics::AppendFailurePoint point) const
        {
          if (!collective_cleanup_success(!append_failure_here(point)))
            throw QuietException();
        }
        std::uint64_t end_offset_for_segment(const unsigned int segment) const
        {
          std::uint64_t result = 0;
          for (const RecordLocation &location : record_locations)
            if (location.segment == segment)
              result = std::max(result, location.record_offset + location.total_size);
          return result;
        }
        bool record_is_available(const unsigned int record) const
        {
          return first_available != numbers::invalid_unsigned_int
                 && record >= first_available && record <= last_available;
        }
        bool segment_is_fully_reclaimed(const unsigned int segment) const
        {
          for (unsigned int record = 0; record < record_locations.size(); ++record)
            if (record_locations[record].segment == segment && record_is_available(record)) return false;
          return true;
        }
        bool segment_is_sparse_obsolete(const unsigned int segment,
                                        const unsigned int last_to_keep) const
        {
          for (unsigned int record = 0; record < record_locations.size(); ++record)
            if (record_locations[record].segment == segment
                && !record_locations[record].ranks.empty()
                && record <= last_to_keep)
              return false;
          return true;
        }
        bool delete_sparse_obsolete_segments(const unsigned int last_to_keep,
                                             TrajectoryCleanupResult &result) const
        {
          bool success = true;
          if (rank == 0)
            try
              {
                for (unsigned int segment = 0; segment <= last_segment(); ++segment)
                  if (segment_is_sparse_obsolete(segment, last_to_keep))
                    {
                      std::error_code error;
                      const std::string filename = segment_filename(segment);
                      if (std::filesystem::exists(filename, error))
                        {
                          if (error || !std::filesystem::remove(filename, error) || error)
                            {
                              success = false;
                              break;
                            }
                          ++result.payload_files_deleted;
                        }
                      else if (error)
                        {
                          success = false;
                          break;
                        }
                    }
              }
            catch (const std::exception &)
              {
                success = false;
              }
          return collective_cleanup_success(success);
        }
        bool delete_fully_reclaimed_segments(TrajectoryCleanupResult &result) const
        {
          bool success = true;
          if (rank == 0)
            try
              {
                for (unsigned int segment = 0; segment <= last_segment(); ++segment)
                  if (segment_is_fully_reclaimed(segment))
                    {
                      std::error_code error;
                      const std::string filename = segment_filename(segment);
                      if (std::filesystem::exists(filename, error))
                        {
                          if (error || !std::filesystem::remove(filename, error) || error)
                            {
                              success = false;
                              break;
                            }
                          ++result.payload_files_deleted;
                        }
                      else if (error)
                        {
                          success = false;
                          break;
                        }
                    }
              }
            catch (const std::exception &)
              {
                success = false;
              }
          return collective_cleanup_success(success);
        }
        unsigned int last_segment() const
        {
          unsigned int result = 0;
          for (const RecordLocation &location : record_locations)
            result = std::max(result, location.segment);
          return result;
        }

        void initialize_fresh_archive()
        {
          bool success = true;
          if (rank == 0)
            try
              {
                std::error_code error;
                if (std::filesystem::exists(directory, error))
                  {
                    // Never recursively replace an arbitrary user directory.
                    // At least one recognized trajectory marker must prove
                    // that this path belongs to a previous generated archive.
                    const bool marker = std::filesystem::exists(directory + manifest_filename, error)
                                        || std::filesystem::exists(directory + "manifest.txt", error)
                                        || std::filesystem::exists(directory + "metadata.txt", error);
                    AssertThrow(!error && marker,
                                ExcMessage("Refusing to reset a non-trajectory directory."));
                    std::filesystem::remove_all(directory, error);
                    AssertThrow(!error, ExcMessage("Could not reset stale trajectory archive."));
                  }
              }
            catch (const std::exception &)
              {
                success = false;
              }
          collective_success(success, "initialize shared trajectory archive");
          Utilities::create_directory(directory, communicator, true);
          success = true;
          if (rank == 0)
            try
              {
                write_manifest_atomically();
              }
            catch (const std::exception &)
              {
                success = false;
              }
          collective_success(success, "publish empty shared trajectory manifest");
        }
        void open_existing_archive()
        {
          AssertThrow(Utilities::fexists(directory + manifest_filename, communicator),
                      ExcMessage("Cannot resume an adjoint trajectory: shared manifest is missing."));
          read_manifest();
        }
        void write_manifest_atomically() const
        {
          AssertThrow(rank == 0, ExcInternalError());
          // Serialize the complete logical journal and every committed byte
          // location to a sibling temporary file. POSIX rename is the sole
          // publication point, so a reader never observes a truncated table.
          const std::string temporary = directory + std::string(manifest_filename) + ".tmp";
          std::ofstream output(temporary, std::ios::trunc);
          AssertThrow(output, ExcMessage("Could not write temporary trajectory manifest."));
          output.precision(std::numeric_limits<double>::max_digits10);
          output << "ASPECT_ADJOINT_TRAJECTORY_SHARED_MPI_V4\n"
                 << "schema_version " << schema_version << '\n'
                 << "storage_mode " << (storage_mode == TrajectoryStorageMode::minimal ? "minimal" : "stokes_cached") << '\n'
                 << "records_per_segment " << records_per_segment << '\n'
                 << "records " << metadata_table.size() << '\n'
                 << "first_available " << first_available << '\n'
                 << "last_available " << last_available << '\n';
          for (unsigned int record = 0; record < metadata_table.size(); ++record)
            {
              const ReplayStepMetadata &metadata = metadata_table[record];
              const RecordLocation &location = record_locations[record];
              output << "record " << metadata.step << ' ' << metadata.time << ' ' << metadata.time_step << ' '
                     << metadata.old_time_step << ' ' << metadata.outgoing_time_step << ' '
                     << (metadata.has_outgoing_step ? 1 : 0) << ' '
                     << (storage_mode == TrajectoryStorageMode::minimal ? 0 : 1) << ' '
                     << location.segment << ' ' << location.record_offset << ' ' << location.total_size << ' '
                     << location.ranks.size() << '\n';
              for (unsigned int p = 0; p < location.ranks.size(); ++p)
                {
                  const RankPayload &payload = location.ranks[p];
                  output << "rank " << p << ' ' << payload.offset << ' ' << payload.size << ' '
                         << payload.fingerprint << ' ' << payload.temperature_bytes << ' '
                         << payload.composition_bytes << ' ' << payload.velocity_bytes << ' '
                         << payload.pressure_bytes << '\n';
                }
            }
          output.close();
          AssertThrow(output, ExcMessage("Writing temporary trajectory manifest failed."));
          std::error_code error;
          std::filesystem::rename(temporary, directory + manifest_filename, error);
          AssertThrow(!error, ExcMessage("Could not publish trajectory manifest: " + error.message()));
        }
        void read_manifest()
        {
          // Rank zero validates and parses the text manifest. The parsed
          // scalar/table entries are then broadcast, avoiding independent
          // filesystem parsing and ensuring all ranks enter later collective
          // MPI-IO operations with identical record locations.
          bool success = true;
          if (rank == 0)
            try
              {
                std::ifstream input(directory + manifest_filename);
                std::string magic, key, mode;
                unsigned int version = 0, segment_records = 0, records = 0;
                input >> magic >> key >> version >> key >> mode >> key >> segment_records
                >> key >> records >> key >> first_available >> key >> last_available;
                AssertThrow(input && magic == "ASPECT_ADJOINT_TRAJECTORY_SHARED_MPI_V4"
                            && key == "last_available" && version == schema_version
                            && segment_records == records_per_segment,
                            ExcMessage("Invalid or incompatible shared trajectory manifest."));
                AssertThrow((mode == "minimal" || mode == "stokes_cached")
                            && (mode == "minimal") == (storage_mode == TrajectoryStorageMode::minimal),
                            ExcMessage("Trajectory storage mode differs from the archive."));
                metadata_table.resize(records);
                record_locations.resize(records);
                payload_bytes.resize(records);
                for (unsigned int record = 0; record < records; ++record)
                  {
                    std::string record_key;
                    unsigned int has_outgoing_step = 0, mode_code = 0, stored_processes = 0;
                    RecordLocation &location = record_locations[record];
                    input >> record_key >> metadata_table[record].step >> metadata_table[record].time
                    >> metadata_table[record].time_step >> metadata_table[record].old_time_step
                    >> metadata_table[record].outgoing_time_step >> has_outgoing_step
                    >> mode_code >> location.segment
                    >> location.record_offset >> location.total_size >> stored_processes;
                    AssertThrow(input && record_key == "record" && metadata_table[record].step == record
                                && (has_outgoing_step == 0 || has_outgoing_step == 1)
                                && mode_code == (storage_mode == TrajectoryStorageMode::minimal ? 0 : 1)
                                && (stored_processes == 0 || stored_processes == n_processes),
                                ExcMessage("Shared trajectory record is incompatible with this MPI run."));
                    metadata_table[record].has_outgoing_step = has_outgoing_step == 1;
                    location.ranks.resize(stored_processes);
                    for (unsigned int p = 0; p < stored_processes; ++p)
                      {
                        std::string rank_key;
                        unsigned int stored_rank = 0;
                        RankPayload &payload = location.ranks[p];
                        input >> rank_key >> stored_rank >> payload.offset >> payload.size >> payload.fingerprint
                        >> payload.temperature_bytes >> payload.composition_bytes
                        >> payload.velocity_bytes >> payload.pressure_bytes;
                        AssertThrow(input && rank_key == "rank" && stored_rank == p,
                                    ExcMessage("Invalid shared trajectory rank index."));
                      }
                    if (stored_processes != 0)
                      set_local_payload_bytes(record);
                  }
              }
            catch (const std::exception &)
              {
                success = false;
              }
          collective_success(success, "read shared trajectory manifest");
          unsigned int records = rank == 0 ? metadata_table.size() : 0;
          records = Utilities::MPI::broadcast(communicator, records, 0);
          first_available = Utilities::MPI::broadcast(communicator, first_available, 0);
          last_available = Utilities::MPI::broadcast(communicator, last_available, 0);
          if (rank != 0)
            {
              metadata_table.resize(records);
              record_locations.resize(records);
              payload_bytes.resize(records);
            }
          for (unsigned int record = 0; record < records; ++record)
            {
              ReplayStepMetadata &metadata = metadata_table[record];
              RecordLocation &location = record_locations[record];
              metadata.step = Utilities::MPI::broadcast(communicator, metadata.step, 0);
              metadata.time = Utilities::MPI::broadcast(communicator, metadata.time, 0);
              metadata.time_step = Utilities::MPI::broadcast(communicator, metadata.time_step, 0);
              metadata.old_time_step = Utilities::MPI::broadcast(communicator, metadata.old_time_step, 0);
              metadata.outgoing_time_step = Utilities::MPI::broadcast(communicator, metadata.outgoing_time_step, 0);
              unsigned int has_outgoing_step = metadata.has_outgoing_step ? 1 : 0;
              has_outgoing_step = Utilities::MPI::broadcast(communicator, has_outgoing_step, 0);
              metadata.has_outgoing_step = has_outgoing_step == 1;
              location.segment = Utilities::MPI::broadcast(communicator, location.segment, 0);
              location.record_offset = Utilities::MPI::broadcast(communicator, location.record_offset, 0);
              location.total_size = Utilities::MPI::broadcast(communicator, location.total_size, 0);
              unsigned int stored_processes = rank == 0 ? location.ranks.size() : 0;
              stored_processes = Utilities::MPI::broadcast(communicator, stored_processes, 0);
              AssertThrow(stored_processes == 0 || stored_processes == n_processes,
                          ExcMessage("Trajectory payload has an incompatible MPI rank count."));
              if (rank != 0) location.ranks.resize(stored_processes);
              for (unsigned int p = 0; p < stored_processes; ++p)
                {
                  RankPayload &payload = location.ranks[p];
                  payload.offset = Utilities::MPI::broadcast(communicator, payload.offset, 0);
                  payload.size = Utilities::MPI::broadcast(communicator, payload.size, 0);
                  payload.fingerprint = Utilities::MPI::broadcast(communicator, payload.fingerprint, 0);
                  payload.temperature_bytes = Utilities::MPI::broadcast(communicator, payload.temperature_bytes, 0);
                  payload.composition_bytes = Utilities::MPI::broadcast(communicator, payload.composition_bytes, 0);
                  payload.velocity_bytes = Utilities::MPI::broadcast(communicator, payload.velocity_bytes, 0);
                  payload.pressure_bytes = Utilities::MPI::broadcast(communicator, payload.pressure_bytes, 0);
                }
              if (stored_processes != 0)
                set_local_payload_bytes(record);
            }
        }
        void set_local_payload_bytes(const unsigned int record)
        {
          const RankPayload &local = record_locations[record].ranks[rank];
          payload_bytes[record].temperature_bytes = local.temperature_bytes;
          payload_bytes[record].composition_bytes = local.composition_bytes;
          payload_bytes[record].velocity_bytes = local.velocity_bytes;
          payload_bytes[record].pressure_bytes = local.pressure_bytes;
        }
        std::string segment_filename(const unsigned int segment) const
        {
          return directory + "trajectory_" + Utilities::int_to_string(segment, 6) + ".bin";
        }
        void collective_success(const bool local_success, const std::string &stage) const
        {
          // Convert rank-local I/O errors into one collective decision before
          // any rank proceeds to the next transaction stage. The barrier is
          // intentional: without it one rank could publish metadata while
          // another rank is still unwinding a failed payload write.
          const unsigned int global_success = Utilities::MPI::min(local_success ? 1U : 0U, communicator);
          const int barrier_status = MPI_Barrier(communicator);
          AssertThrow(global_success == 1 && barrier_status == MPI_SUCCESS,
                      ExcMessage("Adjoint MPI trajectory operation failed collectively while attempting to "
                                 + stage + ". The record was not published."));
        }
        bool collective_cleanup_success(const bool local_success) const
        {
          const unsigned int global_success = Utilities::MPI::min(local_success ? 1U : 0U, communicator);
          return global_success == 1 && MPI_Barrier(communicator) == MPI_SUCCESS;
        }

        TrajectoryStorageMode storage_mode;
        std::string directory;
        MPI_Comm communicator;
        unsigned int rank, n_processes;
        unsigned int records_per_segment;
        std::vector<ReplayStepMetadata> metadata_table;
        std::vector<RecordLocation> record_locations;
        std::vector<TrajectoryStorageStatistics> payload_bytes;
        unsigned int first_available = 0;
        unsigned int last_available = numbers::invalid_unsigned_int;
        Diagnostics::FailureInjection<Diagnostics::AppendFailurePoint>
        append_failure_injection;
    };

    template <int dim>
    void
    Manager<dim>::validate_time_dependent_temperature_setup() const
    {
      using NonlinearSolver = typename Parameters<dim>::NonlinearSolver;

      const std::string impermeable_boundary_message =
        "The time-dependent adjoint currently requires u.n = 0 on physical "
        "boundaries. Prescribed normal inflow/outflow and inflow-dependent "
        "compositional boundary conditions are not supported in v1.";

      AssertThrow(simulator != nullptr, ExcInternalError());
      AssertThrow(this->get_parameters().nonlinear_solver
                  == NonlinearSolver::single_Advection_single_Stokes_adjoint_temperature,
                  ExcMessage("The time-dependent temperature adjoint requires "
                             "Nonlinear solver scheme = single Advection, single Stokes, adjoint temperature."));
      AssertThrow(this->introspection().n_chemical_composition_fields()
                  == this->introspection().n_compositional_fields,
                  ExcMessage("The time-dependent temperature adjoint with compositional controls "
                             "currently supports only compositional fields of type chemical composition."));
      for (unsigned int c = 0; c < this->introspection().n_compositional_fields; ++c)
        {
          const AdvectionField composition_field = AdvectionField::composition(c);
          AssertThrow(composition_field.advection_method(this->introspection())
                      == Parameters<dim>::AdvectionFieldMethod::fem_field,
                      ExcMessage("The time-dependent temperature adjoint with compositional controls "
                                 "currently supports only composition fields with method 'field'."));
          AssertThrow(composition_field.is_discontinuous(this->introspection()) == false,
                      ExcMessage("The time-dependent temperature adjoint with compositional controls "
                                 "supports only continuous composition elements."));
        }
      AssertThrow(this->get_parameters().use_operator_splitting == false,
                  ExcMessage("The time-dependent temperature adjoint v1 does not support operator splitting."));
      AssertThrow(this->get_parameters().include_melt_transport == false,
                  ExcMessage("The time-dependent temperature adjoint v1 does not support melt transport."));
      AssertThrow(this->get_parameters().advection_stabilization_method
                  == Parameters<dim>::AdvectionStabilizationMethod::entropy_viscosity,
                  ExcMessage("The time-dependent temperature adjoint v1 does not support SUPG stabilization. "
                             "Use Stabilization method = entropy viscosity."));
      AssertThrow(this->get_parameters().use_artificial_viscosity_smoothing == false,
                  ExcMessage("The time-dependent temperature adjoint v1 does not support artificial-viscosity smoothing. "
                             "The neighbor maximum used by smoothing needs an explicit transpose before it can be enabled."));
      AssertThrow(simulator->mesh_deformation == nullptr,
                  ExcMessage("The time-dependent temperature adjoint v1 supports only fixed meshes."));
      AssertThrow(this->get_parameters().adaptive_refinement_interval == 0
                  && this->get_parameters().additional_refinement_times.empty(),
                  ExcMessage("The time-dependent temperature adjoint trajectory "
                             "archive supports only a fixed mesh. Disable "
                             "scheduled and periodic mesh refinement."));
      AssertThrow(AdvectionField::temperature().is_discontinuous(this->introspection()) == false,
                  ExcMessage("The time-dependent temperature adjoint v1 supports only continuous temperature elements."));
      AssertThrow(this->get_heating_model_manager().get_active_plugin_names().empty(),
                  ExcMessage("The time-dependent temperature adjoint v1 does not yet support heating model derivatives, "
                             "including latent heat, adiabatic heating, shear heating, or radiogenic heating."));

      // For the standard Box geometry, the boundary id directly identifies
      // the Cartesian normal direction. This is a configuration-level check:
      // it rejects an explicitly prescribed normal velocity component without
      // sampling a boundary function. Curved-geometry velocity plugins may
      // legitimately use a full Cartesian component mask for a tangential
      // field, so their u.n=0 contract cannot be inferred from the mask and
      // is documented as a v1 user requirement instead.
      if (Plugins::plugin_type_matches<const GeometryModel::Box<dim>>(this->get_geometry_model()))
        for (const types::boundary_id boundary_id :
             this->get_boundary_velocity_manager().get_prescribed_boundary_velocity_indicators())
          {
            const unsigned int normal_component = boundary_id / 2;
            const ComponentMask component_mask =
              this->get_boundary_velocity_manager().get_component_mask(boundary_id);
            AssertThrow(component_mask[this->introspection().component_indices.velocities[normal_component]] == false,
                        ExcMessage(impermeable_boundary_message));
          }

      const MaterialModel::NonlinearDependence::ModelDependence model_dependence =
        this->get_material_model().get_model_dependence();
      AssertThrow((static_cast<int>(model_dependence.viscosity)
                   & (static_cast<int>(MaterialModel::NonlinearDependence::pressure)
                      | static_cast<int>(MaterialModel::NonlinearDependence::strain_rate))) == 0,
                  ExcMessage("The time-dependent temperature adjoint v1 supports only temperature- and "
                             "composition-dependent viscosity. Pressure- and strain-rate-dependent "
                             "viscosity remain unsupported."));
      if ((static_cast<int>(model_dependence.viscosity)
           & static_cast<int>(MaterialModel::NonlinearDependence::any_variable)) != 0)
        AssertThrow(this->get_parameters().material_averaging
                    == MaterialModel::MaterialAveraging::none,
                    ExcMessage("The time-dependent temperature adjoint v1 supports eta(T,C) only when "
                               "Material model/Material averaging = none. The transpose of viscosity "
                               "averaging/projection is not part of this v1 layer."));
      AssertThrow((static_cast<int>(model_dependence.thermal_conductivity)
                   & static_cast<int>(MaterialModel::NonlinearDependence::any_variable)) == 0,
                  ExcMessage("The time-dependent temperature adjoint v1 does not yet support thermal-conductivity derivatives."));
      AssertThrow((static_cast<int>(model_dependence.specific_heat)
                   & (static_cast<int>(MaterialModel::NonlinearDependence::pressure)
                      | static_cast<int>(MaterialModel::NonlinearDependence::strain_rate)
                      | static_cast<int>(MaterialModel::NonlinearDependence::compositional_fields))) == 0,
                  ExcMessage("The time-dependent temperature adjoint v1 supports only temperature-dependent specific heat. "
                             "Pressure-, strain-rate-, and composition-dependent specific heat remain unsupported."));
    }

    template <int dim>
    void
    Manager<dim>::recover_stokes_state(const unsigned int record,
                                       LinearAlgebra::BlockVector &state)
    {
      AssertThrow(trajectory_manager.has_archive(), ExcInternalError());
      const ReplayStepMetadata &metadata = trajectory_manager.archive().metadata(record);

      // The v1 dependency audit permits this reconstruction: Stokes depends
      // on T,C and time-dependent boundary data, but not on archived matrices,
      // constraints, pressure, or strain rate. The old fields are not read by
      // the supported incompressible/Picard Stokes path.
      simulator->solution = state;
      simulator->old_solution = state;
      simulator->old_old_solution = state;
      simulator->current_linearization_point = state;
      simulator->time = metadata.time;
      simulator->time_step = metadata.time_step;
      simulator->old_time_step = metadata.old_time_step;
      simulator->timestep_number = metadata.step;
      simulator->last_pressure_normalization_adjustment = 0.0;
      simulator->compute_current_constraints();

      {
        TimerOutput::Scope recovery_timer(this->get_computing_timer(),
                                          "Adjoint trajectory Stokes recovery");
        simulator->rebuild_stokes_matrix = true;
        simulator->rebuild_stokes_preconditioner = true;
        simulator->assemble_and_solve_stokes();
      }

      state = simulator->solution;
      ++trajectory_manager.stokes_recovery_solves;
    }



    template <int dim>
    void
    Manager<dim>::run_trajectory_stokes_recovery_diagnostic()
    {
      AssertThrow(trajectory_manager.has_archive(), ExcInternalError());
      AssertThrow(trajectory_manager.archive().mode()
                  == TrajectoryStorageMode::stokes_cached,
                  ExcMessage("The trajectory Stokes recovery diagnostic needs "
                             "'Trajectory storage mode = stokes cached' so it "
                             "can compare with the forward Stokes state."));

      std::vector<unsigned int> sampled_records(1, 0);
      if (trajectory_manager.archive().size() > 2)
        sampled_records.push_back((trajectory_manager.archive().size() - 1) / 2);
      if (trajectory_manager.archive().size() > 1)
        sampled_records.push_back(trajectory_manager.archive().size() - 1);
      std::sort(sampled_records.begin(), sampled_records.end());
      sampled_records.erase(std::unique(sampled_records.begin(), sampled_records.end()),
                            sampled_records.end());

      const unsigned int velocity_block =
        this->introspection().block_indices.velocities;
      const unsigned int pressure_block =
        this->introspection().block_indices.pressure;

      for (const unsigned int record : sampled_records)
        {
          LinearAlgebra::BlockVector recovered;
          recovered.reinit(this->introspection().index_sets.system_partitioning,
                           this->introspection().index_sets.system_relevant_partitioning,
                           this->get_mpi_communicator());
          recovered = 0.0;
          {
            TimerOutput::Scope read_timer(this->get_computing_timer(),
                                          "Adjoint trajectory read");
            trajectory_manager.archive().load(record,
                                              recovered,
                                              this->introspection().block_indices.temperature,
                                              this->introspection().block_indices.compositional_fields,
                                              velocity_block,
                                              pressure_block);
          }
          const LinearAlgebra::BlockVector forward_stokes_state(recovered);
          recovered.block(velocity_block) = 0.0;
          recovered.block(pressure_block) = 0.0;
          recover_stokes_state(record, recovered);

          LinearAlgebra::Vector velocity_difference(recovered.block(velocity_block));
          velocity_difference.add(-1.0,
                                  forward_stokes_state.block(velocity_block));
          const double velocity_error = velocity_difference.l2_norm()
                                        / std::max(forward_stokes_state.block(velocity_block).l2_norm(),
                                                   1e-50);

          LinearAlgebra::Vector pressure_difference(recovered.block(pressure_block));
          pressure_difference.add(-1.0,
                                  forward_stokes_state.block(pressure_block));
          // A pressure nullspace may leave a constant offset even though the
          // recovered Stokes solution is physically equivalent.
          pressure_difference.add(-pressure_difference.mean_value());
          const double pressure_error = pressure_difference.l2_norm()
                                        / std::max(forward_stokes_state.block(pressure_block).l2_norm(),
                                                   1e-50);

          this->get_pcout() << "   Adjoint trajectory Stokes recovery, record "
                            << record << ": relative velocity error = "
                            << std::setprecision(16) << velocity_error
                            << ", gauge-normalized relative pressure error = "
                            << pressure_error << std::setprecision(6) << std::endl;
          AssertThrow(velocity_error <= trajectory_stokes_recovery_tolerance
                      && pressure_error <= trajectory_stokes_recovery_tolerance,
                      ExcMessage("Adjoint trajectory Stokes recovery did not "
                                 "match the cached forward state within the "
                                 "configured tolerance."));
        }
    }



    template <int dim>
    void
    Manager<dim>::validate_trajectory_replay_capability() const
    {
      AssertThrow(simulator != nullptr, ExcInternalError());
      // Phase A deliberately has an explicit, small capability envelope. It
      // is not inferred from Plugin::save() because an empty payload does not
      // establish that update(), signals, or external resources are replay
      // deterministic.
      AssertThrow(simulator->particle_managers.empty()
                  && !this->get_parameters().volume_of_fluid_tracking_enabled,
                  ExcMessage("Phase-A trajectory replay does not support particles or volume-of-fluid state."));
      AssertThrow(!this->get_parameters().run_postprocessors_on_nonlinear_iterations,
                  ExcMessage("Phase-A trajectory replay does not support postprocessors on nonlinear iterations."));
      AssertThrow(Plugins::plugin_type_matches<const GeometryModel::Box<dim>>(
                    this->get_geometry_model())
                  && Plugins::plugin_type_matches<const MaterialModel::Simple<dim>>(
                    this->get_material_model())
                  && Plugins::plugin_type_matches<const GravityModel::Vertical<dim>>(
                    this->get_gravity_model()),
                  ExcMessage("Phase-A trajectory replay is explicitly limited to the "
                             "stateless Box/Simple/Vertical benchmark capability set. "
                             "Other plugin combinations require a dedicated replay audit."));
      AssertThrow(this->get_boundary_velocity_manager().get_active_plugin_names().empty(),
                  ExcMessage("Phase-A trajectory replay does not yet support prescribed "
                             "velocity plugins, including time-dependent functions."));
    }



    template <int dim>
    void
    Manager<dim>::run_trajectory_replay_diagnostic()
    {
      AssertThrow(trajectory_manager.has_archive() && trajectory_manager.archive().size() > 1,
                  ExcMessage("The trajectory replay diagnostic requires at least one accepted transition."));
      validate_trajectory_replay_capability();
      this->get_pcout() << "ASPECT_ADJOINT_REPLAY diagnostic_begin records="
                        << trajectory_manager.archive().size() << std::endl;

      const LinearAlgebra::BlockVector saved_solution = simulator->solution;
      const LinearAlgebra::BlockVector saved_old_solution = simulator->old_solution;
      const LinearAlgebra::BlockVector saved_old_old_solution = simulator->old_old_solution;
      const LinearAlgebra::BlockVector saved_linearization_point = simulator->current_linearization_point;
      const LinearAlgebra::BlockVector saved_system_rhs = simulator->system_rhs;
      const double saved_time = simulator->time;
      const double saved_time_step = simulator->time_step;
      const double saved_old_time_step = simulator->old_time_step;
      const unsigned int saved_timestep_number = simulator->timestep_number;
      const double saved_pressure_normalization_adjustment =
        simulator->last_pressure_normalization_adjustment;
      const unsigned int saved_stokes_recovery_solves = trajectory_manager.stokes_recovery_solves;

      const auto load_reference_state = [this](const unsigned int record)
      {
        LinearAlgebra::BlockVector locally_owned;
        locally_owned.reinit(this->introspection().index_sets.system_partitioning,
                             this->get_mpi_communicator());
        locally_owned = 0.0;
        trajectory_manager.archive().load(record,
                                          locally_owned,
                                          this->introspection().block_indices.temperature,
                                          this->introspection().block_indices.compositional_fields,
                                          this->introspection().block_indices.velocities,
                                          this->introspection().block_indices.pressure);
        LinearAlgebra::BlockVector result;
        result.reinit(this->introspection().index_sets.system_partitioning,
                      this->introspection().index_sets.system_relevant_partitioning,
                      this->get_mpi_communicator());
        result = locally_owned;
        if (trajectory_manager.archive().mode() == TrajectoryStorageMode::minimal)
          recover_stokes_state(record, result);
        return result;
      };
      const auto relative_difference = [](const LinearAlgebra::Vector &actual,
                                          const LinearAlgebra::Vector &reference)
      {
        LinearAlgebra::Vector difference(actual);
        difference.add(-1.0, reference);
        return difference.l2_norm() / std::max(reference.l2_norm(), 1e-50);
      };
      const auto locally_owned_copy = [this](const LinearAlgebra::BlockVector &source)
      {
        LinearAlgebra::BlockVector result;
        result.reinit(this->introspection().index_sets.system_partitioning,
                      this->get_mpi_communicator());
        result = source;
        return result;
      };
      const auto equal_time = [](const double left, const double right)
      {
        return std::abs(left - right)
               <= 64.0 * std::numeric_limits<double>::epsilon()
               * std::max({1.0, std::abs(left), std::abs(right)});
      };

      try
        {
          const ReplayStepMetadata &initial_metadata = trajectory_manager.archive().metadata(0);
          simulator->solution = load_reference_state(0);
          simulator->old_solution = load_reference_state(0);
          simulator->old_old_solution = load_reference_state(0);
          simulator->current_linearization_point = simulator->solution;
          simulator->time = initial_metadata.time;
          simulator->time_step = initial_metadata.time_step;
          simulator->old_time_step = initial_metadata.old_time_step;
          simulator->timestep_number = initial_metadata.step;
          simulator->last_pressure_normalization_adjustment = 0.0;
          this->get_pcout() << "ASPECT_ADJOINT_REPLAY initial_state_loaded" << std::endl;

          for (unsigned int record = 0; record + 1 < trajectory_manager.archive().size(); ++record)
            {
              const ReplayStepMetadata &metadata = trajectory_manager.archive().metadata(record);
              const ReplayStepMetadata &next_metadata = trajectory_manager.archive().metadata(record + 1);
              AssertThrow(metadata.has_outgoing_step,
                          ExcMessage("A non-final archived state is missing its accepted outgoing time step."));
              AssertThrow(equal_time(metadata.time + metadata.outgoing_time_step,
                                     next_metadata.time),
                          ExcMessage("The accepted trajectory schedule has inconsistent time metadata."));

              this->get_pcout() << "ASPECT_ADJOINT_REPLAY replay_begin step="
                                << metadata.step << std::endl;
              simulator->replay_accepted_timestep(metadata);
              this->get_pcout() << "ASPECT_ADJOINT_REPLAY replay_complete step="
                                << metadata.step << std::endl;

              const LinearAlgebra::BlockVector replayed_solution = simulator->solution;
              const LinearAlgebra::BlockVector replayed_old_solution = simulator->old_solution;
              const LinearAlgebra::BlockVector replayed_old_old_solution = simulator->old_old_solution;
              const LinearAlgebra::BlockVector replayed_linearization_point =
                simulator->current_linearization_point;
              const LinearAlgebra::BlockVector replayed_system_rhs = simulator->system_rhs;
              const LinearAlgebra::BlockVector replayed_locally_owned =
                locally_owned_copy(replayed_solution);
              const double replayed_time = simulator->time;
              const double replayed_time_step = simulator->time_step;
              const double replayed_old_time_step = simulator->old_time_step;
              const unsigned int replayed_timestep_number = simulator->timestep_number;
              const double replayed_pressure_normalization_adjustment =
                simulator->last_pressure_normalization_adjustment;

              this->get_pcout() << "ASPECT_ADJOINT_REPLAY reference_load_begin step="
                                << metadata.step << std::endl;
              const LinearAlgebra::BlockVector reference = load_reference_state(record + 1);
              this->get_pcout() << "ASPECT_ADJOINT_REPLAY reference_load_complete step="
                                << metadata.step << std::endl;
              const LinearAlgebra::BlockVector reference_locally_owned =
                locally_owned_copy(reference);
              LinearAlgebra::BlockVector difference(replayed_locally_owned);
              difference.add(-1.0, reference_locally_owned);
              const double state_relative_difference = difference.l2_norm()
                                                       / std::max(reference_locally_owned.l2_norm(), 1e-50);
              this->get_pcout() << "ASPECT_ADJOINT_REPLAY state_difference_complete step="
                                << metadata.step << std::endl;
              const unsigned int velocity_block = this->introspection().block_indices.velocities;
              const unsigned int pressure_block = this->introspection().block_indices.pressure;
              const double velocity_relative_difference =
                relative_difference(replayed_locally_owned.block(velocity_block),
                                    reference_locally_owned.block(velocity_block));
              LinearAlgebra::Vector pressure_difference(replayed_locally_owned.block(pressure_block));
              pressure_difference.add(-1.0, reference_locally_owned.block(pressure_block));
              pressure_difference.add(-pressure_difference.mean_value());
              const double pressure_relative_difference =
                pressure_difference.l2_norm()
                / std::max(reference_locally_owned.block(pressure_block).l2_norm(), 1e-50);
              this->get_pcout() << "ASPECT_ADJOINT_REPLAY field_differences_complete step="
                                << metadata.step << std::endl;
              const bool metadata_matches =
                replayed_timestep_number == next_metadata.step
                && equal_time(replayed_time, next_metadata.time)
                && equal_time(replayed_time_step, next_metadata.time_step)
                && equal_time(replayed_old_time_step, next_metadata.old_time_step);

              this->get_pcout()
                  << "ASPECT_ADJOINT_REPLAY step=" << metadata.step
                  << " time=" << std::setprecision(16) << metadata.time
                  << " outgoing_dt=" << metadata.outgoing_time_step
                  << " resulting_time=" << replayed_time
                  << " state_l2=" << replayed_locally_owned.l2_norm()
                  << " relative_state_difference=" << state_relative_difference
                  << " relative_velocity_difference=" << velocity_relative_difference
                  << " gauge_relative_pressure_difference=" << pressure_relative_difference
                  << " nonlinear_iterations=" << simulator->nonlinear_iteration + 1
                  << " metadata_match=" << (metadata_matches ? 1 : 0)
                  << std::setprecision(6) << std::endl;
              AssertThrow(metadata_matches,
                          ExcMessage("Accepted-step replay produced inconsistent time-step metadata."));
              AssertThrow(state_relative_difference <= trajectory_replay_tolerance
                          && velocity_relative_difference <= trajectory_replay_tolerance
                          && pressure_relative_difference <= trajectory_replay_tolerance,
                          ExcMessage("Accepted-step replay did not reproduce the archived "
                                     "full-trajectory reference within the configured tolerance."));

              // Loading a minimal reference may have reconstructed a Stokes
              // state and therefore altered runtime matrices. Restore the
              // candidate exactly, then make the next replay step rebuild its
              // transient Stokes runtime state.
              simulator->solution = replayed_solution;
              simulator->old_solution = replayed_old_solution;
              simulator->old_old_solution = replayed_old_old_solution;
              simulator->current_linearization_point = replayed_linearization_point;
              simulator->system_rhs = replayed_system_rhs;
              simulator->time = replayed_time;
              simulator->time_step = replayed_time_step;
              simulator->old_time_step = replayed_old_time_step;
              simulator->timestep_number = replayed_timestep_number;
              simulator->last_pressure_normalization_adjustment =
                replayed_pressure_normalization_adjustment;
              simulator->rebuild_stokes_matrix = true;
              simulator->rebuild_stokes_preconditioner = true;
            }
        }
      catch (...)
        {
          simulator->solution = saved_solution;
          simulator->old_solution = saved_old_solution;
          simulator->old_old_solution = saved_old_old_solution;
          simulator->current_linearization_point = saved_linearization_point;
          simulator->system_rhs = saved_system_rhs;
          simulator->time = saved_time;
          simulator->time_step = saved_time_step;
          simulator->old_time_step = saved_old_time_step;
          simulator->timestep_number = saved_timestep_number;
          simulator->last_pressure_normalization_adjustment =
            saved_pressure_normalization_adjustment;
          trajectory_manager.stokes_recovery_solves = saved_stokes_recovery_solves;
          throw;
        }

      simulator->solution = saved_solution;
      simulator->old_solution = saved_old_solution;
      simulator->old_old_solution = saved_old_old_solution;
      simulator->current_linearization_point = saved_linearization_point;
      simulator->system_rhs = saved_system_rhs;
      simulator->time = saved_time;
      simulator->time_step = saved_time_step;
      simulator->old_time_step = saved_old_time_step;
      simulator->timestep_number = saved_timestep_number;
      simulator->last_pressure_normalization_adjustment =
        saved_pressure_normalization_adjustment;
      simulator->rebuild_stokes_matrix = true;
      simulator->rebuild_stokes_preconditioner = true;
      trajectory_manager.stokes_recovery_solves = saved_stokes_recovery_solves;
    }



    template <int dim>
    void
    Manager<dim>::compare_sparse_replay_state(const unsigned int record,
                                              const LinearAlgebra::BlockVector &state)
    {
      if (!trajectory_manager.has_reference_archive())
        return;

      LinearAlgebra::BlockVector reference_owned;
      reference_owned.reinit(this->introspection().index_sets.system_partitioning,
                             this->get_mpi_communicator());
      trajectory_manager.reference_archive().load(record, reference_owned,
                                                  this->introspection().block_indices.temperature,
                                                  this->introspection().block_indices.compositional_fields,
                                                  this->introspection().block_indices.velocities,
                                                  this->introspection().block_indices.pressure);
      LinearAlgebra::BlockVector state_owned;
      state_owned.reinit(this->introspection().index_sets.system_partitioning,
                         this->get_mpi_communicator());
      state_owned = state;
      LinearAlgebra::BlockVector difference(state_owned);
      difference.add(-1.0, reference_owned);
      const double state_error = difference.l2_norm()
                                 / std::max(reference_owned.l2_norm(), 1e-50);
      const unsigned int velocity_block = this->introspection().block_indices.velocities;
      const unsigned int pressure_block = this->introspection().block_indices.pressure;
      LinearAlgebra::Vector velocity_difference(state_owned.block(velocity_block));
      velocity_difference.add(-1.0, reference_owned.block(velocity_block));
      const double velocity_error = velocity_difference.l2_norm()
                                    / std::max(reference_owned.block(velocity_block).l2_norm(), 1e-50);
      LinearAlgebra::Vector pressure_difference(state_owned.block(pressure_block));
      pressure_difference.add(-1.0, reference_owned.block(pressure_block));
      pressure_difference.add(-pressure_difference.mean_value());
      const double pressure_error = pressure_difference.l2_norm()
                                    / std::max(reference_owned.block(pressure_block).l2_norm(), 1e-50);
      trajectory_manager.sparse_max_state_relative_difference = std::max(trajectory_manager.sparse_max_state_relative_difference, state_error);
      trajectory_manager.sparse_max_velocity_relative_difference = std::max(trajectory_manager.sparse_max_velocity_relative_difference, velocity_error);
      trajectory_manager.sparse_max_pressure_relative_difference = std::max(trajectory_manager.sparse_max_pressure_relative_difference, pressure_error);
      // The sparse reference stores the exact forward Stokes field, whereas
      // minimal sparse replay deterministically recomputes it. The velocity
      // comparison therefore has the documented Stokes-recovery tolerance;
      // the transported state and gauge-normalized pressure retain the
      // stricter trajectory-replay tolerance.
      if (!(state_error <= trajectory_replay_tolerance
            && velocity_error <= trajectory_stokes_recovery_tolerance
            && pressure_error <= trajectory_replay_tolerance))
        {
          // A Debug-mode AssertThrow formats a deep exception trace on every
          // rank. That formatter can itself consume unbounded memory after a
          // sparse replay mismatch. Emit the first numerical evidence once
          // and use the existing quiet collective exit path instead.
          this->get_pcout()
              << "ASPECT_ADJOINT_SPARSE_REPLAY_MISMATCH record=" << record
              << " state_relative_difference=" << state_error
              << " velocity_relative_difference=" << velocity_error
              << " gauge_pressure_relative_difference=" << pressure_error
              << " replay_tolerance=" << trajectory_replay_tolerance
              << " stokes_recovery_tolerance=" << trajectory_stokes_recovery_tolerance
              << std::endl;
          throw QuietException();
        }
    }



    template <int dim>
    void
    Manager<dim>::cleanup_trajectory_after_checkpoint(const AdjointRestartState &state)
    {
      AssertThrow(trajectory_manager.has_archive(), ExcInternalError());
      if (trajectory_manager.configuration().cleanup_policy != TrajectoryCleanupPolicy::safe_incremental)
        return;

      // Cleanup is allowed only after the checkpoint containing this reverse
      // history and next-step frontier has committed. Thus restart can always
      // resume without any payload reclaimed by the following transaction.
      TimerOutput::Scope cleanup_timer(this->get_computing_timer(),
                                       "Adjoint trajectory cleanup");
      const bool inject_cleanup_transaction =
        state.cleanup_frontier == trajectory_cleanup_failure_next_reverse_step;
      const Diagnostics::FailureInjection<Diagnostics::CleanupFailurePoint> injection =
      {
        inject_cleanup_transaction
        ? trajectory_cleanup_failure_injection
        : Diagnostics::CleanupFailurePoint::none,
        trajectory_cleanup_failure_rank
      };
      if (trajectory_manager.configuration().anchor_spacing == 1)
        {
          // In dense mode reverse steps decrease monotonically. Records at or
          // beyond cleanup_frontier have already been consumed and can be
          // removed while older records remain available one by one.
          const TrajectoryCleanupResult cleanup =
            trajectory_manager.archive().reclaim_records_after(state.cleanup_frontier,
                                                               injection);
          if (!cleanup.complete)
            {
              this->get_pcout() << "   Adjoint trajectory cleanup transaction failed collectively; "
                                << "the committed checkpoint remains authoritative." << std::endl;
              if (Utilities::MPI::n_mpi_processes(this->get_mpi_communicator()) > 1)
                throw QuietException();
              AssertThrow(false, ExcMessage("Adjoint trajectory cleanup transaction failed."));
            }
          std::cout << "ASPECT_ADJOINT_MPI_CLEANUP rank="
                    << Utilities::MPI::this_mpi_process(this->get_mpi_communicator())
                    << " next_reverse_step=" << state.next_reverse_step
                    << " cleanup_frontier=" << state.cleanup_frontier
                    << " local_records_deleted=" << cleanup.records_deleted
                    << " local_payload_files_deleted=" << cleanup.payload_files_deleted
                    << " local_bytes_reclaimed=" << cleanup.reclaimed.total_bytes()
                    << std::endl;
          return;
        }

      // Sparse mode reclaims whole replay intervals rather than individual
      // history states. Retain the lower anchor package selected by the next
      // reverse edge; newer anchor payloads are no longer reachable.
      const unsigned int required_anchor = trajectory_manager.required_anchor_for_reverse_step(
                                             state.next_reverse_step);
      AssertThrow(state.required_sparse_anchor == required_anchor,
                  ExcMessage("Sparse cleanup must retain the restart checkpoint's lower anchor."));
      const auto anchor_list = [this]()
      {
        std::vector<unsigned int> result;
        for (unsigned int record = 0; record < trajectory_manager.archive().size(); ++record)
          if (trajectory_manager.archive().has_record(record)
              && (record % trajectory_manager.configuration().anchor_spacing == 0
                  || record + 1 == trajectory_manager.archive().size()))
            result.push_back(record);
        return result;
      };
      const std::vector<unsigned int> live_before = anchor_list();
      const std::size_t archive_bytes_before = trajectory_manager.archive().shared_archive_bytes();
      const unsigned int segments_before = trajectory_manager.archive().shared_segment_count();
      inject_lifecycle_failure(Diagnostics::LifecycleFailurePoint::during_cleanup,
                               state.next_reverse_step);
      const TrajectoryCleanupResult cleanup =
        trajectory_manager.archive().reclaim_sparse_payloads_after(required_anchor,
                                                                   injection);
      if (!cleanup.complete)
        {
          this->get_pcout() << "   Sparse anchor cleanup transaction failed collectively; "
                            << "the committed checkpoint remains authoritative." << std::endl;
          if (Utilities::MPI::n_mpi_processes(this->get_mpi_communicator()) > 1)
            throw QuietException();
          AssertThrow(false, ExcMessage("Sparse anchor cleanup transaction failed."));
        }
      AssertThrow(trajectory_manager.archive().has_record(required_anchor),
                  ExcMessage("Sparse cleanup removed the durable restart anchor."));
      const std::vector<unsigned int> live_after = anchor_list();
      const std::size_t archive_bytes_after = trajectory_manager.archive().shared_archive_bytes();
      const unsigned int segments_after = trajectory_manager.archive().shared_segment_count();
      const auto format_anchors = [](const std::vector<unsigned int> &anchors)
      {
        std::string text;
        for (const unsigned int anchor : anchors)
          text += (text.empty() ? "" : ",") + Utilities::int_to_string(anchor);
        return text.empty() ? std::string("none") : text;
      };
      std::vector<unsigned int> reclaimed_anchors;
      for (const unsigned int anchor : live_before)
        if (std::find(live_after.begin(), live_after.end(), anchor) == live_after.end())
          reclaimed_anchors.push_back(anchor);
      this->get_pcout() << "ASPECT_ADJOINT_SPARSE_CLEANUP next_reverse_step="
                        << state.next_reverse_step
                        << " required_lower_anchor=" << required_anchor
                        << " live_anchors=" << format_anchors(live_after)
                        << " logically_reclaimed_anchors=" << format_anchors(reclaimed_anchors)
                        << " physically_retained_segments=" << segments_after
                        << " physically_deleted_segments=" << segments_before - segments_after
                        << " archive_bytes=" << archive_bytes_after
                        << " reclaimed_bytes=" << archive_bytes_before - archive_bytes_after
                        << std::endl;
    }



    template <int dim>
    void
    Manager<dim>::materialize_forward_replay_window(const unsigned int record)
    {
      AssertThrow(trajectory_manager.configuration().anchor_spacing > 1 && trajectory_manager.has_archive(),
                  ExcInternalError());
      // Materialize one interval [anchor,end] on demand. The cached window also
      // includes anchor-1 and anchor-2, because the first replayed accepted
      // step and a BDF2 reverse operator both require this preceding history.
      const auto interval = trajectory_manager.sparse_interval_for_record(record);
      const unsigned int anchor = interval.first;
      const unsigned int end = interval.second;
      for (const ForwardReplayWindow &window : trajectory_manager.forward_replay_windows)
        if (record >= window.anchor && record <= window.end)
          return;

      const auto load_payload = [this](const unsigned int state_index)
      {
        // TrajectoryStore loads owned entries. Promote them to a ghosted FE
        // read view, and in minimal mode deterministically recover the omitted
        // quasi-static velocity/pressure before replay uses the state.
        LinearAlgebra::BlockVector owned;
        owned.reinit(this->introspection().index_sets.system_partitioning,
                     this->get_mpi_communicator());
        trajectory_manager.archive().load(state_index, owned,
                                          this->introspection().block_indices.temperature,
                                          this->introspection().block_indices.compositional_fields,
                                          this->introspection().block_indices.velocities,
                                          this->introspection().block_indices.pressure);
        LinearAlgebra::BlockVector result;
        result.reinit(this->introspection().index_sets.system_partitioning,
                      this->introspection().index_sets.system_relevant_partitioning,
                      this->get_mpi_communicator());
        result = owned;
        if (trajectory_manager.archive().mode() == TrajectoryStorageMode::minimal)
          recover_stokes_state(state_index, result);
        return result;
      };

      ForwardReplayWindow window;
      window.anchor = anchor;
      window.end = end;
      const unsigned int history_begin = anchor > 1 ? anchor - 2 : 0;
      for (unsigned int state_index = history_begin; state_index <= anchor; ++state_index)
        {
          AssertThrow(trajectory_manager.archive().has_record(state_index),
                      ExcMessage("Sparse anchor package is missing required BDF history."));
          window.states.push_back({state_index, load_payload(state_index)});
        }
      const auto state_at = [&window](const unsigned int state_index)
                            -> const LinearAlgebra::BlockVector &
      {
        for (const ReconstructedTrajectoryState &state : window.states)
          if (state.record == state_index)
            return state.state;
        AssertThrow(false, ExcInternalError());
        return window.states.front().state;
      };

      const ReplayStepMetadata &anchor_metadata = trajectory_manager.archive().metadata(anchor);
      // Hydrate Simulator's three forward history slots at the anchor exactly
      // as they appeared during the original run. Runtime matrices are marked
      // stale because they are not archived and belong to another time level.
      simulator->solution = state_at(anchor);
      simulator->old_solution = state_at(anchor > 0 ? anchor - 1 : anchor);
      simulator->old_old_solution = state_at(anchor > 1 ? anchor - 2 : anchor);
      simulator->current_linearization_point = simulator->solution;
      simulator->time = anchor_metadata.time;
      simulator->time_step = anchor_metadata.time_step;
      simulator->old_time_step = anchor_metadata.old_time_step;
      simulator->timestep_number = anchor_metadata.step;
      simulator->last_pressure_normalization_adjustment = 0.0;
      simulator->rebuild_stokes_matrix = true;
      simulator->rebuild_stokes_preconditioner = true;
      ++trajectory_manager.sparse_anchor_load_count;
      ++trajectory_manager.sparse_interval_replay_count;
      report_mpi_adjoint_memory_checkpoint(
        "sparse replay: anchor state hydrated before accepted-step replay",
        &simulator->solution);

      for (unsigned int transition = anchor; transition < end; ++transition)
        {
          // replay_accepted_timestep follows the recorded accepted dt rather
          // than asking the time-step manager to choose a new one. Its output
          // state transition+1 is retained in memory and optionally compared
          // with the diagnostic dense archive.
          report_mpi_adjoint_memory_checkpoint(
            "sparse replay: before accepted-step replay "
            + Utilities::int_to_string(transition), &simulator->solution);
          simulator->replay_accepted_timestep(trajectory_manager.archive().metadata(transition));
          report_mpi_adjoint_memory_checkpoint(
            "sparse replay: after accepted-step replay "
            + Utilities::int_to_string(transition), &simulator->solution);
          window.states.push_back({transition + 1, simulator->solution});
          ++trajectory_manager.sparse_replay_transition_count;
          report_mpi_adjoint_memory_checkpoint(
            "sparse replay: before dense-reference comparison "
            + Utilities::int_to_string(transition + 1), &window.states.back().state);
          compare_sparse_replay_state(transition + 1, window.states.back().state);
          report_mpi_adjoint_memory_checkpoint(
            "sparse replay: after dense-reference comparison "
            + Utilities::int_to_string(transition + 1), &window.states.back().state);
        }
      trajectory_manager.sparse_peak_window_state_count = std::max(
                                                            trajectory_manager.sparse_peak_window_state_count,
                                                            static_cast<unsigned int>(window.states.size()));
      this->get_pcout() << "ASPECT_ADJOINT_SPARSE_REPLAY anchor=" << anchor
                        << " end=" << end
                        << " anchor_loads=" << trajectory_manager.sparse_anchor_load_count
                        << " replayed_transitions=" << (end - anchor)
                        << " max_state_relative_difference="
                        << trajectory_manager.sparse_max_state_relative_difference
                        << " max_velocity_relative_difference="
                        << trajectory_manager.sparse_max_velocity_relative_difference
                        << " max_gauge_pressure_relative_difference="
                        << trajectory_manager.sparse_max_pressure_relative_difference << std::endl;
      trajectory_manager.forward_replay_windows.emplace_back(std::move(window));
      // Reverse traversal can straddle one interval boundary and request
      // history from the previous window. Two windows are therefore sufficient
      // and bound memory independently of total trajectory length.
      while (trajectory_manager.forward_replay_windows.size() > 2)
        trajectory_manager.forward_replay_windows.pop_front();
    }



    template <int dim>
    const LinearAlgebra::BlockVector &
    Manager<dim>::load_reconstructed_state(const unsigned int record)
    {
      AssertThrow(trajectory_manager.has_archive(), ExcInternalError());
      if (trajectory_manager.configuration().anchor_spacing > 1)
        {
          // Sparse mode delegates ownership to replay windows; references stay
          // valid while the deque retains the containing window. The caller
          // obtains all three snapshot references before later traversal can
          // evict an older window.
          for (const ForwardReplayWindow &window : trajectory_manager.forward_replay_windows)
            for (const ReconstructedTrajectoryState &state : window.states)
              if (state.record == record)
                return state.state;
          materialize_forward_replay_window(record);
          for (const ForwardReplayWindow &window : trajectory_manager.forward_replay_windows)
            for (const ReconstructedTrajectoryState &state : window.states)
              if (state.record == record)
                return state.state;
          AssertThrow(false, ExcMessage("Sparse replay window did not produce the requested state."));
        }
      // Dense archives still cache the last three loaded states to avoid
      // repeated MPI-IO for q_n,q_{n-1},q_{n-2} across adjacent reverse steps.
      for (const auto &cached : trajectory_manager.reverse_state_cache)
        if (cached.record == record)
          {
            report_mpi_adjoint_memory_checkpoint(
              "replay record " + Utilities::int_to_string(record) + " cache hit",
              &cached.state);
            return cached.state;
          }

      ReconstructedTrajectoryState reconstructed;
      reconstructed.record = record;
      report_mpi_adjoint_memory_checkpoint(
        "before replay record " + Utilities::int_to_string(record) + " materialization");
      LinearAlgebra::BlockVector locally_owned_state;
      locally_owned_state.reinit(this->introspection().index_sets.system_partitioning,
                                 this->get_mpi_communicator());
      locally_owned_state = 0.0;
      report_mpi_adjoint_memory_checkpoint(
        "after replay record " + Utilities::int_to_string(record)
        + " owned-vector reinit", &locally_owned_state);

      {
        TimerOutput::Scope read_timer(this->get_computing_timer(),
                                      "Adjoint trajectory read");
        trajectory_manager.archive().load(record,
                                          locally_owned_state,
                                          this->introspection().block_indices.temperature,
                                          this->introspection().block_indices.compositional_fields,
                                          this->introspection().block_indices.velocities,
                                          this->introspection().block_indices.pressure);
      }
      report_mpi_adjoint_memory_checkpoint(
        "after replay record " + Utilities::int_to_string(record)
        + " owned payload load", &locally_owned_state);

      reconstructed.state.reinit(this->introspection().index_sets.system_partitioning,
                                 this->introspection().index_sets.system_relevant_partitioning,
                                 this->get_mpi_communicator());
      reconstructed.state = locally_owned_state;
      report_mpi_adjoint_memory_checkpoint(
        "after replay record " + Utilities::int_to_string(record)
        + " ghost-view materialization", &reconstructed.state);

      if (trajectory_manager.archive().mode() == TrajectoryStorageMode::minimal)
        recover_stokes_state(record, reconstructed.state);

      // The reverse operator needs at most q_n, q_(n-1), and q_(n-2). Keep
      // exactly those recently used reconstructed states; the trajectory
      // itself remains disk-backed and never embeds this history.
      if (trajectory_manager.reverse_state_cache.size() == 3)
        trajectory_manager.reverse_state_cache.erase(trajectory_manager.reverse_state_cache.begin());
      trajectory_manager.reverse_state_cache.emplace_back(std::move(reconstructed));
      report_mpi_adjoint_memory_checkpoint(
        "after replay record " + Utilities::int_to_string(record)
        + " reverse-cache insertion",
        &trajectory_manager.reverse_state_cache.back().state);
      return trajectory_manager.reverse_state_cache.back().state;
    }



    template <int dim>
    ForwardStateView<dim>
    Manager<dim>::reconstruct_forward_snapshot(const unsigned int record)
    {
      AssertThrow(trajectory_manager.has_archive(), ExcInternalError());
      AssertIndexRange(record, trajectory_manager.archive().size());

      TimerOutput::Scope reconstruction_timer(this->get_computing_timer(),
                                              "Adjoint trajectory state reconstruction");
      report_mpi_adjoint_memory_checkpoint(
        "before forward snapshot reconstruction at record "
        + Utilities::int_to_string(record));
      // ForwardStateView is a non-owning description of one discrete time
      // level. At the first two records, unavailable histories collapse onto
      // the oldest physical state exactly as the forward BDF startup did.
      const LinearAlgebra::BlockVector &solution = load_reconstructed_state(record);
      const LinearAlgebra::BlockVector &old_solution =
        load_reconstructed_state(record > 0 ? record - 1 : record);
      const LinearAlgebra::BlockVector &old_old_solution =
        load_reconstructed_state(record > 1 ? record - 2 : record);
      const ReplayStepMetadata &metadata = trajectory_manager.archive().metadata(record);
      ForwardStateView<dim> snapshot(solution,
                                     old_solution,
                                     old_old_solution,
                                     record > 0 ? old_solution : solution,
                                     solution,
                                     metadata.time,
                                     metadata.time_step,
                                     metadata.old_time_step,
                                     metadata.step);
      if (trajectory_manager.configuration().anchor_spacing > 1 && record > 1
          && record % trajectory_manager.configuration().anchor_spacing == 1)
        this->get_pcout() << "ASPECT_ADJOINT_SPARSE_BOUNDARY snapshot=" << record
                          << " history=" << record - 1 << ',' << record - 2
                          << " provider=forward_replay_window" << std::endl;
      report_mpi_adjoint_memory_checkpoint(
        "after replay records for forward snapshot " + Utilities::int_to_string(record),
        &snapshot.solution.get());

      // The advection operator was assembled from the prior physical state.
      // For the supported Picard Stokes model the Stokes material point is
      // determined by the current T,C state, so no separately archived
      // linearization vector is required.
      // Rebuild the primal graph from the fixed mesh, the recorded time, and
      // configured boundary managers. The temporary snapshot keeps a copy
      // only while a legacy adjoint helper needs to homogenize it; it is never
      // an archived constraint record.
      simulator->solution = snapshot.solution.get();
      simulator->old_solution = snapshot.old_solution.get();
      simulator->old_old_solution = snapshot.old_old_solution.get();
      simulator->current_linearization_point = snapshot.advection_linearization_point.get();
      simulator->time = snapshot.time;
      simulator->time_step = snapshot.time_step;
      simulator->old_time_step = snapshot.old_time_step;
      simulator->timestep_number = snapshot.timestep_number;
      simulator->last_pressure_normalization_adjustment = 0.0;
      report_mpi_adjoint_memory_checkpoint(
        "before compute_current_constraints for replay record "
        + Utilities::int_to_string(record),
        &snapshot.solution.get());
      simulator->compute_current_constraints();
      // Boundary constraints can depend on recorded time. Recompute their
      // primal graph and copy it into the snapshot so all subsequent transpose
      // scatters act on the same constrained discrete space as the forward
      // accepted step.
      snapshot.constraints.reinit(simulator->current_constraints.get_local_lines(),
                                  this->introspection().index_sets.system_relevant_set);
      snapshot.constraints.copy_from(simulator->current_constraints);
      report_mpi_adjoint_memory_checkpoint(
        "after compute_current_constraints for replay record "
        + Utilities::int_to_string(record),
        &snapshot.solution.get());

      return snapshot;
    }



    template <int dim>
    void
    Manager<dim>::report_trajectory_statistics(const std::string &stage) const
    {
      if (!trajectory_manager.has_archive())
        return;

      std::size_t local_cache_memory = 0;
      for (const auto &cached : trajectory_manager.reverse_state_cache)
        local_cache_memory += cached.state.memory_consumption();
      const std::size_t local_current_record_memory =
        (trajectory_manager.reverse_state_cache.empty() ? 0
         : trajectory_manager.reverse_state_cache.back().state.memory_consumption());

      diagnostics_manager.report_trajectory(
        trajectory_manager.archive(),
        stage,
        local_cache_memory,
        local_current_record_memory,
        trajectory_manager.stokes_recovery_solves,
        trajectory_manager.sparse_peak_window_state_count,
        trajectory_manager.sparse_replay_transition_count,
        trajectory_manager.sparse_restart_replay_transition_count);
    }



    template <int dim>
    void
    Manager<dim>::report_mpi_adjoint_memory_checkpoint(
      const std::string &stage,
      const LinearAlgebra::BlockVector *state) const
    {
      const LinearAlgebra::BlockVector &reported_state =
        (state != nullptr ? *state : simulator->solution);
      diagnostics_manager.report_memory_checkpoint(stage,
                                                   reported_state,
                                                   trajectory_manager.reverse_state_cache.size());
    }
