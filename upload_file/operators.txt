#ifndef _aspect_adjoint_operators_advection_adjoint_calculator_h
#define _aspect_adjoint_operators_advection_adjoint_calculator_h

#include <aspect/adjoint/core/state.h>
#include <aspect/advection_field.h>
#include <aspect/global.h>
#include <aspect/simulator_access.h>

#include <vector>

namespace aspect
{
  template <int dim> class Simulator;

  namespace Adjoint
  {
    namespace Diagnostics
    {
      template <int dim> class AdvectionOperatorChecks;
    }

    /**
     * Transpose actions for finite-element advection equations in the
     * time-dependent adjoint workflow.
     *
     * In the forward run, one accepted advection step can be viewed
     * schematically as
     *
     * @code
     *   A_n(q_n, u_n, ...) q_n = b_n(q_{n-1}, q_{n-2}, u^*_n, ...)
     * @endcode
     *
     * where @p A_n is the assembled advection matrix at the restored
     * linearization point and the right hand side contains BDF history,
     * extrapolated velocity, material prefactors, and stabilization terms.
     * Given a seed \f$\bar q_n\f$ from the later adjoint calculation, this
     * helper solves \f$A_n^T z_n = \bar q_n\f$ and then applies the transpose
     * of each right-hand-side dependence to the older time levels
     * \f$q_{n-1}\f$ and, for BDF2, \f$q_{n-2}\f$.
     *
     * The vectors passed to the public propagation functions are full-system
     * block vectors. Only the requested field or Stokes blocks are touched,
     * but full vectors are used so that constraints, material-model queries,
     * and ASPECT's existing assembly utilities can be reused without building
     * a separate reduced adjoint data structure.
     *
     * @ingroup Adjoint
     */
    template <int dim>
    class AdvectionAdjointCalculator : public SimulatorAccess<dim>
    {
      public:
        /** Preconditioners for the transposed advection solve. */
        enum class Preconditioner
        {
          none,
          forward_ilu_experimental,
          transpose_ilu
        };

        explicit
        AdvectionAdjointCalculator(Simulator<dim> &simulator);

        /** Select the preconditioner used by subsequent transposed solves. */
        void
        set_preconditioner(const Preconditioner preconditioner);

        /** Enable one none-versus-ILU(A^T) diagnostic per field type. */
        void
        set_run_preconditioner_comparison(const bool run_comparison);

        /**
         * Restore @p snapshot, assemble the forward advection operator for
         * @p advection_field, solve \f$A_n^T z_n = \bar q_n\f$, and add all
         * supported transposed right-hand-side contributions to older
         * adjoints.
         *
         * @p current_field_adjoint is the seed \f$\bar q_n\f$ accumulated at
         * the current accepted state. @p previous_field_adjoint receives
         * contributions for \f$q_{n-1}\f$ and
         * @p previous_previous_field_adjoint receives contributions for
         * \f$q_{n-2}\f$ when the restored step used BDF2. The Stokes right
         * hand sides collect the transpose of velocity dependence; they are
         * solved later by Manager because velocity is constrained by the
         * Stokes equation, not by the advection equation itself.
         */
        void
        propagate_advection_field_adjoint(const AdvectionField &advection_field,
                                          const ForwardStateView<dim> &snapshot,
                                          const LinearAlgebra::BlockVector &current_field_adjoint,
                                          LinearAlgebra::BlockVector &previous_field_adjoint,
                                          LinearAlgebra::BlockVector *previous_previous_field_adjoint,
                                          LinearAlgebra::BlockVector &previous_stokes_adjoint_rhs,
                                          LinearAlgebra::BlockVector *previous_previous_stokes_adjoint_rhs);

        /**
         * Temperature-specific wrapper around propagate_advection_field_adjoint().
         *
         * The temperature equation contains material prefactors such as
         * \f$\rho C_p\f$. When these depend on temperature or composition,
         * the transpose of the temperature advection residual also feeds
         * adjoints of older temperature and composition states. The optional
         * composition-vector arguments provide those slices of the reverse
         * time history.
         */
        void
        propagate_temperature_adjoint(const ForwardStateView<dim> &snapshot,
                                      const LinearAlgebra::BlockVector &current_temperature_adjoint,
                                      LinearAlgebra::BlockVector &previous_temperature_adjoint,
                                      LinearAlgebra::BlockVector *previous_previous_temperature_adjoint,
                                      LinearAlgebra::BlockVector &previous_stokes_adjoint_rhs,
                                      LinearAlgebra::BlockVector *previous_previous_stokes_adjoint_rhs,
                                      std::vector<LinearAlgebra::BlockVector> *previous_composition_adjoints = nullptr,
                                      std::vector<LinearAlgebra::BlockVector> *previous_previous_composition_adjoints = nullptr);

      private:
        friend class Diagnostics::AdvectionOperatorChecks<dim>;

        /**
         * Test-driver implementation of a finite-difference dot-product check
         * for the velocity dependence of one restored advection solve.
         */
        void
        run_velocity_transpose_diagnostic(const AdvectionField &advection_field,
                                          const ForwardStateView<dim> &snapshot,
                                          const LinearAlgebra::BlockVector &current_field_adjoint,
                                          const double epsilon) const;

        /**
         * Run the transposed advection solve and verify that constrained
         * adjoint DoFs on @p advection_field are exactly homogeneous.
         */
        void
        run_transposed_advection_constraint_diagnostic(const AdvectionField &advection_field,
                                                       const ForwardStateView<dim> &snapshot,
                                                       const LinearAlgebra::BlockVector &current_field_adjoint) const;

        /**
         * Run the isolated R_q -> nu_art coefficient-seed diagnostic for one
         * restored advection solve.
         */
        void
        run_artificial_viscosity_coefficient_seed_diagnostic(const AdvectionField &advection_field,
                                                             const ForwardStateView<dim> &snapshot,
                                                             const LinearAlgebra::BlockVector &current_field_adjoint,
                                                             const double epsilon) const;

        /**
         * Run the standalone entropy-viscosity estimator transpose diagnostic
         * for one restored advection solve.
         */
        void
        run_artificial_viscosity_estimator_transpose_diagnostic(const AdvectionField &advection_field,
                                                                const ForwardStateView<dim> &snapshot,
                                                                const double epsilon) const;

        /**
         * Run the frozen-denominator local entropy-residual transpose
         * diagnostic for one restored advection solve.
         */
        void
        run_local_entropy_residual_transpose_diagnostic(const AdvectionField &advection_field,
                                                        const ForwardStateView<dim> &snapshot,
                                                        const double epsilon) const;

        /**
         * Run the scalar denominator diagnostic for global_entropy_variation.
         */
        void
        run_global_entropy_denominator_scalar_diagnostic(const AdvectionField &advection_field,
                                                         const ForwardStateView<dim> &snapshot,
                                                         const double epsilon) const;

        /**
         * Run the standalone frozen-average_field transpose diagnostic for
         * global_entropy_variation(field_history).
         */
        void
        run_global_entropy_variation_transpose_diagnostic(const AdvectionField &advection_field,
                                                          const ForwardStateView<dim> &snapshot,
                                                          const double epsilon) const;

        Preconditioner preconditioner = Preconditioner::transpose_ilu;
        bool run_preconditioner_comparison = false;
        mutable bool temperature_preconditioner_comparison_completed = false;
        mutable bool composition_preconditioner_comparison_completed = false;

        /**
         * Solve the transposed field block for one restored advection system.
         *
         * This is the local algebraic step
         * \f$z=A_n^{-T}\bar q_n\f$. The returned vector is full-system shaped,
         * but only the block belonging to @p advection_field is populated.
         * Constrained degrees of freedom are set to zero because adjoint
         * variables satisfy homogeneous versions of the primal constraints.
         */
        void
        solve_transposed_advection_system(const AdvectionField &advection_field,
                                          const LinearAlgebra::BlockVector &rhs,
                                          LinearAlgebra::BlockVector &solution) const;

        /**
         * Add the transpose of BDF history mass terms to older field states.
         *
         * For the first time step this contributes only to q_{n-1}. For BDF2
         * it splits the seed between q_{n-1} and q_{n-2} using the same
         * variable-time-step coefficients as the forward advection assembly.
         */
        void
        add_history_transpose(const AdvectionField &advection_field,
                              const ForwardStateView<dim> &snapshot,
                              const LinearAlgebra::BlockVector &z,
                              LinearAlgebra::BlockVector &previous_field_adjoint,
                              LinearAlgebra::BlockVector *previous_previous_field_adjoint) const;

        /**
         * Add the transpose of the extrapolated velocity term in the
         * advection residual.
         *
         * The forward residual uses a velocity linearization/extrapolation
         * assembled from earlier Stokes states. This action maps the
         * advection transpose solution @p z into Stokes adjoint right hand
         * sides at q_{n-1} and, for BDF2, q_{n-2}.
         */
        void
        add_velocity_transpose(const AdvectionField &advection_field,
                               const ForwardStateView<dim> &snapshot,
                               const LinearAlgebra::BlockVector &z,
                               LinearAlgebra::BlockVector &previous_stokes_adjoint_rhs,
                               LinearAlgebra::BlockVector *previous_previous_stokes_adjoint_rhs) const;

        /**
         * Add the transpose of the temperature advection prefactor
         * \f$\rho C_p\f$.
         *
         * This term is separate from add_history_transpose() because its
         * derivative passes through the material model. The material-model
         * VJP scatters back to temperature and composition inputs at the old
         * time levels used by the prefactor.
         */
        void
        add_temperature_material_prefactor_transpose(const ForwardStateView<dim> &snapshot,
                                                     const LinearAlgebra::BlockVector &z,
                                                     LinearAlgebra::BlockVector &previous_temperature_adjoint,
                                                     LinearAlgebra::BlockVector *previous_previous_temperature_adjoint,
                                                     std::vector<LinearAlgebra::BlockVector> *previous_composition_adjoints,
                                                     std::vector<LinearAlgebra::BlockVector> *previous_previous_composition_adjoints) const;

        Simulator<dim> *simulator;
    };
  }
}

#ifndef _aspect_adjoint_operators_artificial_viscosity_adjoint_h
#define _aspect_adjoint_operators_artificial_viscosity_adjoint_h

#include <aspect/adjoint/core/state.h>
#include <aspect/advection_field.h>
#include <aspect/global.h>
#include <aspect/simulator_access.h>

#include <vector>

namespace aspect
{
  template <int dim> class Simulator;

  namespace Adjoint
  {
    template <int dim> class AdvectionAdjointCalculator;

    /**
     * Reverse-mode helper for the entropy-viscosity stabilization coefficient
     * used by finite-element advection equations.
     *
     * The forward advection residual sees artificial viscosity in two layers:
     * first the residual contains a cellwise diffusion coefficient
     * \f$\nu_{\mathrm{art},K}\f$, and second that coefficient is produced by
     * the entropy-viscosity estimator from old field values, old velocities,
     * extrema, and residual norms. The time-dependent adjoint reverses those
     * layers in the opposite order:
     *
     * - assemble_coefficient_seed() computes the seed
     *   \f$\bar\nu_{\mathrm{art},K}\f$ generated by the diffusion term after
     *   the transposed advection equation has been solved.
     * - add_estimator_transpose() propagates that seed through the supported
     *   active branches of the entropy-viscosity estimator to q_{n-1},
     *   q_{n-2}, and Stokes adjoint right hand sides.
     *
     * The current implementation assumes that each active cell stays on a
     * single branch of \f$\max(k,\nu_{\mathrm{art}})\f$ for all quadrature
     * points. Mixed branch cells need an additional quadrature-point active
     * set cache before their exact transpose can be enabled.
     */
    template <int dim>
    class ArtificialViscosityAdjoint : public SimulatorAccess<dim>
    {
      public:
        explicit
        ArtificialViscosityAdjoint(Simulator<dim> &simulator);

        /**
         * Assemble the cellwise adjoint seed for the artificial-viscosity
         * coefficient at @p snapshot:
         *
         * \f[
         * \bar\nu_{\mathrm{art},K}
         * =
         * -\Delta t
         * \int_K \chi_{\nu,Kq}\nabla z\cdot\nabla T\,d\Omega.
         * \f]
         *
         * The returned vector is indexed by active cell index. Entries for
         * non-locally-owned cells are left at zero.
         */
        std::vector<double>
        assemble_coefficient_seed(const AdvectionField &advection_field,
                                  const ForwardStateView<dim> &snapshot,
                                  const LinearAlgebra::BlockVector &transpose_solution) const;

        /**
         * Reverse the supported active branches of the entropy-viscosity
         * estimator and add their contributions to the older time levels.
         *
         * @p coefficient_seed is indexed by active cell and represents
         * \f$\bar\nu_{\mathrm{art},K}\f$. Field adjoints receive derivatives
         * with respect to old/old-old advected values and their gradients.
         * Stokes right hand sides receive derivatives with respect to old
         * velocities and strain rates. For temperature, the optional
         * composition adjoints receive material-prefactor derivatives created
         * inside the estimator.
         */
        void
        add_estimator_transpose(const AdvectionField &advection_field,
                                const ForwardStateView<dim> &snapshot,
                                const std::vector<double> &coefficient_seed,
                                LinearAlgebra::BlockVector &previous_field_adjoint,
                                LinearAlgebra::BlockVector *previous_previous_field_adjoint,
                                LinearAlgebra::BlockVector &previous_stokes_adjoint_rhs,
                                LinearAlgebra::BlockVector *previous_previous_stokes_adjoint_rhs,
                                std::vector<LinearAlgebra::BlockVector> *previous_composition_adjoints = nullptr,
                                std::vector<LinearAlgebra::BlockVector> *previous_previous_composition_adjoints = nullptr) const;

      private:
        friend class AdvectionAdjointCalculator<dim>;

        /**
         * Run a local finite-difference dot-product diagnostic for the
         * derivative of the assembled advection residual with respect to an
         * independent cellwise artificial-viscosity coefficient. This isolates
         * the R_q -> nu_art layer from the entropy-viscosity estimator itself.
         */
        void
        run_coefficient_seed_diagnostic(const AdvectionField &advection_field,
                                        const ForwardStateView<dim> &snapshot,
                                        const LinearAlgebra::BlockVector &transpose_solution,
                                        const std::vector<double> &coefficient_seed,
                                        const double epsilon) const;

        /**
         * Run a finite-difference dot-product diagnostic for the currently
         * implemented transpose of the entropy-viscosity estimator. The test
         * starts from an artificial cellwise seed @f$\bar\nu_{\rm art}@f$ and
         * compares add_estimator_transpose() with a recomputation of the
         * forward estimator under perturbations of old/old-old field and
         * velocity states.
         */
        void
        run_estimator_transpose_diagnostic(const AdvectionField &advection_field,
                                           const ForwardStateView<dim> &snapshot,
                                           const double epsilon) const;

        /**
         * Run a frozen-denominator diagnostic for the local
         * max-residual/entropy-deviation numerator of the alpha=2
         * entropy-viscosity estimator.
         */
        void
        run_local_entropy_residual_transpose_diagnostic(const AdvectionField &advection_field,
                                                        const ForwardStateView<dim> &snapshot,
                                                        const double epsilon) const;

        /**
         * Run a scalar diagnostic for the denominator derivative
         * nu_entropy = c_R h^2 Rmax / global_entropy_variation.
         */
        void
        run_global_entropy_denominator_scalar_diagnostic(const AdvectionField &advection_field,
                                                         const ForwardStateView<dim> &snapshot,
                                                         const double epsilon) const;

        /**
         * Run a standalone diagnostic for
         * global_entropy_variation(field_history), keeping average_field
         * frozen at the value supplied by the extrapolated field range.
         */
        void
        run_global_entropy_variation_transpose_diagnostic(const AdvectionField &advection_field,
                                                          const ForwardStateView<dim> &snapshot,
                                                          const double epsilon) const;

        /**
         * Add the transpose of global_entropy_variation(field_history), with
         * average_field frozen at the extrapolated-range value.
         */
        void
        add_global_entropy_variation_transpose(const AdvectionField &advection_field,
                                               const double bar_entropy_variation,
                                               double &bar_average_field,
                                               LinearAlgebra::BlockVector &previous_field_adjoint,
                                               LinearAlgebra::BlockVector *previous_previous_field_adjoint) const;

        /**
         * Add the transpose of
         * average_field = 0.5 * (min(extrapolated field) + max(extrapolated field))
         * to the old/old-old advection field states.
         */
        void
        add_extrapolated_field_average_transpose(const AdvectionField &advection_field,
                                                 const ForwardStateView<dim> &snapshot,
                                                 const double bar_average_field,
                                                 LinearAlgebra::BlockVector &previous_field_adjoint,
                                                 LinearAlgebra::BlockVector *previous_previous_field_adjoint) const;

        Simulator<dim> *simulator;
    };
  }
}

#ifndef _aspect_adjoint_operators_boundary_conditions_h
#define _aspect_adjoint_operators_boundary_conditions_h

#include <aspect/adjoint/core/state.h>

namespace aspect
{
  namespace Adjoint
  {
    /**
     * Rebuild the constraint space used by an adjoint calculation from an
     * accepted forward snapshot.  The forward constraint graph is retained
     * exactly (in particular the affine Cartesian-component relations used by
     * curved free-slip boundaries), while every inhomogeneity is removed.
     *
     * Fixed boundary data are not controls in the v1 adjoint. Consequently
     * their variations, and hence the corresponding adjoint essential values,
     * are zero. Fixed heat-flux boundaries do not contribute constraint lines
     * and therefore remain natural boundaries here.
     */
    template <int dim>
    class BoundaryConditions
    {
      public:
        void
        build_homogeneous_constraints(const ForwardStateView<dim> &snapshot,
                                      AffineConstraints<double> &constraints) const
        {
          build_homogeneous_constraints(snapshot.constraints, constraints);
        }

        void
        build_homogeneous_constraints(const AffineConstraints<double> &forward_constraints,
                                      AffineConstraints<double> &constraints) const
        {
          constraints.copy_from(forward_constraints);

          // Do not use set_zero() on a solution vector here. A free-slip
          // constraint on a curved face can be an affine relation between
          // velocity components; preserving its entries is essential.
          for (const auto &line : constraints.get_lines())
            constraints.set_inhomogeneity(line.index, 0.0);

          // Keep this verification next to the transformation: it guards
          // against a future "homogenization" shortcut that would retain
          // only constrained indices and silently drop affine entries (for
          // example curved free-slip, hanging-node, or periodic relations).
          const auto &forward_lines = forward_constraints.get_lines();
          const auto &homogeneous_lines = constraints.get_lines();
          AssertThrow(forward_lines.size() == homogeneous_lines.size(),
                      ExcInternalError());
          for (unsigned int line_number = 0; line_number < forward_lines.size(); ++line_number)
            {
              AssertThrow(forward_lines[line_number].index
                          == homogeneous_lines[line_number].index,
                          ExcInternalError());
              AssertThrow(forward_lines[line_number].entries
                          == homogeneous_lines[line_number].entries,
                          ExcInternalError());
              AssertThrow(homogeneous_lines[line_number].inhomogeneity == 0.0,
                          ExcInternalError());
            }

          constraints.close();
        }

        void
        build_stokes_adjoint_constraints(const ForwardStateView<dim> &snapshot,
                                         AffineConstraints<double> &constraints) const
        {
          build_homogeneous_constraints(snapshot, constraints);
        }

        void
        build_temperature_adjoint_constraints(const ForwardStateView<dim> &snapshot,
                                              AffineConstraints<double> &constraints) const
        {
          build_homogeneous_constraints(snapshot, constraints);
        }
    };
  }
}

#include <aspect/adjoint/core/kernel_calculator.h>
#include <aspect/postprocess/forward_boundary_reduced_stress.h>
#include <aspect/simulator_access.h>

namespace aspect
{
  template <int dim> class Simulator;

  namespace Adjoint
  {
    /**
     * Apply the transpose and the explicit material derivatives of the
     * dynamic-topography postprocessing chain at either the top or bottom
     * boundary. The boundary kind is explicit so that the CBF denominator,
     * outward normal, and pressure-mean correction remain tied to the
     * selected physical boundary.
     *
     * The supplied seed is a ghosted, distributed full-system block vector.
     * Its meaningful entries are temperature-component DoFs on the selected
     * boundary face
     * support points and contain the assembled derivative with respect to
     * topography. @p pressure_weight is the corresponding
     * quadrature-space seed contracted with
     * \f$((rho-rho_{above})|g|)^{-1}\f$, which is needed by the transpose of
     * the selected-boundary mean pressure subtraction.
     */
    template <int dim>
    class DynamicTopographyBoundaryOperator : public SimulatorAccess<dim>
    {
      public:
        explicit DynamicTopographyBoundaryOperator(const Simulator<dim> &simulator);

        /** Add the state transpose to an existing physical adjoint RHS. */
        void
        add_state_transpose(const LinearAlgebra::BlockVector &topography_adjoint,
                            const double boundary_area,
                            const double pressure_weight,
                            const double density_exterior,
                            LinearAlgebra::BlockVector &rhs,
                            const Postprocess::DynamicTopographyBoundary boundary) const;

        /** Add direct raw-density and raw-viscosity kernel contributions. */
        void
        add_direct_kernel_contributions(const LinearAlgebra::BlockVector &topography_adjoint,
                                        const LinearAlgebra::BlockVector &topography_vector,
                                        const double density_exterior,
                                        const std::string &objective_name,
                                        KernelRepository<dim> &kernels,
                                        const Postprocess::DynamicTopographyBoundary boundary) const;
    };
  }
}

#include <aspect/simulator_access.h>

#include <deal.II/fe/fe_values.h>

#include <vector>

namespace aspect
{
  template <int dim> class Simulator;

  namespace Adjoint
  {
    /**
     * Quadrature-point adjoint seeds for material model outputs.
     */
    template <int dim>
    struct MaterialModelOutputAdjoints
    {
      MaterialModelOutputAdjoints(const unsigned int n_points = 0);

      std::vector<double> viscosities;
      std::vector<double> densities;
      std::vector<double> specific_heat;
      std::vector<double> thermal_conductivities;
    };



    /**

      /**
       * Minimal matrix wrapper used so SolverGMRES can apply the transpose of
       * ASPECT's already assembled advection block without explicitly forming
       * a second sparse matrix. The forward assembly still owns the matrix;
       * this class only redirects vmult() to Tvmult().
       */
      class TransposedSparseMatrix
      {
        public:
          explicit
          TransposedSparseMatrix(const LinearAlgebra::SparseMatrix &matrix)
            :
            matrix(matrix)
          {}

          void
          vmult(LinearAlgebra::Vector &dst,
                const LinearAlgebra::Vector &src) const
          {
            matrix.Tvmult(dst, src);
          }

        private:
          const LinearAlgebra::SparseMatrix &matrix;
      };
    }

    template <int dim>
    void
    AdvectionAdjointCalculator<dim>::solve_transposed_advection_system(const AdvectionField &advection_field,
                                                                       const LinearAlgebra::BlockVector &rhs,
                                                                       LinearAlgebra::BlockVector &solution) const
    {
      // The accepted forward step solves A_n q_n=b_n. Given a seed bar(q_n),
      // the implicit-function transpose first solves
      //
      //                         A_n^T z_n = bar(q_n).
      //
      // Subsequent routines contract z_n with derivatives of b_n-A_n q_n
      // with respect to history fields, velocity, and material coefficients.
      // Reusing the assembled forward block is important: it preserves the
      // exact quadrature, stabilization, constraint rows, and linearization
      // used by the primal time step.
      const unsigned int field_block_index =
        advection_field.block_index(this->introspection());

      // The transposed solution is a constraint-distribution target below;
      // keep it owned-only and writable. Replay snapshots, not adjoint
      // accumulation vectors, carry ghosted FE views.
      solution.reinit(this->introspection().index_sets.system_partitioning,
                      this->get_mpi_communicator());
      solution = 0.0;

      const double advection_solver_tolerance =
        advection_field.is_temperature()
        ? this->get_parameters().temperature_solver_tolerance
        : this->get_parameters().composition_solver_tolerance;
      const double tolerance =
        std::max(1e-50,
                 advection_solver_tolerance
                 * rhs.block(field_block_index).l2_norm());

      SolverControl solver_control(1000, tolerance);
      SolverGMRES<LinearAlgebra::Vector> solver(
        solver_control,
        SolverGMRES<LinearAlgebra::Vector>::AdditionalData(
          this->get_parameters().advection_gmres_restart_length,
          true));

      if (rhs.block(field_block_index).l2_norm() <= std::numeric_limits<double>::min())
        return;

      // Constraint rows in the forward matrix encode prescribed primal
      // values. Their adjoint right-hand-side entries must be zero; otherwise
      // a terminal functional could reintroduce fixed boundary data into z.
      LinearAlgebra::BlockVector homogeneous_rhs(rhs);
      this->get_current_constraints().set_zero(homogeneous_rhs);

      this->get_pcout() << "   Solving transposed "
                        << advection_field.name(this->introspection())
                        << " system... " << std::flush;

      const LinearAlgebra::SparseMatrix &advection_matrix =
        simulator->system_matrix.block(field_block_index, field_block_index);
      TransposedSparseMatrix transposed_advection_matrix(advection_matrix);

      LinearAlgebra::SparseMatrix explicit_transposed_advection_matrix;
      LinearAlgebra::Vector distributed_field(
        this->introspection().index_sets.system_partitioning[field_block_index],
        this->get_mpi_communicator());

      bool &comparison_completed = advection_field.is_temperature()
                                   ? temperature_preconditioner_comparison_completed
                                   : composition_preconditioner_comparison_completed;
      if (run_preconditioner_comparison && !comparison_completed)
        {
          // This diagnostic solves the same A^T system twice. It checks only
          // preconditioning quality: the identity-preconditioned result and
          // ILU(A^T)-preconditioned result must agree to solver tolerance.
          // Forming A^T explicitly is necessary for ILU construction, not for
          // the matrix-vector product used by GMRES.
          LinearAlgebra::Vector z_none(
            this->introspection().index_sets.system_partitioning[field_block_index],
            this->get_mpi_communicator());
          LinearAlgebra::Vector z_ilu(z_none);
          z_none = 0.0;
          z_ilu = 0.0;

          SolverControl none_solver_control(1000, tolerance);
          SolverGMRES<LinearAlgebra::Vector> none_solver(
            none_solver_control,
            SolverGMRES<LinearAlgebra::Vector>::AdditionalData(
              this->get_parameters().advection_gmres_restart_length, true));
          PreconditionIdentity identity_preconditioner;
          none_solver.solve(transposed_advection_matrix,
                            z_none,
                            homogeneous_rhs.block(field_block_index),
                            identity_preconditioner);

          EpetraExt::RowMatrix_Transpose transpose_builder;
          const Epetra_CrsMatrix &epetra_transposed_matrix =
            dynamic_cast<const Epetra_CrsMatrix &>(
              transpose_builder(const_cast<Epetra_CrsMatrix &>(
                                  advection_matrix.trilinos_matrix())));
          explicit_transposed_advection_matrix.reinit(epetra_transposed_matrix);
          LinearAlgebra::PreconditionILU ilu_preconditioner;
          ilu_preconditioner.initialize(explicit_transposed_advection_matrix);

          SolverControl ilu_solver_control(1000, tolerance);
          SolverGMRES<LinearAlgebra::Vector> ilu_solver(
            ilu_solver_control,
            SolverGMRES<LinearAlgebra::Vector>::AdditionalData(
              this->get_parameters().advection_gmres_restart_length, true));
          ilu_solver.solve(transposed_advection_matrix,
                           z_ilu,
                           homogeneous_rhs.block(field_block_index),
                           ilu_preconditioner);

          LinearAlgebra::Vector difference(z_ilu);
          difference.add(-1.0, z_none);
          LinearAlgebra::Vector none_residual(z_none);
          LinearAlgebra::Vector ilu_residual(z_ilu);
          transposed_advection_matrix.vmult(none_residual, z_none);
          transposed_advection_matrix.vmult(ilu_residual, z_ilu);
          none_residual.add(-1.0, homogeneous_rhs.block(field_block_index));
          ilu_residual.add(-1.0, homogeneous_rhs.block(field_block_index));

          const double relative_difference =
            difference.l2_norm() / std::max(z_none.l2_norm(), 1e-50);
          this->get_pcout() << "   Adjoint "
                            << advection_field.name(this->introspection())
                            << " preconditioner comparison: relative ||z_ILU - z_none||_2 = "
                            << relative_difference
                            << ", none residual = " << none_residual.l2_norm()
                            << " (" << none_solver_control.last_step() << " iterations)"
                            << ", ILU(A^T) residual = " << ilu_residual.l2_norm()
                            << " (" << ilu_solver_control.last_step() << " iterations)."
                            << std::endl;
          comparison_completed = true;
        }

      if (preconditioner == Preconditioner::none)
        {
          PreconditionIdentity identity_preconditioner;
          TimerOutput::Scope solve_timer(
            this->get_computing_timer(),
            advection_field.is_temperature()
            ? "Solve adjoint temperature system"
            : "Solve adjoint composition system");
          solver.solve(transposed_advection_matrix,
                       distributed_field,
                       homogeneous_rhs.block(field_block_index),
                       identity_preconditioner);
        }
      else
        {
          LinearAlgebra::PreconditionILU ilu_preconditioner;
          LinearAlgebra::PreconditionILU::AdditionalData preconditioner_data;
          {
            TimerOutput::Scope setup_timer(
              this->get_computing_timer(),
              advection_field.is_temperature()
              ? "Build adjoint temperature preconditioner"
              : "Build adjoint composition preconditioner");
            if (preconditioner == Preconditioner::transpose_ilu)
              {
                // Keep the validated adjoint action above, but explicitly
                // transpose a short-lived matrix for ILU(A^T). The EpetraExt
                // result is copied before the transposer is destroyed.
                EpetraExt::RowMatrix_Transpose transpose_builder;
                const Epetra_CrsMatrix &epetra_transposed_matrix =
                  dynamic_cast<const Epetra_CrsMatrix &>(
                    transpose_builder(const_cast<Epetra_CrsMatrix &>(
                                        advection_matrix.trilinos_matrix())));
                explicit_transposed_advection_matrix.reinit(epetra_transposed_matrix);
                ilu_preconditioner.initialize(explicit_transposed_advection_matrix,
                                              preconditioner_data);
              }
            else
              // This experimental option intentionally applies the forward
              // ILU(A) to the transpose solve. It can be cheaper to build but
              // is not algebraically equivalent to ILU(A^T), so it remains a
              // named opt-in preconditioner rather than the default.
              simulator->build_advection_preconditioner(advection_field,
                                                        ilu_preconditioner,
                                                        0.0);
          }

          try
            {
              TimerOutput::Scope solve_timer(
                this->get_computing_timer(),
                advection_field.is_temperature()
                ? "Solve adjoint temperature system"
                : "Solve adjoint composition system");
              solver.solve(transposed_advection_matrix,
                           distributed_field,
                           homogeneous_rhs.block(field_block_index),
                           ilu_preconditioner);
            }
          catch (const std::exception &)
            {
              // Match the forward recovery policy. This changes only the
              // factorization, never the adjoint operator or constraints.
              this->get_pcout() << "retrying transposed solve with different "
                                << "preconditioner... " << std::flush;
              if (preconditioner == Preconditioner::transpose_ilu)
                {
                  preconditioner_data.ilu_atol = 1e-5;
                  TimerOutput::Scope setup_timer(
                    this->get_computing_timer(),
                    advection_field.is_temperature()
                    ? "Build adjoint temperature preconditioner"
                    : "Build adjoint composition preconditioner");
                  ilu_preconditioner.initialize(explicit_transposed_advection_matrix,
                                                preconditioner_data);
                }
              else
                {
                  TimerOutput::Scope setup_timer(
                    this->get_computing_timer(),
                    advection_field.is_temperature()
                    ? "Build adjoint temperature preconditioner"
                    : "Build adjoint composition preconditioner");
                  simulator->build_advection_preconditioner(advection_field,
                                                            ilu_preconditioner,
                                                            1e-5);
                }
              TimerOutput::Scope solve_timer(
                this->get_computing_timer(),
                advection_field.is_temperature()
                ? "Solve adjoint temperature system"
                : "Solve adjoint composition system");
              solver.solve(transposed_advection_matrix,
                           distributed_field,
                           homogeneous_rhs.block(field_block_index),
                           ilu_preconditioner);
            }
        }

      solution.block(field_block_index) = distributed_field;
      // Reconstruct dependent DoFs from the homogeneous constraint graph.
      // This is distinct from set_zero(): the latter destroys the affine
      // component relations used for curved free-slip boundaries.
      this->get_current_constraints().distribute(solution);

      const char *preconditioner_name =
        (preconditioner == Preconditioner::none ? "none" :
         preconditioner == Preconditioner::forward_ilu_experimental ? "ILU(A), experimental" :
         "ILU(A^T)");
      this->get_pcout() << solver_control.last_step()
                        << " iterations (" << preconditioner_name
                        << " preconditioner)." << std::endl;
    }



    template <int dim>
    void
    AdvectionAdjointCalculator<dim>::add_history_transpose(const AdvectionField &advection_field,
                                                           const ForwardStateView<dim> &snapshot,
                                                           const LinearAlgebra::BlockVector &z,
                                                           LinearAlgebra::BlockVector &previous_field_adjoint,
                                                           LinearAlgebra::BlockVector *previous_previous_field_adjoint) const
    {
      // z is solver output and deliberately owned-only/writable. FEValues
      // needs locally relevant entries, so materialize a short-lived ghosted
      // read view without changing the solver vector itself.
      LinearAlgebra::BlockVector z_read;
      z_read.reinit(this->introspection().index_sets.system_partitioning,
                    this->introspection().index_sets.system_relevant_partitioning,
                    this->get_mpi_communicator());
      z_read = z;

      const unsigned int quadrature_degree =
        this->get_fe().base_element(advection_field.base_element(this->introspection())).degree + 1;
      const QGauss<dim> quadrature_formula(quadrature_degree);
      const unsigned int dofs_per_cell = this->get_fe().dofs_per_cell;
      const FEValuesExtractors::Scalar field_extractor =
        advection_field.scalar_extractor(this->introspection());

      FEValues<dim> fe_values(this->get_mapping(),
                              this->get_fe(),
                              quadrature_formula,
                              update_values |
                              update_gradients |
                              update_quadrature_points |
                              update_JxW_values);

      std::vector<double> z_values(quadrature_formula.size());
      Vector<double> local_previous(dofs_per_cell);
      Vector<double> local_previous_previous(dofs_per_cell);
      std::vector<types::global_dof_index> local_dof_indices(dofs_per_cell);

      const bool use_bdf2_scheme = (snapshot.timestep_number > 1);
      const double previous_coefficient =
        use_bdf2_scheme ? (1.0 + snapshot.time_step / snapshot.old_time_step) : 1.0;
      const double previous_previous_coefficient =
        use_bdf2_scheme
        ? -(snapshot.time_step * snapshot.time_step)
        / (snapshot.old_time_step * (snapshot.time_step + snapshot.old_time_step))
        : 0.0;

      // The history portion of b_n is mass-like. In the variable-step BDF2
      // convention used by the forward assembly it contains
      //
      //   integral rho*C_p [a_1 q_{n-1}+a_2 q_{n-2}] phi_i dx,
      //   a_1 = 1+dt/dt_old,
      //   a_2 = -dt^2/[dt_old(dt+dt_old)].
      //
      // Its transpose therefore integrates z phi_j with the same coefficients
      // and scatters to the two history slots. For composition rho*C_p is
      // replaced by one. Constraints are applied during local-to-global
      // distribution, making this the transpose on the constrained FE space.
      for (const auto &cell : this->get_dof_handler().active_cell_iterators())
        if (cell->is_locally_owned())
          {
            fe_values.reinit(cell);
            fe_values[field_extractor]
            .get_function_values(z_read, z_values);

            local_previous = 0.0;
            local_previous_previous = 0.0;
            cell->get_dof_indices(local_dof_indices);

            MaterialModel::MaterialModelInputs<dim> in(fe_values,
                                                       cell,
                                                       this->introspection(),
                                                       simulator->current_linearization_point);
            MaterialModel::MaterialModelOutputs<dim> out(fe_values.n_quadrature_points,
                                                         this->n_compositional_fields());
            in.requested_properties =
              MaterialModel::MaterialProperties::equation_of_state_properties;
            this->get_material_model().evaluate(in, out);

            if (this->get_parameters().formulation_temperature_equation ==
                Parameters<dim>::Formulation::TemperatureEquation::reference_density_profile)
              for (unsigned int q = 0; q < fe_values.n_quadrature_points; ++q)
                out.densities[q] =
                  this->get_adiabatic_conditions().density(fe_values.quadrature_point(q));

            for (unsigned int q = 0; q < fe_values.n_quadrature_points; ++q)
              {
                const double density_c_p =
                  advection_field.is_temperature()
                  ? out.densities[q] * out.specific_heat[q]
                  : 1.0;
                const double JxW = fe_values.JxW(q);

                for (unsigned int i = 0; i < dofs_per_cell; ++i)
                  {
                    const double phi_i =
                      fe_values[field_extractor].value(i, q);
                    const double contribution = z_values[q] * density_c_p * phi_i * JxW;

                    local_previous(i) += previous_coefficient * contribution;
                    if (use_bdf2_scheme)
                      local_previous_previous(i) += previous_previous_coefficient * contribution;
                  }
              }

            this->get_current_constraints().distribute_local_to_global(local_previous,
                                                                       local_dof_indices,
                                                                       previous_field_adjoint);

            if (use_bdf2_scheme)
              {
                AssertThrow(previous_previous_field_adjoint != nullptr,
                            ExcInternalError());
                this->get_current_constraints().distribute_local_to_global(local_previous_previous,
                                                                           local_dof_indices,
                                                                           *previous_previous_field_adjoint);
              }
          }

      previous_field_adjoint.compress(VectorOperation::add);
      if (use_bdf2_scheme)
        previous_previous_field_adjoint->compress(VectorOperation::add);
    }



    template <int dim>
    void
    AdvectionAdjointCalculator<dim>::add_velocity_transpose(const AdvectionField &advection_field,
                                                            const ForwardStateView<dim> &snapshot,
                                                            const LinearAlgebra::BlockVector &z,
                                                            LinearAlgebra::BlockVector &previous_stokes_adjoint_rhs,
                                                            LinearAlgebra::BlockVector *previous_previous_stokes_adjoint_rhs) const
    {
      // z is an owned-only solver output. Materialize a local ghosted view
      // solely for the FE evaluation below.
      LinearAlgebra::BlockVector z_read;
      z_read.reinit(this->introspection().index_sets.system_partitioning,
                    this->introspection().index_sets.system_relevant_partitioning,
                    this->get_mpi_communicator());
      z_read = z;

      const unsigned int quadrature_degree =
        this->get_fe().base_element(advection_field.base_element(this->introspection())).degree + 1;
      const QGauss<dim> quadrature_formula(quadrature_degree);
      const unsigned int dofs_per_cell = this->get_fe().dofs_per_cell;
      const FEValuesExtractors::Scalar field_extractor =
        advection_field.scalar_extractor(this->introspection());

      FEValues<dim> fe_values(this->get_mapping(),
                              this->get_fe(),
                              quadrature_formula,
                              update_values |
                              update_gradients |
                              update_quadrature_points |
                              update_JxW_values);

      std::vector<double> z_values(quadrature_formula.size());
      std::vector<Tensor<1,dim>> field_gradients(quadrature_formula.size());
      Vector<double> local_previous(dofs_per_cell);
      Vector<double> local_previous_previous(dofs_per_cell);
      std::vector<types::global_dof_index> local_dof_indices(dofs_per_cell);

      const bool use_bdf2_scheme = (snapshot.timestep_number > 1);
      const double gamma1 =
        use_bdf2_scheme ? (1.0 + snapshot.time_step / snapshot.old_time_step) : 1.0;
      const double gamma2 =
        use_bdf2_scheme ? -snapshot.time_step / snapshot.old_time_step : 0.0;

      // The forward transport contribution has the form
      //
      //   dt integral rho*C_p (u*.grad(q_n)) phi_i dx,
      //   u* = gamma1 u_{n-1}+gamma2 u_{n-2}.
      //
      // Contracting with z produces, for a velocity basis function psi_j,
      //
      //   -dt integral rho*C_p z (psi_j.grad(q_n)) dx.
      //
      // The minus sign follows the b-Aq residual convention. Multiplication by
      // gamma1/gamma2 then sends this seed to the two quasi-static Stokes
      // states from which u* was extrapolated.
      for (const auto &cell : this->get_dof_handler().active_cell_iterators())
        if (cell->is_locally_owned())
          {
            fe_values.reinit(cell);
            fe_values[field_extractor]
            .get_function_values(z_read, z_values);
            fe_values[field_extractor]
            .get_function_gradients(simulator->solution, field_gradients);

            local_previous = 0.0;
            local_previous_previous = 0.0;
            cell->get_dof_indices(local_dof_indices);

            MaterialModel::MaterialModelInputs<dim> in(fe_values,
                                                       cell,
                                                       this->introspection(),
                                                       simulator->current_linearization_point);
            MaterialModel::MaterialModelOutputs<dim> out(fe_values.n_quadrature_points,
                                                         this->n_compositional_fields());
            in.requested_properties =
              MaterialModel::MaterialProperties::equation_of_state_properties;
            this->get_material_model().evaluate(in, out);

            if (this->get_parameters().formulation_temperature_equation ==
                Parameters<dim>::Formulation::TemperatureEquation::reference_density_profile)
              for (unsigned int q = 0; q < fe_values.n_quadrature_points; ++q)
                out.densities[q] =
                  this->get_adiabatic_conditions().density(fe_values.quadrature_point(q));

            for (unsigned int q = 0; q < fe_values.n_quadrature_points; ++q)
              {
                const double density_c_p =
                  advection_field.is_temperature()
                  ? out.densities[q] * out.specific_heat[q]
                  : 1.0;
                const double scale =
                  -snapshot.time_step * density_c_p * z_values[q] * fe_values.JxW(q);

                for (unsigned int i = 0; i < dofs_per_cell; ++i)
                  if (this->introspection().component_masks.velocities[
                        this->get_fe().system_to_component_index(i).first])
                    {
                      const Tensor<1,dim> phi_u =
                        fe_values[this->introspection().extractors.velocities].value(i, q);
                      const double contribution =
                        scale * (phi_u * field_gradients[q]);

                      local_previous(i) += gamma1 * contribution;
                      if (use_bdf2_scheme)
                        local_previous_previous(i) += gamma2 * contribution;
                    }
              }

            this->get_current_constraints().distribute_local_to_global(local_previous,
                                                                       local_dof_indices,
                                                                       previous_stokes_adjoint_rhs);

            if (use_bdf2_scheme)
              {
                AssertThrow(previous_previous_stokes_adjoint_rhs != nullptr,
                            ExcInternalError());
                this->get_current_constraints().distribute_local_to_global(local_previous_previous,
                                                                           local_dof_indices,
                                                                           *previous_previous_stokes_adjoint_rhs);
              }
          }

      previous_stokes_adjoint_rhs.compress(VectorOperation::add);
      if (use_bdf2_scheme)
        previous_previous_stokes_adjoint_rhs->compress(VectorOperation::add);
    }



    template <int dim>
    void
    AdvectionAdjointCalculator<dim>::add_temperature_material_prefactor_transpose(
      const ForwardStateView<dim> &snapshot,
      const LinearAlgebra::BlockVector &z,
      LinearAlgebra::BlockVector &previous_temperature_adjoint,
      LinearAlgebra::BlockVector *previous_previous_temperature_adjoint,
      std::vector<LinearAlgebra::BlockVector> *previous_composition_adjoints,
      std::vector<LinearAlgebra::BlockVector> *previous_previous_composition_adjoints) const
    {
      if (this->get_parameters().formulation_temperature_equation ==
          Parameters<dim>::Formulation::TemperatureEquation::reference_density_profile)
        return;

      // z is the owned-only output of the transposed advection solve. This
      // material-prefactor transpose evaluates it with FEValues, so create a
      // short-lived locally-relevant read view at the FE boundary.
      LinearAlgebra::BlockVector z_read;
      z_read.reinit(this->introspection().index_sets.system_partitioning,
                    this->introspection().index_sets.system_relevant_partitioning,
                    this->get_mpi_communicator());
      z_read = z;

      const unsigned int quadrature_degree =
        this->get_fe().base_element(this->introspection().base_elements.temperature).degree + 1;
      const QGauss<dim> quadrature_formula(quadrature_degree);
      const unsigned int dofs_per_cell = this->get_fe().dofs_per_cell;

      FEValues<dim> fe_values(this->get_mapping(),
                              this->get_fe(),
                              quadrature_formula,
                              update_values |
                              update_gradients |
                              update_quadrature_points |
                              update_JxW_values);

      std::vector<double> z_values(quadrature_formula.size());
      std::vector<double> temperature_values(quadrature_formula.size());
      std::vector<Tensor<1, dim>> temperature_gradients(quadrature_formula.size());
      std::vector<double> old_temperature_values(quadrature_formula.size());
      std::vector<double> old_old_temperature_values(quadrature_formula.size());
      std::vector<Tensor<1, dim>> current_velocity_values(quadrature_formula.size());
      std::vector<types::global_dof_index> local_dof_indices(dofs_per_cell);

      const bool use_bdf2_scheme = (snapshot.timestep_number > 1);
      const double bdf2_factor = use_bdf2_scheme
                                 ? (2.0 * snapshot.time_step + snapshot.old_time_step)
                                 / (snapshot.time_step + snapshot.old_time_step)
                                 : 1.0;
      const double gamma1 =
        use_bdf2_scheme ? (1.0 + snapshot.time_step / snapshot.old_time_step) : 1.0;
      const double gamma2 =
        use_bdf2_scheme ? -snapshot.time_step / snapshot.old_time_step : 0.0;

      // Differentiate the temperature equation through a=rho(T,C) C_p(T,C).
      // With the forward residual rearranged as b-A T_n, the local seed is
      //
      //   bar(a)_q = z_q [H_q-dt u_q.grad(T_n)-bdf T_n] JxW_q,
      //
      // where H_q is the BDF history combination. The product rule gives
      // bar(rho)=C_p bar(a) and bar(C_p)=rho bar(a). The material-model VJP
      // maps these output seeds to temperature and composition inputs, and
      // the scatter applies the same extrapolation weights gamma1/gamma2 used
      // when those inputs were evaluated in the forward assembly.
      for (const auto &cell : this->get_dof_handler().active_cell_iterators())
        if (cell->is_locally_owned())
          {
            fe_values.reinit(cell);
            cell->get_dof_indices(local_dof_indices);

            fe_values[this->introspection().extractors.temperature]
            .get_function_values(z_read, z_values);
            fe_values[this->introspection().extractors.temperature]
            .get_function_values(simulator->solution, temperature_values);
            fe_values[this->introspection().extractors.temperature]
            .get_function_gradients(simulator->solution, temperature_gradients);
            fe_values[this->introspection().extractors.temperature]
            .get_function_values(simulator->old_solution, old_temperature_values);
            fe_values[this->introspection().extractors.temperature]
            .get_function_values(simulator->old_old_solution, old_old_temperature_values);
            fe_values[this->introspection().extractors.velocities]
            .get_function_values(simulator->current_linearization_point, current_velocity_values);

            MaterialModel::MaterialModelInputs<dim> in(fe_values,
                                                       cell,
                                                       this->introspection(),
                                                       simulator->current_linearization_point);
            MaterialModel::MaterialModelOutputs<dim> out(fe_values.n_quadrature_points,
                                                         this->n_compositional_fields());
            in.requested_properties =
              MaterialModel::MaterialProperties::equation_of_state_properties;
            this->get_material_model().evaluate(in, out);

            MaterialModelOutputAdjoints<dim> bar_out(fe_values.n_quadrature_points);

            for (unsigned int q = 0; q < fe_values.n_quadrature_points; ++q)
              {
                const double old_field_term =
                  use_bdf2_scheme
                  ? (old_temperature_values[q] * (1.0 + snapshot.time_step / snapshot.old_time_step)
                     - old_old_temperature_values[q] * (snapshot.time_step * snapshot.time_step)
                     / (snapshot.old_time_step * (snapshot.time_step + snapshot.old_time_step)))
                  : old_temperature_values[q];
                const double operator_field_term =
                  snapshot.time_step * (current_velocity_values[q] * temperature_gradients[q])
                  + bdf2_factor * temperature_values[q];
                const double coefficient_seed =
                  z_values[q] * (old_field_term - operator_field_term) * fe_values.JxW(q);

                if (coefficient_seed == 0.0)
                  continue;

                bar_out.densities[q] += out.specific_heat[q] * coefficient_seed;
                bar_out.specific_heat[q] += out.densities[q] * coefficient_seed;
              }

            MaterialModelInputAdjoints<dim> bar_in(fe_values.n_quadrature_points,
                                                   this->n_compositional_fields());
            const MaterialModelAdjoint<dim> material_model_adjoint(*simulator);
            material_model_adjoint.apply_derivative_transpose(in,
                                                              out,
                                                              bar_out,
                                                              bar_in);

            const MaterialModelInputAdjointScatter<dim> material_input_scatter(*simulator);
            material_input_scatter.scatter_temperature_advection_inputs(
              fe_values,
              local_dof_indices,
              bar_in,
              gamma1,
              gamma2,
              previous_temperature_adjoint,
              previous_previous_temperature_adjoint,
              previous_composition_adjoints,
              previous_previous_composition_adjoints);
          }

      previous_temperature_adjoint.compress(VectorOperation::add);
      if (use_bdf2_scheme)
        {
          AssertThrow(previous_previous_temperature_adjoint != nullptr,
                      ExcInternalError());
          previous_previous_temperature_adjoint->compress(VectorOperation::add);
        }
      for (unsigned int c = 0; c < this->introspection().n_compositional_fields; ++c)
        if (previous_composition_adjoints != nullptr)
          {
            (*previous_composition_adjoints)[c].compress(VectorOperation::add);
            if (use_bdf2_scheme)
              {
                AssertThrow(previous_previous_composition_adjoints != nullptr,
                            ExcInternalError());
                (*previous_previous_composition_adjoints)[c].compress(VectorOperation::add);
              }
          }
    }































    template <int dim>
    void
    AdvectionAdjointCalculator<dim>::propagate_advection_field_adjoint(const AdvectionField &advection_field,
                                                                       const ForwardStateView<dim> &snapshot,
                                                                       const LinearAlgebra::BlockVector &current_field_adjoint,
                                                                       LinearAlgebra::BlockVector &previous_field_adjoint,
                                                                       LinearAlgebra::BlockVector *previous_previous_field_adjoint,
                                                                       LinearAlgebra::BlockVector &previous_stokes_adjoint_rhs,
                                                                       LinearAlgebra::BlockVector *previous_previous_stokes_adjoint_rhs)
    {
      // Each propagation call realizes one discrete reverse-time edge:
      // restore state n, reassemble A_n, solve A_n^T z_n=bar(q_n), and apply
      // (partial R_n/partial x)^T z_n to every predecessor x. The scoped state
      // guard makes this transaction exception-safe and prevents replay of an
      // old time level from leaking time, constraints, matrices, or pressure
      // normalization into the enclosing reverse sweep.
      ScopedSimulatorState<dim> simulator_state(*simulator);

      simulator->solution = snapshot.solution.get();
      simulator->old_solution = snapshot.old_solution.get();
      simulator->old_old_solution = snapshot.old_old_solution.get();
      simulator->current_linearization_point = snapshot.advection_linearization_point.get();
      simulator->time = snapshot.time;
      simulator->time_step = snapshot.time_step;
      simulator->old_time_step = snapshot.old_time_step;
      simulator->timestep_number = snapshot.timestep_number;
      simulator->last_pressure_normalization_adjustment =
        snapshot.pressure_normalization_adjustment;

      BoundaryConditions<dim> boundary_conditions;
      boundary_conditions.build_temperature_adjoint_constraints(snapshot,
                                                                simulator->current_constraints);

      simulator->assemble_advection_system(advection_field);

      LinearAlgebra::BlockVector z;
      solve_transposed_advection_system(advection_field, current_field_adjoint, z);

      if (this->get_parameters().advection_stabilization_method
          == Parameters<dim>::AdvectionStabilizationMethod::entropy_viscosity)
        {
          ArtificialViscosityAdjoint<dim> artificial_viscosity_adjoint(*simulator);
          // The diffusion part of the assembled residual depends on the
          // cellwise artificial-viscosity coefficient. First collect the seed
          // for that coefficient from the residual transpose; then push it
          // further back through the entropy-viscosity estimator to old field
          // and old velocity states.
          const std::vector<double> artificial_viscosity_seed =
            artificial_viscosity_adjoint.assemble_coefficient_seed(advection_field, snapshot, z);

          double local_seed_norm_square = 0.0;
          for (const double value : artificial_viscosity_seed)
            local_seed_norm_square += value * value;

          this->get_pcout() << "   Artificial-viscosity coefficient adjoint seed norm: "
                            << std::sqrt(local_seed_norm_square) << std::endl;

          artificial_viscosity_adjoint.add_estimator_transpose(
            advection_field,
            snapshot,
            artificial_viscosity_seed,
            previous_field_adjoint,
            previous_previous_field_adjoint,
            previous_stokes_adjoint_rhs,
            previous_previous_stokes_adjoint_rhs);
        }

      // Reverse the computation graph in dependency order. Stabilization is
      // handled first because its coefficient was used while forming A_n;
      // the ordinary BDF history and transport-velocity dependencies are then
      // accumulated into the same n-1/n-2 adjoint slots. Addition is valid
      // because all three are independent summands of the forward residual.
      add_history_transpose(advection_field,
                            snapshot,
                            z,
                            previous_field_adjoint,
                            previous_previous_field_adjoint);
      // The remaining explicit velocity dependence of the advection residual
      // is handled outside the artificial-viscosity estimator, because it
      // comes from the ordinary transport term rather than stabilization.
      add_velocity_transpose(advection_field,
                             snapshot,
                             z,
                             previous_stokes_adjoint_rhs,
                             previous_previous_stokes_adjoint_rhs);

    }



    template <int dim>
    void
    AdvectionAdjointCalculator<dim>::propagate_temperature_adjoint(const ForwardStateView<dim> &snapshot,
                                                                   const LinearAlgebra::BlockVector &current_temperature_adjoint,
                                                                   LinearAlgebra::BlockVector &previous_temperature_adjoint,
                                                                   LinearAlgebra::BlockVector *previous_previous_temperature_adjoint,
                                                                   LinearAlgebra::BlockVector &previous_stokes_adjoint_rhs,
                                                                   LinearAlgebra::BlockVector *previous_previous_stokes_adjoint_rhs,
                                                                   std::vector<LinearAlgebra::BlockVector> *previous_composition_adjoints,
                                                                   std::vector<LinearAlgebra::BlockVector> *previous_previous_composition_adjoints)
    {
      // Temperature propagation is intentionally not just a call to the
      // generic field propagation: rho*C_p can depend on temperature and
      // composition, so the transpose must pass through the material model
      // before applying the common BDF history and velocity terms.
      ScopedSimulatorState<dim> simulator_state(*simulator);

      simulator->solution = snapshot.solution.get();
      simulator->old_solution = snapshot.old_solution.get();
      simulator->old_old_solution = snapshot.old_old_solution.get();
      simulator->current_linearization_point = snapshot.advection_linearization_point.get();
      simulator->time = snapshot.time;
      simulator->time_step = snapshot.time_step;
      simulator->old_time_step = snapshot.old_time_step;
      simulator->timestep_number = snapshot.timestep_number;
      simulator->last_pressure_normalization_adjustment =
        snapshot.pressure_normalization_adjustment;

      BoundaryConditions<dim> boundary_conditions;
      boundary_conditions.build_temperature_adjoint_constraints(snapshot,
                                                                simulator->current_constraints);

      simulator->assemble_advection_system(AdvectionField::temperature());

      LinearAlgebra::BlockVector z;
      solve_transposed_advection_system(AdvectionField::temperature(),
                                        current_temperature_adjoint,
                                        z);

      add_temperature_material_prefactor_transpose(
        snapshot,
        z,
        previous_temperature_adjoint,
        previous_previous_temperature_adjoint,
        previous_composition_adjoints,
        previous_previous_composition_adjoints);

      if (this->get_parameters().advection_stabilization_method
          == Parameters<dim>::AdvectionStabilizationMethod::entropy_viscosity)
        {
          ArtificialViscosityAdjoint<dim> artificial_viscosity_adjoint(*simulator);
          // The temperature artificial-viscosity estimator can depend on
          // temperature, composition, and velocity through residuals and
          // material prefactors. The optional composition slices allow those
          // dependencies to be accumulated alongside the temperature history.
          const std::vector<double> artificial_viscosity_seed =
            artificial_viscosity_adjoint.assemble_coefficient_seed(AdvectionField::temperature(),
                                                                   snapshot,
                                                                   z);

          double local_seed_norm_square = 0.0;
          for (const double value : artificial_viscosity_seed)
            local_seed_norm_square += value * value;

          this->get_pcout() << "   Artificial-viscosity coefficient adjoint seed norm: "
                            << std::sqrt(local_seed_norm_square) << std::endl;

          artificial_viscosity_adjoint.add_estimator_transpose(
            AdvectionField::temperature(),
            snapshot,
            artificial_viscosity_seed,
            previous_temperature_adjoint,
            previous_previous_temperature_adjoint,
            previous_stokes_adjoint_rhs,
            previous_previous_stokes_adjoint_rhs,
            previous_composition_adjoints,
            previous_previous_composition_adjoints);
        }

      add_history_transpose(AdvectionField::temperature(),
                            snapshot,
                            z,
                            previous_temperature_adjoint,
                            previous_previous_temperature_adjoint);
      add_velocity_transpose(AdvectionField::temperature(),
                             snapshot,
                             z,
                             previous_stokes_adjoint_rhs,
                             previous_previous_stokes_adjoint_rhs);

    }
  }

    ArtificialViscosityAdjoint<dim>::ArtificialViscosityAdjoint(Simulator<dim> &simulator_object)

    template <int dim>
    std::vector<double>
    ArtificialViscosityAdjoint<dim>::assemble_coefficient_seed(const AdvectionField &advection_field,
                                                               const ForwardStateView<dim> &snapshot,
                                                               const LinearAlgebra::BlockVector &transpose_solution) const
    {
      AssertThrow(this->get_parameters().advection_stabilization_method
                  == Parameters<dim>::AdvectionStabilizationMethod::entropy_viscosity,
                  ExcMessage("ArtificialViscosityAdjoint currently supports only entropy-viscosity stabilization."));

      const Vector<double> &artificial_viscosity_per_cell =
        simulator->last_advection_artificial_viscosity_per_cell;
      const Vector<double> &minimum_physical_diffusion_per_cell =
        simulator->last_advection_physical_diffusion_min_per_cell;
      const Vector<double> &maximum_physical_diffusion_per_cell =
        simulator->last_advection_physical_diffusion_max_per_cell;
      AssertThrow(artificial_viscosity_per_cell.size()
                  == this->get_dof_handler().get_triangulation().n_active_cells(),
                  ExcMessage("The artificial-viscosity adjoint seed requires the advection "
                             "system to be assembled first so that the cellwise stabilization "
                             "coefficients are available."));
      AssertThrow(minimum_physical_diffusion_per_cell.size() == artificial_viscosity_per_cell.size() &&
                  maximum_physical_diffusion_per_cell.size() == artificial_viscosity_per_cell.size(),
                  ExcMessage("The artificial-viscosity adjoint seed requires cached physical diffusion "
                             "bounds from the same temperature advection assembly."));

      std::vector<double> coefficient_seed(artificial_viscosity_per_cell.size(), 0.0);

      const unsigned int quadrature_degree =
        this->get_fe().base_element(advection_field.base_element(this->introspection())).degree + 1;
      const QGauss<dim> quadrature_formula(quadrature_degree);
      const FEValuesExtractors::Scalar field_extractor =
        advection_field.scalar_extractor(this->introspection());

      FEValues<dim> fe_values(this->get_mapping(),
                              this->get_fe(),
                              quadrature_formula,
                              update_gradients |
                              update_quadrature_points |
                              update_JxW_values);

      std::vector<Tensor<1, dim>> field_gradients(quadrature_formula.size());
      std::vector<Tensor<1, dim>> transpose_gradients(quadrature_formula.size());

      // Reverse the stabilized diffusion contribution on each cell K. In the
      // forward residual the artificial coefficient enters as
      //
      //   R_i^K += dt * integral_K nu_K grad(q_n) . grad(phi_i) dx,
      //
      // where nu_K=max(kappa,nu_art,K) is constant on K. Contracting its
      // derivative with the advection adjoint z gives
      //
      //   bar(nu_art,K) = -dt * integral_K grad(z) . grad(q_n) dx.
      //
      // The sign is the one induced by the residual convention used by the
      // reverse time step. Cells where physical diffusion wins the
      // max(kappa,nu_art,K) active set receive no artificial-viscosity seed.
      // A strict inequality is required here: differentiating through a tie
      // would require a separately specified generalized derivative.
      for (const auto &cell : this->get_dof_handler().active_cell_iterators())
        if (cell->is_locally_owned())
          {
            const unsigned int active_cell_index = cell->active_cell_index();
            const double artificial_viscosity =
              artificial_viscosity_per_cell[active_cell_index];
            const double minimum_physical_diffusion =
              minimum_physical_diffusion_per_cell[active_cell_index];
            const double maximum_physical_diffusion =
              maximum_physical_diffusion_per_cell[active_cell_index];

            if (artificial_viscosity <= minimum_physical_diffusion)
              continue;

            AssertThrow(artificial_viscosity > maximum_physical_diffusion,
                        ExcMessage("The artificial-viscosity adjoint seed currently supports only cells "
                                   "where max(k, nu_art) selects the same branch at every quadrature point. "
                                   "A cell with mixed physical/artificial diffusion branches needs a "
                                   "quadrature-point active-set cache."));

            fe_values.reinit(cell);

            fe_values[field_extractor]
            .get_function_gradients(this->get_solution(),
                                    field_gradients);
            fe_values[field_extractor]
            .get_function_gradients(transpose_solution,
                                    transpose_gradients);

            for (unsigned int q = 0; q < fe_values.n_quadrature_points; ++q)
              coefficient_seed[active_cell_index] -=
                snapshot.time_step
                * (transpose_gradients[q] * field_gradients[q])
                * fe_values.JxW(q);
          }

      return coefficient_seed;
    }







    template <int dim>
    void
    ArtificialViscosityAdjoint<dim>::add_global_entropy_variation_transpose(
      const AdvectionField &advection_field,
      const double bar_entropy_variation,
      double &bar_average_field,
      LinearAlgebra::BlockVector &previous_field_adjoint,
      LinearAlgebra::BlockVector *previous_previous_field_adjoint) const
    {
      if (bar_entropy_variation == 0.0)
        return;

      const Quadrature<dim> &quadrature_formula =
        advection_field.is_temperature()
        ? this->introspection().quadratures.temperature
        : this->introspection().quadratures.compositional_fields[advection_field.compositional_variable];
      const unsigned int dofs_per_cell = this->get_fe().dofs_per_cell;
      const FEValuesExtractors::Scalar field_extractor =
        advection_field.scalar_extractor(this->introspection());

      const std::pair<double, double> global_field_range =
        simulator->get_extrapolated_advection_field_range(advection_field);
      const double average_field =
        0.5 * (global_field_range.first + global_field_range.second);

      // For alpha=2 ASPECT normalizes the entropy residual by
      //
      //   Delta_E = max(max_x E(x)-<E>, <E>-min_x E(x)),
      //   E(x)     = (q_h(x)-q_avg)^2,
      //   q_h      = (q_{n-1}+q_{n-2})/2.
      //
      // This first pass reconstructs the four global scalars that define the
      // selected branch. Their MPI reductions are part of the discrete
      // operator: the transpose must use the same globally selected extrema
      // on every rank.

      FEValues<dim> fe_values(this->get_mapping(),
                              this->get_fe(),
                              quadrature_formula,
                              update_values |
                              update_JxW_values);

      std::vector<double> old_field_values(quadrature_formula.size());
      std::vector<double> old_old_field_values(quadrature_formula.size());
      std::vector<types::global_dof_index> local_dof_indices(dofs_per_cell);

      double local_min_entropy = std::numeric_limits<double>::max();
      double local_max_entropy = std::numeric_limits<double>::lowest();
      double local_entropy_integrated = 0.0;
      double local_area = 0.0;

      for (const auto &cell : this->get_dof_handler().active_cell_iterators())
        if (cell->is_locally_owned())
          {
            fe_values.reinit(cell);
            fe_values[field_extractor].get_function_values(simulator->old_solution,
                                                           old_field_values);
            fe_values[field_extractor].get_function_values(simulator->old_old_solution,
                                                           old_old_field_values);
            for (unsigned int q = 0; q < fe_values.n_quadrature_points; ++q)
              {
                const double field_history =
                  0.5 * (old_field_values[q] + old_old_field_values[q]);
                const double entropy =
                  (field_history - average_field) * (field_history - average_field);
                local_min_entropy = std::min(local_min_entropy, entropy);
                local_max_entropy = std::max(local_max_entropy, entropy);
                local_entropy_integrated += entropy * fe_values.JxW(q);
                local_area += fe_values.JxW(q);
              }
          }

      const double global_min_entropy =
        -Utilities::MPI::max(-local_min_entropy, this->get_mpi_communicator());
      const double global_max_entropy =
        Utilities::MPI::max(local_max_entropy, this->get_mpi_communicator());
      const double global_entropy_integrated =
        Utilities::MPI::sum(local_entropy_integrated, this->get_mpi_communicator());
      const double global_area =
        Utilities::MPI::sum(local_area, this->get_mpi_communicator());
      const double average_entropy =
        global_entropy_integrated / global_area;

      const bool upper_branch =
        (global_max_entropy - average_entropy)
        >= (average_entropy - global_min_entropy);
      // Reverse the selected branch of Delta_E. For the upper branch,
      // bar(E_max)=bar(Delta_E) and bar(<E>)=-bar(Delta_E); the signs reverse
      // for the lower branch. At equality the forward comparison selects the
      // upper branch, which also fixes the adjoint convention at the kink.
      const double bar_max_entropy = upper_branch ? bar_entropy_variation : 0.0;
      const double bar_min_entropy = upper_branch ? 0.0 : -bar_entropy_variation;
      const double bar_average_entropy = upper_branch
                                         ? -bar_entropy_variation
                                         : bar_entropy_variation;
      const double extremum_tolerance =
        1e-12 * std::max({1.0,
                          std::abs(global_min_entropy),
                          std::abs(global_max_entropy)
                         });

      Vector<double> local_previous(dofs_per_cell);
      Vector<double> local_previous_previous(dofs_per_cell);
      double local_bar_average_field = 0.0;

      for (const auto &cell : this->get_dof_handler().active_cell_iterators())
        if (cell->is_locally_owned())
          {
            fe_values.reinit(cell);
            cell->get_dof_indices(local_dof_indices);
            fe_values[field_extractor].get_function_values(simulator->old_solution,
                                                           old_field_values);
            fe_values[field_extractor].get_function_values(simulator->old_old_solution,
                                                           old_old_field_values);
            local_previous = 0.0;
            local_previous_previous = 0.0;

            for (unsigned int q = 0; q < fe_values.n_quadrature_points; ++q)
              {
                const double field_history =
                  0.5 * (old_field_values[q] + old_old_field_values[q]);
                const double entropy =
                  (field_history - average_field) * (field_history - average_field);

                // <E> is a volume average, so its transpose distributes the
                // scalar seed as JxW/|Omega|. The seed of a global extremum is
                // placed at all quadrature points in its tolerance-defined
                // active set. This mirrors the forward quadrature sampling;
                // it is not a derivative of the continuous pointwise max.
                double bar_entropy =
                  bar_average_entropy * fe_values.JxW(q) / global_area;
                if (upper_branch
                    && std::abs(entropy - global_max_entropy) <= extremum_tolerance)
                  bar_entropy += bar_max_entropy;
                if (!upper_branch
                    && std::abs(entropy - global_min_entropy) <= extremum_tolerance)
                  bar_entropy += bar_min_entropy;

                // E=(q_h-q_avg)^2 gives
                //   bar(q_h)   =  2(q_h-q_avg) bar(E),
                //   bar(q_avg) = -2(q_h-q_avg) bar(E).
                // Because q_h is the arithmetic midpoint, both stored
                // histories receive one half of bar(q_h).
                const double bar_field_history =
                  2.0 * (field_history - average_field) * bar_entropy;
                local_bar_average_field +=
                  -2.0 * (field_history - average_field) * bar_entropy;

                for (unsigned int i = 0; i < dofs_per_cell; ++i)
                  if (this->get_fe().system_to_component_index(i).first
                      == advection_field.component_index(this->introspection()))
                    {
                      const double contribution =
                        0.5 * bar_field_history
                        * fe_values[field_extractor].value(i, q);
                      local_previous(i) += contribution;
                      local_previous_previous(i) += contribution;
                    }
              }

            this->get_current_constraints().distribute_local_to_global(local_previous,
                                                                       local_dof_indices,
                                                                       previous_field_adjoint);
            AssertThrow(previous_previous_field_adjoint != nullptr,
                        ExcInternalError());
            this->get_current_constraints().distribute_local_to_global(local_previous_previous,
                                                                       local_dof_indices,
                                                                       *previous_previous_field_adjoint);
          }

      previous_field_adjoint.compress(VectorOperation::add);
      previous_previous_field_adjoint->compress(VectorOperation::add);
      bar_average_field +=
        Utilities::MPI::sum(local_bar_average_field, this->get_mpi_communicator());
    }



    template <int dim>
    void
    ArtificialViscosityAdjoint<dim>::add_extrapolated_field_average_transpose(
      const AdvectionField &advection_field,
      const ForwardStateView<dim> &snapshot,
      const double bar_average_field,
      LinearAlgebra::BlockVector &previous_field_adjoint,
      LinearAlgebra::BlockVector *previous_previous_field_adjoint) const
    {
      if (bar_average_field == 0.0)
        return;

      if (snapshot.timestep_number > 1)
        AssertThrow(previous_previous_field_adjoint != nullptr,
                    ExcInternalError());

      const QIterated<dim> quadrature_formula(QTrapezoid<1>(),
                                              advection_field.polynomial_degree(this->introspection()));
      const unsigned int dofs_per_cell = this->get_fe().dofs_per_cell;
      const FEValuesExtractors::Scalar field_extractor =
        advection_field.scalar_extractor(this->introspection());

      const std::pair<double, double> global_field_range =
        simulator->get_extrapolated_advection_field_range(advection_field);
      const double global_min_field = global_field_range.first;
      const double global_max_field = global_field_range.second;
      const double extremum_tolerance =
        1e-12 * std::max({1.0,
                          std::abs(global_min_field),
                          std::abs(global_max_field)
                         });

      // q_avg=(q_min+q_max)/2 is formed from extrema of the extrapolated field
      //
      //   q* = (1+dt/dt_old) q_{n-1} - (dt/dt_old) q_{n-2}.
      //
      // The min/max operation is nonsmooth. Split each extremum seed evenly
      // over every sampled quadrature location in its tolerance-defined
      // active set. This deterministic generalized derivative is the same
      // convention used by the finite-difference diagnostics.
      FEValues<dim> fe_values(this->get_mapping(),
                              this->get_fe(),
                              quadrature_formula,
                              update_values);

      std::vector<double> old_field_values(quadrature_formula.size());
      std::vector<double> old_old_field_values(quadrature_formula.size());
      std::vector<types::global_dof_index> local_dof_indices(dofs_per_cell);

      unsigned int local_min_count = 0;
      unsigned int local_max_count = 0;

      for (const auto &cell : this->get_dof_handler().active_cell_iterators())
        if (cell->is_locally_owned())
          {
            fe_values.reinit(cell);
            fe_values[field_extractor].get_function_values(simulator->old_solution,
                                                           old_field_values);
            if (snapshot.timestep_number > 1)
              fe_values[field_extractor].get_function_values(simulator->old_old_solution,
                                                             old_old_field_values);

            for (unsigned int q = 0; q < fe_values.n_quadrature_points; ++q)
              {
                const double extrapolated_field =
                  (snapshot.timestep_number > 1)
                  ? (1.0 + snapshot.time_step / snapshot.old_time_step) * old_field_values[q]
                  - snapshot.time_step / snapshot.old_time_step * old_old_field_values[q]
                  : old_field_values[q];
                if (std::abs(extrapolated_field - global_min_field) <= extremum_tolerance)
                  ++local_min_count;
                if (std::abs(extrapolated_field - global_max_field) <= extremum_tolerance)
                  ++local_max_count;
              }
          }

      const unsigned int global_min_count =
        Utilities::MPI::sum(local_min_count, this->get_mpi_communicator());
      const unsigned int global_max_count =
        Utilities::MPI::sum(local_max_count, this->get_mpi_communicator());
      AssertThrow(global_min_count > 0 && global_max_count > 0,
                  ExcInternalError());

      this->get_pcout() << "   "
                        << (advection_field.is_temperature() ? "Temperature" : "Composition")
                        << "-advection extrapolated field range active min/max locations = "
                        << global_min_count << "/" << global_max_count
                        << std::endl;

      // Since q_avg=0.5(q_min+q_max), both extrema receive half of the
      // incoming seed. Division by the active-set cardinality ensures that a
      // plateau represents one global extremum rather than multiplying the
      // derivative by the number of MPI-owned samples on that plateau.
      const double bar_min_extrapolated =
        0.5 * bar_average_field / static_cast<double>(global_min_count);
      const double bar_max_extrapolated =
        0.5 * bar_average_field / static_cast<double>(global_max_count);
      const double old_factor =
        (snapshot.timestep_number > 1)
        ? 1.0 + snapshot.time_step / snapshot.old_time_step
        : 1.0;
      const double old_old_factor =
        (snapshot.timestep_number > 1)
        ? -snapshot.time_step / snapshot.old_time_step
        : 0.0;

      // Apply the transpose of the extrapolation map. BDF1 has only q_{n-1};
      // BDF2 sends old_factor*bar(q*) to q_{n-1} and
      // old_old_factor*bar(q*) to q_{n-2}. The FE scatter below is precisely
      // the transpose of evaluating those history fields at quadrature
      // points.
      Vector<double> local_previous(dofs_per_cell);
      Vector<double> local_previous_previous(dofs_per_cell);

      for (const auto &cell : this->get_dof_handler().active_cell_iterators())
        if (cell->is_locally_owned())
          {
            fe_values.reinit(cell);
            cell->get_dof_indices(local_dof_indices);
            fe_values[field_extractor].get_function_values(simulator->old_solution,
                                                           old_field_values);
            if (snapshot.timestep_number > 1)
              fe_values[field_extractor].get_function_values(simulator->old_old_solution,
                                                             old_old_field_values);
            local_previous = 0.0;
            local_previous_previous = 0.0;

            for (unsigned int q = 0; q < fe_values.n_quadrature_points; ++q)
              {
                const double extrapolated_field =
                  (snapshot.timestep_number > 1)
                  ? old_factor * old_field_values[q]
                  + old_old_factor * old_old_field_values[q]
                  : old_field_values[q];
                double bar_extrapolated_field = 0.0;
                if (std::abs(extrapolated_field - global_min_field) <= extremum_tolerance)
                  bar_extrapolated_field += bar_min_extrapolated;
                if (std::abs(extrapolated_field - global_max_field) <= extremum_tolerance)
                  bar_extrapolated_field += bar_max_extrapolated;

                if (bar_extrapolated_field == 0.0)
                  continue;

                for (unsigned int i = 0; i < dofs_per_cell; ++i)
                  if (this->get_fe().system_to_component_index(i).first
                      == advection_field.component_index(this->introspection()))
                    {
                      const double phi_field =
                        fe_values[field_extractor].value(i, q);
                      local_previous(i) +=
                        old_factor * bar_extrapolated_field * phi_field;
                      if (snapshot.timestep_number > 1)
                        local_previous_previous(i) +=
                          old_old_factor * bar_extrapolated_field * phi_field;
                    }
              }

            this->get_current_constraints().distribute_local_to_global(local_previous,
                                                                       local_dof_indices,
                                                                       previous_field_adjoint);
            if (snapshot.timestep_number > 1)
              this->get_current_constraints().distribute_local_to_global(local_previous_previous,
                                                                         local_dof_indices,
                                                                         *previous_previous_field_adjoint);
          }

      previous_field_adjoint.compress(VectorOperation::add);
      if (snapshot.timestep_number > 1)
        previous_previous_field_adjoint->compress(VectorOperation::add);
    }



















    template <int dim>
    void
    ArtificialViscosityAdjoint<dim>::add_estimator_transpose(
      const AdvectionField &advection_field,
      const ForwardStateView<dim> &snapshot,
      const std::vector<double> &coefficient_seed,
      LinearAlgebra::BlockVector &previous_field_adjoint,
      LinearAlgebra::BlockVector *previous_previous_field_adjoint,
      LinearAlgebra::BlockVector &previous_stokes_adjoint_rhs,
      LinearAlgebra::BlockVector *previous_previous_stokes_adjoint_rhs,
      std::vector<LinearAlgebra::BlockVector> *previous_composition_adjoints,
      std::vector<LinearAlgebra::BlockVector> *previous_previous_composition_adjoints) const
    {
      // The transpose of a zero coefficient seed is exactly zero. Returning
      // here also avoids calling compress(add) on untouched vectors, whose
      // preceding action may legitimately have been an insert from another
      // reverse contribution.
      const bool locally_active_seed =
        std::any_of(coefficient_seed.begin(), coefficient_seed.end(),
                    [](const double value)
      {
        return value != 0.0;
      });
      if (Utilities::MPI::max(locally_active_seed ? 1 : 0,
                              this->get_mpi_communicator()) == 0)
        return;

      const unsigned int stabilization_alpha =
        this->get_parameters().stabilization_alpha;
      AssertThrow(stabilization_alpha == 1 || stabilization_alpha == 2,
                  ExcMessage("The strict artificial-viscosity estimator adjoint supports only alpha = 1 or alpha = 2."));
      const unsigned int quadrature_degree =
        this->get_fe().base_element(advection_field.base_element(this->introspection())).degree
        + (this->get_parameters().stokes_velocity_degree + 1) / 2;
      const QGauss<dim> quadrature_formula(quadrature_degree);
      const unsigned int dofs_per_cell = this->get_fe().dofs_per_cell;
      const FEValuesExtractors::Scalar field_extractor =
        advection_field.scalar_extractor(this->introspection());

      const std::pair<double, double> global_field_range =
        simulator->get_extrapolated_advection_field_range(advection_field);
      const double average_field =
        0.5 * (global_field_range.first + global_field_range.second);
      const double global_field_variation =
        global_field_range.second - global_field_range.first;
      const double global_entropy_variation =
        stabilization_alpha == 2
        ? simulator->get_entropy_variation(average_field, advection_field)
        : numbers::signaling_nan<double>();
      const double global_max_velocity =
        simulator->get_maximal_velocity(simulator->old_solution);

      // The forward cell coefficient is the smaller of a transport cap and
      // an entropy-residual estimate:
      //
      //   nu_max = beta h a_max u_max,
      //   nu_E2  = c_R h^2 max_q(|R_q| |q_q-q_avg|) / Delta_E,
      //   nu_E1  = c_R h L u_max max_q|R_q|
      //            / (u_global (q_global,max-q_global,min)),
      //   nu_art = min(nu_max,nu_Ealpha).
      //
      // Here a=rho*C_p for temperature and one for composition. The remainder
      // of this routine first reconstructs the active quadrature points and
      // branch, then traverses these expressions in reverse order.

      FEValues<dim> fe_values(this->get_mapping(),
                              this->get_fe(),
                              quadrature_formula,
                              update_values |
                              update_gradients |
                              update_hessians |
                              update_quadrature_points);

      std::vector<double> old_field_values(quadrature_formula.size());
      std::vector<double> old_old_field_values(quadrature_formula.size());
      std::vector<Tensor<1, dim>> old_field_gradients(quadrature_formula.size());
      std::vector<Tensor<1, dim>> old_old_field_gradients(quadrature_formula.size());
      std::vector<double> old_field_laplacians(quadrature_formula.size());
      std::vector<double> old_old_field_laplacians(quadrature_formula.size());
      std::vector<Tensor<1, dim>> old_velocity_values(quadrature_formula.size());
      std::vector<Tensor<1, dim>> old_old_velocity_values(quadrature_formula.size());
      std::vector<SymmetricTensor<2, dim>> old_strain_rates(quadrature_formula.size());
      std::vector<SymmetricTensor<2, dim>> old_old_strain_rates(quadrature_formula.size());
      std::vector<double> old_temperature_values(quadrature_formula.size());
      std::vector<double> old_old_temperature_values(quadrature_formula.size());
      std::vector<double> old_pressure_values(quadrature_formula.size());
      std::vector<double> old_old_pressure_values(quadrature_formula.size());
      std::vector<std::vector<double>> old_composition_values(this->introspection().n_compositional_fields,
                                                               std::vector<double>(quadrature_formula.size()));
      std::vector<std::vector<double>> old_old_composition_values(this->introspection().n_compositional_fields,
                                                                   std::vector<double>(quadrature_formula.size()));

      Vector<double> local_previous_field(dofs_per_cell);
      Vector<double> local_previous_previous_field(dofs_per_cell);
      Vector<double> local_previous_stokes(dofs_per_cell);
      Vector<double> local_previous_previous_stokes(dofs_per_cell);
      std::vector<types::global_dof_index> local_dof_indices(dofs_per_cell);

      const Vector<double> &minimum_physical_diffusion_per_cell =
        simulator->last_advection_physical_diffusion_min_per_cell;
      const Vector<double> &maximum_physical_diffusion_per_cell =
        simulator->last_advection_physical_diffusion_max_per_cell;
      double local_bar_global_entropy_variation = 0.0;
      double local_bar_average_field = 0.0;
      const bool use_bdf2_scheme = (snapshot.timestep_number > 1);

      // Work cell by cell because the forward estimator stores one
      // artificial-viscosity coefficient per active cell. Only cells with a
      // nonzero seed and an artificial-diffusion active branch participate in
      // the reverse pass.
      for (const auto &cell : this->get_dof_handler().active_cell_iterators())
        if (cell->is_locally_owned())
          {
            const unsigned int active_cell_index = cell->active_cell_index();
            const double bar_nu = coefficient_seed[active_cell_index];
            if (bar_nu == 0.0)
              continue;

            const double artificial_viscosity =
              simulator->last_advection_artificial_viscosity_per_cell[active_cell_index];
            if (artificial_viscosity <= minimum_physical_diffusion_per_cell[active_cell_index])
              continue;

            AssertThrow(artificial_viscosity > maximum_physical_diffusion_per_cell[active_cell_index],
                        ExcMessage("The artificial-viscosity estimator adjoint currently supports only cells "
                                   "where max(k, nu_art) selects the same branch at every quadrature point."));

            fe_values.reinit(cell);
            cell->get_dof_indices(local_dof_indices);

            fe_values[field_extractor]
            .get_function_values(simulator->old_solution,
                                 old_field_values);
            fe_values[field_extractor]
            .get_function_values(simulator->old_old_solution,
                                 old_old_field_values);
            fe_values[field_extractor]
            .get_function_gradients(simulator->old_solution,
                                    old_field_gradients);
            fe_values[field_extractor]
            .get_function_gradients(simulator->old_old_solution,
                                    old_old_field_gradients);
            fe_values[field_extractor]
            .get_function_laplacians(simulator->old_solution,
                                     old_field_laplacians);
            fe_values[field_extractor]
            .get_function_laplacians(simulator->old_old_solution,
                                     old_old_field_laplacians);
            fe_values[this->introspection().extractors.velocities]
            .get_function_values(simulator->old_solution,
                                 old_velocity_values);
            fe_values[this->introspection().extractors.velocities]
            .get_function_values(simulator->old_old_solution,
                                 old_old_velocity_values);
            fe_values[this->introspection().extractors.velocities]
            .get_function_symmetric_gradients(simulator->old_solution,
                                              old_strain_rates);
            fe_values[this->introspection().extractors.velocities]
            .get_function_symmetric_gradients(simulator->old_old_solution,
                                              old_old_strain_rates);
            if (advection_field.is_temperature())
              {
                fe_values[this->introspection().extractors.temperature]
                .get_function_values(simulator->old_solution,
                                     old_temperature_values);
                fe_values[this->introspection().extractors.temperature]
                .get_function_values(simulator->old_old_solution,
                                     old_old_temperature_values);
                fe_values[this->introspection().extractors.pressure]
                .get_function_values(simulator->old_solution,
                                     old_pressure_values);
                fe_values[this->introspection().extractors.pressure]
                .get_function_values(simulator->old_old_solution,
                                     old_old_pressure_values);
                for (unsigned int c = 0; c < this->introspection().n_compositional_fields; ++c)
                  {
                    fe_values[this->introspection().extractors.compositional_fields[c]]
                    .get_function_values(simulator->old_solution,
                                         old_composition_values[c]);
                    fe_values[this->introspection().extractors.compositional_fields[c]]
                    .get_function_values(simulator->old_old_solution,
                                         old_old_composition_values[c]);
                  }
              }

            // The estimator contains nested max operations. Retain the
            // winning quadrature locations and the primitive values needed by
            // their vector-Jacobian products. This deliberately follows the
            // same strict '>' tie-breaking as the forward estimator.
            double max_velocity = 0.0;
            unsigned int max_velocity_q = numbers::invalid_unsigned_int;
            double max_velocity_norm = 0.0;
            double max_strain_rate_norm = 0.0;
            double max_residual_measure = 0.0;
            unsigned int max_residual_q = numbers::invalid_unsigned_int;
            double max_residual_value = 0.0;
            double max_residual_entropy_weight = 0.0;
            Tensor<1, dim> max_residual_velocity;
            Tensor<1, dim> max_residual_temperature_gradient;
            double max_residual_advection_prefactor = 1.0;
            double max_residual_transport_residual = 0.0;
            unsigned int max_advection_prefactor_q = numbers::invalid_unsigned_int;

            const double physical_diffusion =
              maximum_physical_diffusion_per_cell[active_cell_index];
            const double h = cell->diameter();
            const double gamma = this->get_parameters().stabilization_gamma;
            double max_advection_prefactor = 1.0;
            std::vector<double> advection_prefactor_values(fe_values.n_quadrature_points, 1.0);
            std::vector<double> conductivity_values(fe_values.n_quadrature_points, physical_diffusion);

            if (advection_field.is_temperature())
              {
                // For temperature, the entropy residual uses material
                // prefactors evaluated at the time-centered old states. These
                // values are needed both to decide the active max locations
                // and to push derivatives back through rho*C_p where needed.
                MaterialModel::MaterialModelInputs<dim> in(fe_values,
                                                           cell,
                                                           this->introspection(),
                                                           simulator->current_linearization_point);
                MaterialModel::MaterialModelOutputs<dim> out(fe_values.n_quadrature_points,
                                                             this->n_compositional_fields());
                in.requested_properties =
                  MaterialModel::MaterialProperties::equation_of_state_properties |
                  MaterialModel::MaterialProperties::thermal_conductivity;

                for (unsigned int q = 0; q < fe_values.n_quadrature_points; ++q)
                  {
                    in.temperature[q] =
                      0.5 * (old_temperature_values[q] + old_old_temperature_values[q]);
                    in.pressure[q] =
                      0.5 * (old_pressure_values[q] + old_old_pressure_values[q]);
                    in.velocity[q] =
                      0.5 * (old_velocity_values[q] + old_old_velocity_values[q]);
                    in.strain_rate[q] =
                      0.5 * (old_strain_rates[q] + old_old_strain_rates[q]);
                    for (unsigned int c = 0; c < this->introspection().n_compositional_fields; ++c)
                      in.composition[q][c] =
                        0.5 * (old_composition_values[c][q]
                               + old_old_composition_values[c][q]);
                  }
                this->get_material_model().evaluate(in, out);

                if (this->get_parameters().formulation_temperature_equation ==
                    Parameters<dim>::Formulation::TemperatureEquation::reference_density_profile)
                  for (unsigned int q = 0; q < fe_values.n_quadrature_points; ++q)
                    out.densities[q] =
                      this->get_adiabatic_conditions().density(fe_values.quadrature_point(q));

                max_advection_prefactor = 0.0;
                for (unsigned int q = 0; q < fe_values.n_quadrature_points; ++q)
                  {
                    advection_prefactor_values[q] =
                      out.densities[q] * out.specific_heat[q];
                    conductivity_values[q] = out.thermal_conductivities[q];
                    if (advection_prefactor_values[q] > max_advection_prefactor)
                      {
                        max_advection_prefactor =
                          advection_prefactor_values[q];
                        max_advection_prefactor_q = q;
                      }
                  }
              }

            auto add_advection_prefactor_composition_transpose =
              [&](const unsigned int q,
                  const double bar_advection_prefactor)
            {
              if (bar_advection_prefactor == 0.0
                  || !advection_field.is_temperature())
                return;

              if (this->get_parameters().formulation_temperature_equation ==
                  Parameters<dim>::Formulation::TemperatureEquation::reference_density_profile)
                return;

              // Local VJP for rho*C_p at a single quadrature point of the
              // entropy-viscosity estimator. The scatter weights are 0.5 and
              // 0.5 because this estimator evaluates material inputs at the
              // midpoint of old and old-old states rather than the BDF
              // extrapolation used in the advection matrix.
              MaterialModel::MaterialModelInputs<dim> in(fe_values,
                                                         cell,
                                                         this->introspection(),
                                                         simulator->current_linearization_point);
              MaterialModel::MaterialModelOutputs<dim> out(fe_values.n_quadrature_points,
                                                           this->n_compositional_fields());

              in.requested_properties =
                MaterialModel::MaterialProperties::equation_of_state_properties;
              for (unsigned int qp = 0; qp < fe_values.n_quadrature_points; ++qp)
                {
                  in.temperature[qp] =
                    0.5 * (old_temperature_values[qp] + old_old_temperature_values[qp]);
                  in.pressure[qp] =
                    0.5 * (old_pressure_values[qp] + old_old_pressure_values[qp]);
                  in.velocity[qp] =
                    0.5 * (old_velocity_values[qp] + old_old_velocity_values[qp]);
                  in.strain_rate[qp] =
                    0.5 * (old_strain_rates[qp] + old_old_strain_rates[qp]);
                  for (unsigned int cc = 0; cc < this->introspection().n_compositional_fields; ++cc)
                    in.composition[qp][cc] =
                      0.5 * (old_composition_values[cc][qp]
                             + old_old_composition_values[cc][qp]);
                }

              this->get_material_model().evaluate(in, out);

              MaterialModelOutputAdjoints<dim> bar_out(fe_values.n_quadrature_points);
              bar_out.densities[q] += out.specific_heat[q] * bar_advection_prefactor;
              bar_out.specific_heat[q] += out.densities[q] * bar_advection_prefactor;

              MaterialModelInputAdjoints<dim> bar_in(fe_values.n_quadrature_points,
                                                     this->n_compositional_fields());
              const MaterialModelAdjoint<dim> material_model_adjoint(*simulator);
              material_model_adjoint.apply_derivative_transpose(in,
                                                                out,
                                                                bar_out,
                                                                bar_in);

              const MaterialModelInputAdjointScatter<dim> material_input_scatter(*simulator);
              material_input_scatter.scatter_temperature_advection_inputs(
                fe_values,
                local_dof_indices,
                bar_in,
                0.5,
                0.5,
                previous_field_adjoint,
                use_bdf2_scheme ? previous_previous_field_adjoint : nullptr,
                previous_composition_adjoints,
                use_bdf2_scheme ? previous_previous_composition_adjoints : nullptr);
            };

            for (unsigned int q = 0; q < fe_values.n_quadrature_points; ++q)
              {
                // Reconstruct the forward active quantities: the maximum
                // velocity measure and the maximum residual measure. Their
                // quadrature locations are saved so the adjoint can follow the
                // same nonsmooth max branches selected in the forward
                // estimator.
                const Tensor<1, dim> velocity =
                  0.5 * (old_velocity_values[q] + old_old_velocity_values[q]);
                const double velocity_norm = velocity.norm();
                const SymmetricTensor<2, dim> strain_rate =
                  0.5 * (old_strain_rates[q] + old_old_strain_rates[q]);
                const double strain_rate_norm = strain_rate.norm();
                const double velocity_measure =
                  velocity_norm + gamma * h * strain_rate_norm;

                if (velocity_measure > max_velocity)
                  {
                    max_velocity = velocity_measure;
                    max_velocity_q = q;
                    max_velocity_norm = velocity_norm;
                    max_strain_rate_norm = strain_rate_norm;
                  }

                // At a quadrature point the strong residual reconstructed by
                // the estimator is
                //
                //   R = a[(q_{n-1}-q_{n-2})/dt_old
                //         + u_mid.grad(q_mid)] - kappa Laplace(q_mid).
                //
                // alpha=2 multiplies |R| by |q_mid-q_avg| before the cellwise
                // maximum; alpha=1 uses |R| directly.
                const double field_dt =
                  snapshot.old_time_step == 0.0
                  ? 0.0
                  : (old_field_values[q] - old_old_field_values[q])
                  / snapshot.old_time_step;
                const Tensor<1, dim> field_gradient =
                  0.5 * (old_field_gradients[q] + old_old_field_gradients[q]);
                const double laplacian =
                  0.5 * (old_field_laplacians[q] + old_old_field_laplacians[q]);
                const double transport_residual =
                  field_dt + velocity * field_gradient;
                const double residual =
                  advection_prefactor_values[q] * transport_residual
                  - conductivity_values[q] * laplacian;
                const double field =
                  0.5 * (old_field_values[q] + old_old_field_values[q]);
                const double entropy_weight =
                  stabilization_alpha == 2
                  ? std::abs(field - average_field)
                  : 1.0;
                const double residual_measure = std::abs(residual) * entropy_weight;

                if (residual_measure > max_residual_measure)
                  {
                    max_residual_measure = residual_measure;
                    max_residual_q = q;
                    max_residual_value = residual;
                    max_residual_entropy_weight = entropy_weight;
                    max_residual_velocity = velocity;
                    max_residual_temperature_gradient = field_gradient;
                    max_residual_advection_prefactor =
                      advection_prefactor_values[q];
                    max_residual_transport_residual = transport_residual;
                  }
              }

            const double beta =
              this->get_parameters().stabilization_beta[advection_field.field_index()];
            const double c_R =
              this->get_parameters().stabilization_c_R[advection_field.field_index()];
            const double maximum_viscosity =
              beta * max_advection_prefactor * max_velocity * h;

            double bar_max_velocity = 0.0;
            double bar_max_advection_prefactor = 0.0;
            double bar_residual_measure = 0.0;

            // Reverse nu_art=min(nu_max,nu_E). Degenerate first-step or zero
            // normalization cases use the same cap fallback as the forward
            // code. Away from a tie, exactly one branch receives bar(nu).
            if (std::abs(global_max_velocity) < 1e-50)
              continue;

            if (snapshot.timestep_number <= 1
                || (stabilization_alpha == 2
                    && std::abs(global_entropy_variation) < 1e-50)
                || std::abs(global_field_variation) < 1e-50)
              {
                // Degenerate/first-step fallback: ASPECT's forward estimator
                // uses the maximum-viscosity cap directly, so the seed only
                // reaches the selected max-velocity and max-prefactor paths.
                bar_max_velocity += beta * max_advection_prefactor * h * bar_nu;
                bar_max_advection_prefactor += beta * max_velocity * h * bar_nu;
              }
            else
              {
                const double entropy_viscosity =
                  (stabilization_alpha == 2)
                  ? c_R * h * h * max_residual_measure / global_entropy_variation
                  : c_R * h * simulator->global_Omega_diameter
                  * max_velocity * max_residual_measure
                  / (global_max_velocity * global_field_variation);

                if (maximum_viscosity < entropy_viscosity)
                  {
                    // The beta*h*u_max cap is active.
                    bar_max_velocity += beta * max_advection_prefactor * h * bar_nu;
                    bar_max_advection_prefactor += beta * max_velocity * h * bar_nu;
                  }
                else if (stabilization_alpha == 2)
                  {
                    // The alpha=2 entropy-viscosity branch is active. Its
                    // denominator is a global entropy variation, so accumulate
                    // a scalar seed and apply its transpose after the cell
                    // loop/MPI reduction.
                    const double entropy_viscosity =
                      c_R * h * h * max_residual_measure / global_entropy_variation;
                    bar_residual_measure +=
                      c_R * h * h / global_entropy_variation * bar_nu;
                    local_bar_global_entropy_variation +=
                      -bar_nu * entropy_viscosity / global_entropy_variation;
                  }
                else
                  {
                    // The alpha=1 entropy-viscosity branch uses a global
                    // field range and global velocity normalization. This
                    // implementation propagates the local max-velocity and
                    // residual-measure pieces used by the supported v1 tests.
                    bar_max_velocity +=
                      c_R * h * simulator->global_Omega_diameter
                      * max_residual_measure
                      / (global_max_velocity * global_field_variation)
                      * bar_nu;
                    bar_residual_measure +=
                      c_R * h * simulator->global_Omega_diameter
                      * max_velocity
                      / (global_max_velocity * global_field_variation)
                      * bar_nu;
                  }
              }

            local_previous_field = 0.0;
            local_previous_previous_field = 0.0;
            local_previous_stokes = 0.0;
            local_previous_previous_stokes = 0.0;

            if (bar_max_velocity != 0.0
                && max_velocity_q != numbers::invalid_unsigned_int
                && max_velocity > 0.0)
              {
                const Tensor<1, dim> velocity =
                  0.5 * (old_velocity_values[max_velocity_q]
                         + old_old_velocity_values[max_velocity_q]);
                const SymmetricTensor<2, dim> strain_rate =
                  0.5 * (old_strain_rates[max_velocity_q]
                         + old_old_strain_rates[max_velocity_q]);
                // u_measure=|u|+gamma h |eps(u)|. Its VJP uses the normalized
                // velocity and strain tensors; a zero norm contributes the
                // zero subgradient. Both inputs are midpoint states, hence
                // the factor 0.5 in the FE scatter to n-1 and n-2.
                const Tensor<1, dim> bar_velocity =
                  (max_velocity_norm > 0.0)
                  ? bar_max_velocity * velocity / max_velocity_norm
                  : Tensor<1, dim>();
                const SymmetricTensor<2, dim> bar_strain_rate =
                  (max_strain_rate_norm > 0.0)
                  ? bar_max_velocity * gamma * h * strain_rate / max_strain_rate_norm
                  : SymmetricTensor<2, dim>();

                for (unsigned int i = 0; i < dofs_per_cell; ++i)
                  if (this->introspection().component_masks.velocities[
                        this->get_fe().system_to_component_index(i).first])
                    {
                      const Tensor<1, dim> phi_u =
                        fe_values[this->introspection().extractors.velocities].value(i, max_velocity_q);
                      const SymmetricTensor<2, dim> eps_phi_u =
                        fe_values[this->introspection().extractors.velocities].symmetric_gradient(i, max_velocity_q);
                      const double contribution =
                        0.5 * ((bar_velocity * phi_u) + (bar_strain_rate * eps_phi_u));
                      local_previous_stokes(i) += contribution;
                      if (snapshot.timestep_number > 1)
                        local_previous_previous_stokes(i) += contribution;
                      else
                        local_previous_stokes(i) += contribution;
                    }
              }

            if (bar_max_advection_prefactor != 0.0
                && max_advection_prefactor_q != numbers::invalid_unsigned_int)
              add_advection_prefactor_composition_transpose(max_advection_prefactor_q,
                                                            bar_max_advection_prefactor);

            if (bar_residual_measure != 0.0
                && max_residual_q != numbers::invalid_unsigned_int)
              {
                // Reverse the selected residual-measure quadrature point:
                // abs(residual) contributes sign(residual), alpha=2 also
                // contributes through abs(field-average_field), and the
                // transport residual splits into velocity, field gradient, and
                // time-derivative terms.
                const unsigned int q = max_residual_q;
                const double residual_sign =
                  (max_residual_value > 0.0) ? 1.0 : -1.0;
                const double field =
                  0.5 * (old_field_values[q] + old_old_field_values[q]);
                const double entropy_weight_sign =
                  (field - average_field > 0.0) ? 1.0 : -1.0;

                // Reverse
                //   M=|R| w,  w=|q_mid-q_avg| (alpha=2),
                //   R=a r_transport-kappa Laplace(q_mid),
                //   r_transport=q_t+u_mid.grad(q_mid).
                // The scalar/tensor seeds below are the partial derivatives
                // of this local computation graph before FE transposition.
                const double bar_residual =
                  residual_sign * max_residual_entropy_weight * bar_residual_measure;
                const double bar_field =
                  (stabilization_alpha == 2)
                  ? entropy_weight_sign * std::abs(max_residual_value) * bar_residual_measure
                  : 0.0;
                if (stabilization_alpha == 2)
                  local_bar_average_field -=
                    entropy_weight_sign * std::abs(max_residual_value) * bar_residual_measure;
                const double bar_transport_residual =
                  max_residual_advection_prefactor * bar_residual;
                const double bar_advection_prefactor =
                  max_residual_transport_residual * bar_residual;
                const Tensor<1, dim> bar_velocity =
                  bar_transport_residual * max_residual_temperature_gradient;
                const Tensor<1, dim> bar_temperature_gradient =
                  bar_transport_residual * max_residual_velocity;
                const double bar_laplacian =
                  -conductivity_values[q] * bar_residual;

                add_advection_prefactor_composition_transpose(q,
                                                              bar_advection_prefactor);

                for (unsigned int i = 0; i < dofs_per_cell; ++i)
                  {
                    const unsigned int component =
                      this->get_fe().system_to_component_index(i).first;

                    if (component == advection_field.component_index(this->introspection()))
                      {
                        const double phi_field =
                          fe_values[field_extractor].value(i, q);
                        const Tensor<1, dim> grad_phi_field =
                          fe_values[field_extractor].gradient(i, q);
                        const double laplacian_phi_field =
                          trace(fe_values[field_extractor].hessian(i, q));

                        // q_mid contributes through the entropy weight,
                        // gradient, and Laplacian. The time difference
                        // contributes +bar(r_transport)/dt_old to q_{n-1}
                        // and the negative counterpart to q_{n-2}.
                        double previous_contribution =
                          0.5 * bar_field * phi_field
                          + 0.5 * (bar_temperature_gradient * grad_phi_field)
                          + 0.5 * bar_laplacian * laplacian_phi_field;
                        double previous_previous_contribution =
                          previous_contribution;

                        if (snapshot.old_time_step != 0.0)
                          {
                            previous_contribution += bar_transport_residual / snapshot.old_time_step * phi_field;
                            previous_previous_contribution -= bar_transport_residual / snapshot.old_time_step * phi_field;
                          }

                        local_previous_field(i) += previous_contribution;
                        if (snapshot.timestep_number > 1)
                          local_previous_previous_field(i) += previous_previous_contribution;
                        else
                          // On the first step ASPECT stores only q_{n-1}; both
                          // midpoint/history contributions collapse onto that
                          // single previous state.
                          local_previous_field(i) += previous_previous_contribution;
                      }

                    if (this->introspection().component_masks.velocities[component])
                      {
                        const Tensor<1, dim> phi_u =
                          fe_values[this->introspection().extractors.velocities].value(i, q);
                        const double contribution = 0.5 * (bar_velocity * phi_u);
                        local_previous_stokes(i) += contribution;
                        if (snapshot.timestep_number > 1)
                          local_previous_previous_stokes(i) += contribution;
                        else
                          // Same BDF1 collapse as for field values: there is no
                          // separate q_{n-2} Stokes state on the first step.
                          local_previous_stokes(i) += contribution;
                      }
                  }
              }

            this->get_current_constraints().distribute_local_to_global(local_previous_field,
                                                                       local_dof_indices,
                                                                       previous_field_adjoint);
            this->get_current_constraints().distribute_local_to_global(local_previous_stokes,
                                                                       local_dof_indices,
                                                                       previous_stokes_adjoint_rhs);

            if (snapshot.timestep_number > 1)
              {
                AssertThrow(previous_previous_field_adjoint != nullptr &&
                            previous_previous_stokes_adjoint_rhs != nullptr,
                            ExcInternalError());
                this->get_current_constraints().distribute_local_to_global(local_previous_previous_field,
                                                                           local_dof_indices,
                                                                           *previous_previous_field_adjoint);
                this->get_current_constraints().distribute_local_to_global(local_previous_previous_stokes,
                                                                           local_dof_indices,
                                                                           *previous_previous_stokes_adjoint_rhs);
              }
          }

      previous_field_adjoint.compress(VectorOperation::add);
      previous_stokes_adjoint_rhs.compress(VectorOperation::add);
      if (previous_composition_adjoints != nullptr)
        for (auto &composition_adjoint : *previous_composition_adjoints)
          composition_adjoint.compress(VectorOperation::add);

      if (snapshot.timestep_number > 1)
        {
          previous_previous_field_adjoint->compress(VectorOperation::add);
          previous_previous_stokes_adjoint_rhs->compress(VectorOperation::add);
          if (previous_previous_composition_adjoints != nullptr)
            for (auto &composition_adjoint : *previous_previous_composition_adjoints)
              composition_adjoint.compress(VectorOperation::add);
        }

      if (stabilization_alpha == 2 && snapshot.timestep_number > 1)
        {
          // Finish the two nonlocal paths only after the cell loop. Every
          // rank may contribute to bar(Delta_E) and bar(q_avg), while the
          // primal quantities were MPI-global. Summing the scalar seeds first
          // is therefore the transpose of broadcasting those primal scalars;
          // the two helpers then transpose the entropy variation and the
          // extrapolated min/max range, respectively.
          const double bar_global_entropy_variation =
            Utilities::MPI::sum(local_bar_global_entropy_variation,
                                this->get_mpi_communicator());
          double bar_average_field =
            Utilities::MPI::sum(local_bar_average_field,
                                this->get_mpi_communicator());
          add_global_entropy_variation_transpose(advection_field,
                                                 bar_global_entropy_variation,
                                                 bar_average_field,
                                                 previous_field_adjoint,
                                                 previous_previous_field_adjoint);
          add_extrapolated_field_average_transpose(advection_field,
                                                   snapshot,
                                                   bar_average_field,
                                                   previous_field_adjoint,
                                                   previous_previous_field_adjoint);
        }
    }
  }

    template <int dim>
    void
    DynamicTopographyBoundaryOperator<dim>::add_state_transpose(
      const LinearAlgebra::BlockVector &topography_adjoint,
      const double boundary_area,
      const double pressure_weight,
      const double density_exterior,
      LinearAlgebra::BlockVector &rhs,
      const Postprocess::DynamicTopographyBoundary boundary) const
    {
      AssertThrow(boundary_area > std::numeric_limits<double>::min(),
                  ExcMessage("DynamicTopographyBoundaryOperator found zero selected-boundary area."));

      // The forward CBF postprocessor computes nodal topography from reduced
      // normal stress as
      //   h_a = (M_a)^{-1} r_a / (Delta rho_a |g_a|),
      // where M_a is the diagonal GLL face mass and r_a is the boundary-normal
      // projection of the discrete Stokes residual. This routine applies the
      // exact reverse chain
      //   bar r_a += (M_a)^{-T} bar h_a/(Delta rho_a |g_a|)
      // followed by the transpose of the volume Stokes bilinear form.
      const types::boundary_id boundary_id = this->get_geometry_model().
                                             translate_symbolic_boundary_name_to_id(boundary == Postprocess::DynamicTopographyBoundary::surface ? "top" : "bottom");
      const unsigned int quadrature_degree =
        this->get_fe().base_element(this->introspection().base_elements.velocities).degree + 1;
      const QGauss<dim> quadrature_formula(quadrature_degree);
      const QGauss<dim-1> quadrature_formula_face(quadrature_degree);
      const QGaussLobatto<dim-1> cbf_face_quadrature_formula(quadrature_degree);
      const unsigned int dofs_per_cell = this->get_fe().dofs_per_cell;

      FEValues<dim> fe_values(this->get_mapping(), this->get_fe(), quadrature_formula,
                              update_values | update_gradients | update_quadrature_points | update_JxW_values);
      FEFaceValues<dim> fe_face_values(this->get_mapping(), this->get_fe(), quadrature_formula_face,
                                       update_values | update_gradients | update_normal_vectors |
                                       update_quadrature_points | update_JxW_values);
      FEFaceValues<dim> fe_cbf_face_values(this->get_mapping(), this->get_fe(), cbf_face_quadrature_formula,
                                           update_values | update_JxW_values);
      const std::vector<Point<dim-1>> face_support_points =
        this->get_fe().base_element(this->introspection().base_elements.temperature).get_unit_face_support_points();
      const Quadrature<dim-1> support_quadrature(face_support_points);
      FEFaceValues<dim> fe_support_values(this->get_mapping(), this->get_fe(), support_quadrature,
                                          update_values | update_gradients | update_normal_vectors |
                                          update_quadrature_points);

      Vector<double> local_rhs(dofs_per_cell);
      Vector<double> local_cbf_mass_matrix(dofs_per_cell);
      Vector<double> local_cbf_state_adjoint(dofs_per_cell);
      std::vector<types::global_dof_index> local_dof_indices(dofs_per_cell);
      std::vector<types::global_dof_index> face_dof_indices(this->get_fe().dofs_per_face);
      LinearAlgebra::BlockVector cbf_mass_matrix(this->introspection().index_sets.system_partitioning,
                                                 this->introspection().index_sets.system_relevant_partitioning,
                                                 this->get_mpi_communicator());
      cbf_mass_matrix = 0.0;

      // Reassemble precisely the diagonal GLL face mass used by the forward
      // CBF calculation. It is a global nodal quantity: neighboring boundary
      // cells contribute to the same face DoF and compress(add) completes the
      // distributed sum. Its construction remains independent of bar h.
      for (const auto &cell : this->get_dof_handler().active_cell_iterators())
        if (cell->is_locally_owned())
          for (const unsigned int f : cell->face_indices())
            if (cell->at_boundary(f) && cell->face(f)->boundary_id() == boundary_id)
              {
                fe_cbf_face_values.reinit(cell, f);
                local_cbf_mass_matrix = 0.0;
                for (unsigned int q = 0; q < cbf_face_quadrature_formula.size(); ++q)
                  for (unsigned int i = 0; i < dofs_per_cell; ++i)
                    {
                      const Tensor<1, dim> phi_u =
                        fe_cbf_face_values[this->introspection().extractors.velocities].value(i, q);
                      local_cbf_mass_matrix(i) += phi_u * phi_u * fe_cbf_face_values.JxW(q);
                    }
                cell->distribute_local_to_global(local_cbf_mass_matrix, cbf_mass_matrix);
              }
      cbf_mass_matrix.compress(VectorOperation::add);

      // Convert each topography seed into an adjoint of the CBF residual. The
      // temperature support points are also the locations where the forward
      // code samples density and gravity, so using any other quadrature here
      // would transpose a different discrete operator.
      for (const auto &cell : this->get_dof_handler().active_cell_iterators())
        if (cell->is_locally_owned())
          for (const unsigned int f : cell->face_indices())
            if (cell->at_boundary(f) && cell->face(f)->boundary_id() == boundary_id)
              {
                fe_values.reinit(cell);
                fe_support_values.reinit(cell, f);
                local_rhs = 0.0;
                local_cbf_state_adjoint = 0.0;

                MaterialModel::MaterialModelInputs<dim> support_in(fe_support_values, cell,
                                                                   this->introspection(), this->get_solution());
                MaterialModel::MaterialModelOutputs<dim> support_out(fe_support_values.n_quadrature_points,
                                                                     this->n_compositional_fields());
                support_in.requested_properties = MaterialModel::MaterialProperties::density;
                this->get_material_model().evaluate(support_in, support_out);
                cell->get_dof_indices(local_dof_indices);
                cell->face(f)->get_dof_indices(face_dof_indices);

                for (unsigned int face_i = 0; face_i < face_dof_indices.size(); ++face_i)
                  {
                    const std::pair<unsigned int, unsigned int> component_index =
                      this->get_fe().face_system_to_component_index(face_i);
                    const unsigned int component = component_index.first;
                    const unsigned int support_index = component_index.second;
                    if (component == this->introspection().component_indices.temperature)
                      {
                        const Tensor<1,dim> normal = fe_support_values.normal_vector(support_index);
                        const Tensor<1,dim> gravity = this->get_gravity_model().gravity_vector(
                                                        fe_support_values.quadrature_point(support_index));
                        const double gravity_norm = gravity.norm();
                        const double density_contrast = support_out.densities[support_index] - density_exterior;
                        AssertThrow(std::abs(density_contrast) > std::numeric_limits<double>::min(),
                                    ExcMessage("DynamicTopographyBoundaryOperator encountered zero density contrast at the selected boundary."));
                        AssertThrow(gravity_norm > std::numeric_limits<double>::min(),
                                    ExcMessage("DynamicTopographyBoundaryOperator encountered zero gravity magnitude at the selected boundary."));
                        const double coefficient = topography_adjoint(face_dof_indices[face_i]) /
                                                   (density_contrast * gravity_norm);
                        // M is diagonal in the GLL nodal basis. Distribute the
                        // scalar normal-stress seed over velocity components
                        // through n dot phi_u and divide by the assembled nodal
                        // mass. This is M^{-T}=M^{-1} for the diagonal mass.
                        for (unsigned int velocity_face_i = 0; velocity_face_i < face_dof_indices.size(); ++velocity_face_i)
                          {
                            const std::pair<unsigned int, unsigned int> velocity_component_index =
                              this->get_fe().face_system_to_component_index(velocity_face_i);
                            if (this->introspection().component_masks.velocities[velocity_component_index.first] == false)
                              continue;
                            const auto local_dof = std::find(local_dof_indices.begin(), local_dof_indices.end(),
                                                             face_dof_indices[velocity_face_i]);
                            AssertThrow(local_dof != local_dof_indices.end(), ExcInternalError());
                            const double mass = cbf_mass_matrix(face_dof_indices[velocity_face_i]);
                            if (mass <= std::numeric_limits<double>::min())
                              continue;
                            const unsigned int local_index = local_dof - local_dof_indices.begin();
                            const Tensor<1, dim> phi_u = fe_support_values[
                                                           this->introspection().extractors.velocities].value(local_index, support_index);
                            local_cbf_state_adjoint(local_index) += coefficient * (phi_u * normal) / mass;
                          }
                      }
                  }

                MaterialModel::MaterialModelInputs<dim> volume_in(fe_values, cell,
                                                                  this->introspection(), this->get_solution());
                MaterialModel::MaterialModelOutputs<dim> volume_out(fe_values.n_quadrature_points,
                                                                    this->n_compositional_fields());
                volume_in.requested_properties = MaterialModel::MaterialProperties::viscosity;
                this->get_material_model().evaluate(volume_in, volume_out);
                // The CBF residual uses the Stokes bilinear form
                //   a(u,v)=int 2 eta eps(u):eps(v)-p div(v) dV.
                // With bar r represented by local_cbf_state_adjoint, applying
                // a(.,.)^T produces the velocity/pressure RHS below. The loop
                // order is algebraic only; it does not introduce a new
                // continuous boundary formulation.
                for (unsigned int q = 0; q < quadrature_formula.size(); ++q)
                  for (unsigned int j = 0; j < dofs_per_cell; ++j)
                    if (local_cbf_state_adjoint(j) != 0.0)
                      {
                        const SymmetricTensor<2,dim> epsilon_phi_j =
                          fe_values[this->introspection().extractors.velocities].symmetric_gradient(j, q);
                        const double div_phi_j =
                          fe_values[this->introspection().extractors.velocities].divergence(j, q);
                        for (unsigned int i = 0; i < dofs_per_cell; ++i)
                          {
                            const SymmetricTensor<2,dim> epsilon_phi_i =
                              fe_values[this->introspection().extractors.velocities].symmetric_gradient(i, q);
                            const double pressure_phi_i =
                              fe_values[this->introspection().extractors.pressure].value(i, q);
                            local_rhs(i) += local_cbf_state_adjoint(j) *
                                            (2.0 * volume_out.viscosities[q] * (epsilon_phi_j * epsilon_phi_i)
                                             - this->get_pressure_scaling() * div_phi_j * pressure_phi_i) * fe_values.JxW(q);
                          }
                      }

                fe_face_values.reinit(cell, f);
                // Dynamic topography subtracts the boundary mean pressure.
                // Reverse that scalar mean by spreading pressure_weight/A
                // uniformly over the selected boundary pressure test space.
                for (unsigned int q = 0; q < quadrature_formula_face.size(); ++q)
                  for (unsigned int i = 0; i < dofs_per_cell; ++i)
                    local_rhs(i) += this->get_pressure_scaling() * pressure_weight / boundary_area
                                    * fe_face_values[this->introspection().extractors.pressure].value(i, q)
                                    * fe_face_values.JxW(q);
                this->get_current_constraints().distribute_local_to_global(local_rhs, local_dof_indices, rhs);
              }
      rhs.compress(VectorOperation::add);

      if (this->get_parameters().pressure_normalization != "no")
        {
          // ASPECT's pressure normalization quotients out the constant
          // pressure mode. Project the assembled adjoint RHS onto the same
          // quotient space: first measure its component along the scaled
          // constant-pressure vector, then subtract the corresponding
          // boundary-average functional. The MPI sum is required because the
          // pressure vector is distributed by ownership.
          const unsigned int pressure_block_index = this->introspection().block_indices.pressure;
          const double inverse_pressure_scaling = 1.0 / this->get_pressure_scaling();
          const IndexSet local_pressure_dofs = rhs.block(pressure_block_index).locally_owned_elements();
          double local_pressure_constant_rhs_dot = 0.0;
          for (unsigned int k = 0; k < local_pressure_dofs.n_elements(); ++k)
            {
              const types::global_dof_index local_index = local_pressure_dofs.nth_index_in_set(k);
              local_pressure_constant_rhs_dot += rhs.block(pressure_block_index)[local_index] * inverse_pressure_scaling;
            }
          const double pressure_constant_rhs_dot = Utilities::MPI::sum(local_pressure_constant_rhs_dot,
                                                                       this->get_mpi_communicator());
          for (const auto &cell : this->get_dof_handler().active_cell_iterators())
            if (cell->is_locally_owned())
              for (const unsigned int f : cell->face_indices())
                if (cell->at_boundary(f) && cell->face(f)->boundary_id() == boundary_id)
                  {
                    fe_face_values.reinit(cell, f);
                    local_rhs = 0.0;
                    cell->get_dof_indices(local_dof_indices);
                    for (unsigned int q = 0; q < quadrature_formula_face.size(); ++q)
                      for (unsigned int i = 0; i < dofs_per_cell; ++i)
                        local_rhs(i) += -pressure_constant_rhs_dot * this->get_pressure_scaling() / boundary_area
                                        * fe_face_values[this->introspection().extractors.pressure].value(i, q)
                                        * fe_face_values.JxW(q);
                    this->get_current_constraints().distribute_local_to_global(local_rhs, local_dof_indices, rhs);
                  }
          rhs.compress(VectorOperation::add);
        }
    }



    template <int dim>
    void
    DynamicTopographyBoundaryOperator<dim>::add_direct_kernel_contributions(
      const LinearAlgebra::BlockVector &topography_adjoint,
      const LinearAlgebra::BlockVector &topography_vector,
      const double density_exterior,
      const std::string &objective_name,
      KernelRepository<dim> &kernels,
      const Postprocess::DynamicTopographyBoundary boundary) const
    {
      // Besides its dependence through the Stokes state, h=r/(Delta rho |g|)
      // has direct material derivatives. This routine accumulates cellwise
      // density and viscosity kernels for
      //   bar rho   = -bar h * h / Delta rho
      // and
      //   bar eta   = bar h/(Delta rho |g|) * dr/deta.
      // The second term includes the eta-dependence of the CBF volume residual;
      // the first includes both the denominator and buoyancy-force paths.
      const unsigned int n_active_cells = this->get_triangulation().n_active_cells();
      const unsigned int quadrature_degree =
        this->get_fe().base_element(this->introspection().base_elements.velocities).degree + 1;
      const QGauss<dim> quadrature_formula(quadrature_degree);
      const QGauss<dim-1> face_quadrature_formula(quadrature_degree);
      const QGaussLobatto<dim-1> cbf_face_quadrature_formula(quadrature_degree);
      const types::boundary_id boundary_id = this->get_geometry_model().
                                             translate_symbolic_boundary_name_to_id(boundary == Postprocess::DynamicTopographyBoundary::surface ? "top" : "bottom");
      const unsigned int dofs_per_cell = this->get_fe().dofs_per_cell;
      FEValues<dim> fe_values(this->get_mapping(), this->get_fe(), quadrature_formula,
                              update_values | update_gradients | update_quadrature_points | update_JxW_values);
      FEFaceValues<dim> fe_cbf_face_values(this->get_mapping(), this->get_fe(), cbf_face_quadrature_formula,
                                           update_values | update_JxW_values);
      const std::vector<Point<dim-1>> face_support_points =
        this->get_fe().base_element(this->introspection().base_elements.temperature).get_unit_face_support_points();
      const Quadrature<dim-1> support_quadrature(face_support_points);
      FEFaceValues<dim> fe_support_values(this->get_mapping(), this->get_fe(), support_quadrature,
                                          update_values | update_gradients | update_normal_vectors |
                                          update_quadrature_points);
      Vector<double> cell_volumes(n_active_cells);
      Vector<double> density_surface(n_active_cells);
      Vector<double> viscosity_surface(n_active_cells);
      Vector<double> local_cbf_mass_matrix(dofs_per_cell);
      Vector<double> local_density_cbf_rhs(dofs_per_cell);
      Vector<double> local_viscosity_cbf_rhs(dofs_per_cell);
      std::vector<types::global_dof_index> local_dof_indices(dofs_per_cell);
      std::vector<types::global_dof_index> face_dof_indices(this->get_fe().dofs_per_face);
      std::vector<double> support_topo_values(support_quadrature.size());
      LinearAlgebra::BlockVector cbf_mass_matrix(this->introspection().index_sets.system_partitioning,
                                                 this->introspection().index_sets.system_relevant_partitioning,
                                                 this->get_mpi_communicator());
      cbf_mass_matrix = 0.0;

      // Kernels are stored as cell averages. Accumulate physical cell volumes
      // now so the integrated sensitivities formed below can be normalized
      // once, after all boundary faces and shared topography DoFs are handled.
      for (const auto &cell : this->get_dof_handler().active_cell_iterators())
        if (cell->is_locally_owned())
          {
            fe_values.reinit(cell);
            const unsigned int cell_index = cell->active_cell_index();
            for (unsigned int q = 0; q < quadrature_formula.size(); ++q)
              cell_volumes(cell_index) += fe_values.JxW(q);
          }

      for (const auto &cell : this->get_dof_handler().active_cell_iterators())
        if (cell->is_locally_owned())
          for (const unsigned int f : cell->face_indices())
            if (cell->at_boundary(f) && cell->face(f)->boundary_id() == boundary_id)
              {
                fe_cbf_face_values.reinit(cell, f);
                local_cbf_mass_matrix = 0.0;
                for (unsigned int q = 0; q < cbf_face_quadrature_formula.size(); ++q)
                  for (unsigned int i = 0; i < dofs_per_cell; ++i)
                    {
                      const Tensor<1, dim> phi_u =
                        fe_cbf_face_values[this->introspection().extractors.velocities].value(i, q);
                      local_cbf_mass_matrix(i) += phi_u * phi_u * fe_cbf_face_values.JxW(q);
                    }
                cell->distribute_local_to_global(local_cbf_mass_matrix, cbf_mass_matrix);
              }
      cbf_mass_matrix.compress(VectorOperation::add);

      // Different boundary cells can see the same continuous topography DoF.
      // Record its explicit denominator derivative once by global DoF index,
      // rather than once per adjacent face. The associated active cell selects
      // where this surface sensitivity is represented in the cellwise kernel.
      std::map<types::global_dof_index, std::pair<unsigned int, double>> density_surface_by_topography_dof;
      for (const auto &cell : this->get_dof_handler().active_cell_iterators())
        if (cell->is_locally_owned())
          for (const unsigned int f : cell->face_indices())
            if (cell->at_boundary(f) && cell->face(f)->boundary_id() == boundary_id)
              {
                fe_support_values.reinit(cell, f);
                MaterialModel::MaterialModelInputs<dim> support_in(fe_support_values, cell,
                                                                   this->introspection(), this->get_solution());
                MaterialModel::MaterialModelOutputs<dim> support_out(fe_support_values.n_quadrature_points,
                                                                     this->n_compositional_fields());
                support_in.requested_properties = MaterialModel::MaterialProperties::density;
                this->get_material_model().evaluate(support_in, support_out);
                fe_support_values[this->introspection().extractors.temperature].get_function_values(
                  topography_vector, support_topo_values);
                cell->face(f)->get_dof_indices(face_dof_indices);
                for (unsigned int face_i = 0; face_i < face_dof_indices.size(); ++face_i)
                  {
                    const std::pair<unsigned int, unsigned int> component_index =
                      this->get_fe().face_system_to_component_index(face_i);
                    const unsigned int support_index = component_index.second;
                    if (component_index.first == this->introspection().component_indices.temperature)
                      {
                        const Tensor<1,dim> gravity = this->get_gravity_model().gravity_vector(
                                                        fe_support_values.quadrature_point(support_index));
                        const double gravity_norm = gravity.norm();
                        const double density_contrast = support_out.densities[support_index] - density_exterior;
                        AssertThrow(std::abs(density_contrast) > std::numeric_limits<double>::min(),
                                    ExcMessage("DynamicTopographyBoundaryOperator encountered zero density contrast at the selected boundary."));
                        AssertThrow(gravity_norm > std::numeric_limits<double>::min(),
                                    ExcMessage("DynamicTopographyBoundaryOperator encountered zero gravity magnitude at the selected boundary."));
                        (void)gravity_norm;
                        density_surface_by_topography_dof[face_dof_indices[face_i]] =
                          std::make_pair(cell->active_cell_index(),
                                         -topography_adjoint(face_dof_indices[face_i]) *
                                         support_topo_values[support_index] / density_contrast);
                      }
                  }
              }

      for (const auto &cell : this->get_dof_handler().active_cell_iterators())
        if (cell->is_locally_owned())
          for (const unsigned int f : cell->face_indices())
            if (cell->at_boundary(f) && cell->face(f)->boundary_id() == boundary_id)
              {
                fe_values.reinit(cell);
                fe_support_values.reinit(cell, f);
                local_density_cbf_rhs = 0.0;
                local_viscosity_cbf_rhs = 0.0;
                MaterialModel::MaterialModelInputs<dim> volume_in(fe_values, cell,
                                                                  this->introspection(), this->get_solution());
                MaterialModel::MaterialModelOutputs<dim> volume_out(fe_values.n_quadrature_points,
                                                                    this->n_compositional_fields());
                volume_in.requested_properties = MaterialModel::MaterialProperties::viscosity;
                this->get_material_model().evaluate(volume_in, volume_out);
                // Differentiate the two material coefficients in the forward
                // Stokes residual. For density, dr/drho contains -g dot phi_u;
                // for viscosity, dr/deta contains 2 eps(phi_u):eps(u). These
                // local vectors are subsequently projected through the same
                // diagonal CBF mass and boundary normal used by the state path.
                for (unsigned int q = 0; q < quadrature_formula.size(); ++q)
                  {
                    const Tensor<1, dim> gravity = this->get_gravity_model().gravity_vector(
                                                     fe_values.quadrature_point(q));
                    for (unsigned int i = 0; i < dofs_per_cell; ++i)
                      {
                        const Tensor<1, dim> phi_u =
                          fe_values[this->introspection().extractors.velocities].value(i, q);
                        const SymmetricTensor<2, dim> epsilon_phi_u =
                          fe_values[this->introspection().extractors.velocities].symmetric_gradient(i, q);
                        local_density_cbf_rhs(i) -= (gravity * phi_u) * fe_values.JxW(q);
                        local_viscosity_cbf_rhs(i) += 2.0 * (epsilon_phi_u * volume_in.strain_rate[q]) * fe_values.JxW(q);
                      }
                  }
                MaterialModel::MaterialModelInputs<dim> support_in(fe_support_values, cell,
                                                                   this->introspection(), this->get_solution());
                MaterialModel::MaterialModelOutputs<dim> support_out(fe_support_values.n_quadrature_points,
                                                                     this->n_compositional_fields());
                support_in.requested_properties = MaterialModel::MaterialProperties::density;
                this->get_material_model().evaluate(support_in, support_out);
                cell->get_dof_indices(local_dof_indices);
                cell->face(f)->get_dof_indices(face_dof_indices);
                for (unsigned int face_i = 0; face_i < face_dof_indices.size(); ++face_i)
                  {
                    const std::pair<unsigned int, unsigned int> component_index =
                      this->get_fe().face_system_to_component_index(face_i);
                    const unsigned int support_index = component_index.second;
                    if (component_index.first == this->introspection().component_indices.temperature)
                      {
                        const Tensor<1,dim> normal = fe_support_values.normal_vector(support_index);
                        const Tensor<1,dim> gravity = this->get_gravity_model().gravity_vector(
                                                        fe_support_values.quadrature_point(support_index));
                        const double gravity_norm = gravity.norm();
                        const double density_contrast = support_out.densities[support_index] - density_exterior;
                        double density_force_normal_stress_derivative = 0.0;
                        double normal_stress_derivative = 0.0;
                        AssertThrow(std::abs(density_contrast) > std::numeric_limits<double>::min(),
                                    ExcMessage("DynamicTopographyBoundaryOperator encountered zero density contrast at the selected boundary."));
                        AssertThrow(gravity_norm > std::numeric_limits<double>::min(),
                                    ExcMessage("DynamicTopographyBoundaryOperator encountered zero gravity magnitude at the selected boundary."));
                        for (unsigned int velocity_face_i = 0; velocity_face_i < face_dof_indices.size(); ++velocity_face_i)
                          {
                            const std::pair<unsigned int, unsigned int> velocity_component_index =
                              this->get_fe().face_system_to_component_index(velocity_face_i);
                            if (this->introspection().component_masks.velocities[velocity_component_index.first] == false)
                              continue;
                            const auto local_dof = std::find(local_dof_indices.begin(), local_dof_indices.end(),
                                                             face_dof_indices[velocity_face_i]);
                            AssertThrow(local_dof != local_dof_indices.end(), ExcInternalError());
                            const double mass = cbf_mass_matrix(face_dof_indices[velocity_face_i]);
                            if (mass <= std::numeric_limits<double>::min())
                              continue;
                            const unsigned int local_index = local_dof - local_dof_indices.begin();
                            const Tensor<1, dim> phi_u = fe_support_values[
                                                           this->introspection().extractors.velocities].value(local_index, support_index);
                            density_force_normal_stress_derivative +=
                              (phi_u * normal) * local_density_cbf_rhs(local_index) / mass;
                            normal_stress_derivative += (phi_u * normal) * local_viscosity_cbf_rhs(local_index) / mass;
                          }
                        density_surface(cell->active_cell_index()) -= topography_adjoint(face_dof_indices[face_i]) *
                                                                      density_force_normal_stress_derivative / (density_contrast * gravity_norm);
                        viscosity_surface(cell->active_cell_index()) -= topography_adjoint(face_dof_indices[face_i]) *
                                                                        normal_stress_derivative / (density_contrast * gravity_norm);
                      }
                  }
              }
      // Add the explicit 1/Delta-rho derivative after the residual-mediated
      // term. Then convert integrated cell sensitivities to the cell-average
      // convention used by KernelRepository.
      for (const auto &entry : density_surface_by_topography_dof)
        density_surface(entry.second.first) += entry.second.second;
      for (const auto &cell : this->get_dof_handler().active_cell_iterators())
        if (cell->is_locally_owned())
          {
            const unsigned int cell_index = cell->active_cell_index();
            AssertThrow(cell_volumes(cell_index) > std::numeric_limits<double>::min(),
                        ExcMessage("DynamicTopographyBoundaryOperator encountered a locally owned cell with zero volume."));
            density_surface(cell_index) /= cell_volumes(cell_index);
            viscosity_surface(cell_index) /= cell_volumes(cell_index);
          }
      const std::string contribution_name = boundary == Postprocess::DynamicTopographyBoundary::surface
                                            ? "dynamic topography surface objective"
                                            : "dynamic topography CMB objective";
      kernels.add_contribution({objective_name, contribution_name, PhysicalProperty::density},
                               density_surface);
      kernels.add_contribution({objective_name, contribution_name, PhysicalProperty::viscosity},
                               viscosity_surface);
    }

    template <int dim>
    double
    MaterialModelAdjoint<dim>::seeded_output_scalar(const MaterialModel::MaterialModelOutputs<dim> &out,
                                                    const MaterialModelOutputAdjoints<dim> &bar_out,
                                                    const unsigned int q) const
    {
      // Contract the material outputs y=(eta,rho,c_p,k) with their adjoint
      // seeds. For one quadrature point this scalar is
      //   phi(x)=bar_eta*eta(x)+bar_rho*rho(x)
      //          +bar_c_p*c_p(x)+bar_k*k(x).
      // Differentiating phi with respect to a material input x is exactly the
      // vector-Jacobian product J_material(x)^T bar_y required by reverse mode.
      return bar_out.viscosities[q] * out.viscosities[q]
             + bar_out.densities[q] * out.densities[q]
             + bar_out.specific_heat[q] * out.specific_heat[q]
             + bar_out.thermal_conductivities[q] * out.thermal_conductivities[q];
    }



    template <int dim>
    void
    MaterialModelAdjoint<dim>::apply_derivative_transpose(const MaterialModel::MaterialModelInputs<dim> &in,
                                                          const MaterialModel::MaterialModelOutputs<dim> &out,
                                                          const MaterialModelOutputAdjoints<dim> &bar_out,
                                                          MaterialModelInputAdjoints<dim> &bar_in) const
    {
      // bar_in is an accumulation target, not an output to overwrite. Multiple
      // residual terms can seed the same material evaluation, so every
      // derivative below is added to the existing temperature, pressure, or
      // composition adjoint at its quadrature point.
      const unsigned int n_points = in.n_evaluation_points();
      AssertDimension(out.n_evaluation_points(), n_points);
      (void) out;
      AssertDimension(bar_out.viscosities.size(), n_points);
      AssertDimension(bar_out.densities.size(), n_points);
      AssertDimension(bar_out.specific_heat.size(), n_points);
      AssertDimension(bar_out.thermal_conductivities.size(), n_points);
      AssertDimension(bar_in.temperature.size(), n_points);
      AssertDimension(bar_in.pressure.size(), n_points);
      AssertDimension(bar_in.composition.size(), n_points);

      const unsigned int n_compositional_fields =
        n_points > 0 ? in.composition[0].size() : 0;
      for (unsigned int q = 0; q < n_points; ++q)
        {
          AssertDimension(bar_in.composition[q].size(), n_compositional_fields);
          AssertDimension(in.composition[q].size(), n_compositional_fields);
        }

      const bool seed_viscosity = has_nonzero_entry(bar_out.viscosities);
      const bool seed_density = has_nonzero_entry(bar_out.densities);
      const bool seed_specific_heat = has_nonzero_entry(bar_out.specific_heat);
      const bool seed_thermal_conductivity =
        has_nonzero_entry(bar_out.thermal_conductivities);

      // A zero output covector has a zero transpose action. Besides avoiding
      // material-model evaluations, this guard prevents rejecting unsupported
      // dependencies that are irrelevant to the active objective path.
      if (!seed_viscosity
          && !seed_density
          && !seed_specific_heat
          && !seed_thermal_conductivity)
        return;

      const MaterialModel::NonlinearDependence::ModelDependence model_dependence =
        this->get_material_model().get_model_dependence();

      // The finite-difference backend perturbs scalar T, p, and C inputs only.
      // Strain rate is tensor-valued and pressure-dependent viscosity is not
      // part of the audited v1 Stokes/material transpose. Fail before forming
      // a partial VJP whenever a nonzero viscosity seed would require either
      // omitted derivative.
      AssertThrow(!(seed_viscosity
                    && dependence_includes(model_dependence.viscosity,
                                           MaterialModel::NonlinearDependence::strain_rate)),
                  ExcMessage("The adjoint generated a non-zero viscosity seed, but the active material "
                             "model declares strain-rate dependence and the current material-adjoint "
                             "backend does not support strain-rate derivatives. Continuing would produce "
                             "an incomplete discrete derivative."));
      AssertThrow(!(seed_viscosity
                    && dependence_includes(model_dependence.viscosity,
                                           MaterialModel::NonlinearDependence::pressure)),
                  ExcMessage("The adjoint generated a non-zero viscosity seed, but the active material "
                             "model declares pressure-dependent viscosity. The v1 material-adjoint "
                             "layer supports only eta(T,C), so eta(p) must remain disabled."));

      auto any_seeded_output_depends_on =
        [&](const MaterialModel::NonlinearDependence::Dependence variable)
      {
        return (seed_viscosity
                && dependence_includes(model_dependence.viscosity, variable))
               || (seed_density
                   && dependence_includes(model_dependence.density, variable))
               || (seed_specific_heat
                   && dependence_includes(model_dependence.specific_heat, variable))
               || (seed_thermal_conductivity
                   && dependence_includes(model_dependence.thermal_conductivity, variable));
      };

      // Request only outputs that carry a nonzero seed. A material plugin may
      // allocate expensive additional outputs, so the transpose must preserve
      // the same requested-properties contract as an ordinary forward call.
      MaterialModel::MaterialProperties::Property requested_properties =
        MaterialModel::MaterialProperties::uninitialized;
      if (seed_viscosity)
        requested_properties = add_property(requested_properties,
                                            MaterialModel::MaterialProperties::viscosity);
      if (seed_density)
        requested_properties = add_property(requested_properties,
                                            MaterialModel::MaterialProperties::density);
      if (seed_specific_heat)
        requested_properties = add_property(requested_properties,
                                            MaterialModel::MaterialProperties::specific_heat);
      if (seed_thermal_conductivity)
        requested_properties = add_property(requested_properties,
                                            MaterialModel::MaterialProperties::thermal_conductivity);

      auto evaluate_seeded_scalar = [&](MaterialModel::MaterialModelInputs<dim> material_in,
                                        const unsigned int q)
      {
        // Evaluate the scalar contraction phi rather than finite-differencing
        // every material output separately. By linearity,
        //   d phi/dx = sum_j bar_y_j dy_j/dx = J^T bar_y,
        // which needs two material evaluations per active scalar input.
        MaterialModel::MaterialModelOutputs<dim> material_out(n_points,
                                                              n_compositional_fields);
        material_in.requested_properties = requested_properties;
        this->get_material_model().evaluate(material_in, material_out);

        return seeded_output_scalar(material_out, bar_out, q);
      };

      if (any_seeded_output_depends_on(MaterialModel::NonlinearDependence::temperature))
        // Central differences approximate the local VJP component by
        // [phi(T_q+eps)-phi(T_q-eps)]/(2 eps). Only point q is perturbed;
        // material plugins are assumed point-local under the standard ASPECT
        // material-model interface.
        for (unsigned int q = 0; q < n_points; ++q)
          {
            MaterialModel::MaterialModelInputs<dim> in_plus = in;
            MaterialModel::MaterialModelInputs<dim> in_minus = in;
            const double epsilon = finite_difference_step(in.temperature[q]);

            in_plus.temperature[q] += epsilon;
            in_minus.temperature[q] -= epsilon;

            bar_in.temperature[q] +=
              (evaluate_seeded_scalar(std::move(in_plus), q)
               - evaluate_seeded_scalar(std::move(in_minus), q))
              / (2.0 * epsilon);
          }

      if (any_seeded_output_depends_on(MaterialModel::NonlinearDependence::pressure))
        // Apply the identical scalar-contraction construction to pressure.
        // Callers that cannot scatter bar_p into a pressure space reject a
        // nonzero result explicitly rather than silently dropping it.
        for (unsigned int q = 0; q < n_points; ++q)
          {
            MaterialModel::MaterialModelInputs<dim> in_plus = in;
            MaterialModel::MaterialModelInputs<dim> in_minus = in;
            const double epsilon = finite_difference_step(in.pressure[q]);

            in_plus.pressure[q] += epsilon;
            in_minus.pressure[q] -= epsilon;

            bar_in.pressure[q] +=
              (evaluate_seeded_scalar(std::move(in_plus), q)
               - evaluate_seeded_scalar(std::move(in_minus), q))
              / (2.0 * epsilon);
          }

      if (any_seeded_output_depends_on(MaterialModel::NonlinearDependence::compositional_fields))
        // Each compositional component is an independent material input. The
        // nested q,c loop therefore forms every column of the local material
        // Jacobian that is touched by the output covector.
        for (unsigned int q = 0; q < n_points; ++q)
          for (unsigned int c = 0; c < n_compositional_fields; ++c)
            {
              MaterialModel::MaterialModelInputs<dim> in_plus = in;
              MaterialModel::MaterialModelInputs<dim> in_minus = in;
              const double epsilon = finite_difference_step(in.composition[q][c]);

              in_plus.composition[q][c] += epsilon;
              in_minus.composition[q][c] -= epsilon;

              bar_in.composition[q][c] +=
                (evaluate_seeded_scalar(std::move(in_plus), q)
                 - evaluate_seeded_scalar(std::move(in_minus), q))
                / (2.0 * epsilon);
            }
    }

    MaterialModelInputAdjointScatter<dim>::MaterialModelInputAdjointScatter(const Simulator<dim> &simulator)
    {
      this->initialize_simulator(simulator);
    }



    template <int dim>
    void
    MaterialModelInputAdjointScatter<dim>::scatter_temperature_advection_inputs(
      const FEValuesBase<dim, dim> &fe_values,
      const std::vector<types::global_dof_index> &local_dof_indices,
      const MaterialModelInputAdjoints<dim> &bar_in,
      const double previous_weight,
      const double previous_previous_weight,
      LinearAlgebra::BlockVector &previous_temperature_adjoint,
      LinearAlgebra::BlockVector *previous_previous_temperature_adjoint,
      std::vector<LinearAlgebra::BlockVector> *previous_composition_adjoints,
      std::vector<LinearAlgebra::BlockVector> *previous_previous_composition_adjoints) const
    {
      // Material properties in the advection residual are evaluated at the
      // extrapolated field
      //   x* = w_{n-1} x_{n-1} + w_{n-2} x_{n-2}.
      // Hence the reverse interpolation sends a quadrature seed bar_x* to
      // w_{n-1} P^T bar_x* and w_{n-2} P^T bar_x*, where P evaluates FE basis
      // functions at quadrature points. This routine performs that P^T scatter
      // for temperature and every compositional field.
      const unsigned int n_points = fe_values.n_quadrature_points;
      const unsigned int dofs_per_cell = this->get_fe().dofs_per_cell;
      AssertDimension(local_dof_indices.size(), dofs_per_cell);
      AssertDimension(bar_in.temperature.size(), n_points);
      AssertDimension(bar_in.pressure.size(), n_points);
      AssertDimension(bar_in.composition.size(), n_points);

      for (const double pressure_adjoint : bar_in.pressure)
        // The audited temperature-advection path has no pressure FE target for
        // a material input adjoint. A nonzero bar_p would therefore make the
        // derivative incomplete and must be rejected before assembly.
        AssertThrow(pressure_adjoint == 0.0,
                    ExcMessage("The material adjoint generated a non-zero pressure input adjoint, "
                               "but the v1 temperature-advection material-input scatter does not "
                               "support pressure DoFs. Continuing would silently drop part of the "
                               "discrete derivative."));

      const unsigned int n_compositional_fields =
        n_points > 0 ? bar_in.composition[0].size() : 0;

      Vector<double> local_previous_temperature(dofs_per_cell);
      Vector<double> local_previous_previous_temperature(dofs_per_cell);
      std::vector<Vector<double>> local_previous_composition(n_compositional_fields,
                                                              Vector<double>(dofs_per_cell));
      std::vector<Vector<double>> local_previous_previous_composition(n_compositional_fields,
                                                                       Vector<double>(dofs_per_cell));

      // Assemble local P^T actions. If BDF1 supplies no n-2 vector, fold its
      // (normally zero) weight into n-1; this keeps the routine valid for the
      // common two-level call shape without allocating a dummy global vector.
      for (unsigned int q = 0; q < n_points; ++q)
        {
          AssertDimension(bar_in.composition[q].size(), n_compositional_fields);

          if (bar_in.temperature[q] != 0.0)
            for (unsigned int i = 0; i < dofs_per_cell; ++i)
              if (this->get_fe().system_to_component_index(i).first
                  == this->introspection().component_indices.temperature)
                {
                  const double contribution =
                    bar_in.temperature[q]
                    * fe_values[this->introspection().extractors.temperature].value(i, q);
                  local_previous_temperature(i) += previous_weight * contribution;
                  if (previous_previous_temperature_adjoint != nullptr)
                    local_previous_previous_temperature(i) += previous_previous_weight * contribution;
                  else
                    local_previous_temperature(i) += previous_previous_weight * contribution;
                }

          for (unsigned int c = 0; c < n_compositional_fields; ++c)
            if (bar_in.composition[q][c] != 0.0)
              {
                if (previous_composition_adjoints == nullptr)
                  continue;

                AssertDimension(previous_composition_adjoints->size(), n_compositional_fields);
                if (previous_previous_composition_adjoints != nullptr)
                  AssertDimension(previous_previous_composition_adjoints->size(),
                                  n_compositional_fields);

                const AdvectionField composition_field = AdvectionField::composition(c);
                const FEValuesExtractors::Scalar composition_extractor =
                  composition_field.scalar_extractor(this->introspection());
                const unsigned int composition_component =
                  composition_field.component_index(this->introspection());

                for (unsigned int i = 0; i < dofs_per_cell; ++i)
                  if (this->get_fe().system_to_component_index(i).first
                      == composition_component)
                    {
                      const double contribution =
                        bar_in.composition[q][c]
                        * fe_values[composition_extractor].value(i, q);
                      local_previous_composition[c](i) += previous_weight * contribution;
                      if (previous_previous_composition_adjoints != nullptr)
                        local_previous_previous_composition[c](i) +=
                          previous_previous_weight * contribution;
                      else
                        local_previous_composition[c](i) +=
                          previous_previous_weight * contribution;
                    }
              }
        }

      this->get_current_constraints().distribute_local_to_global(
        local_previous_temperature,
        local_dof_indices,
        previous_temperature_adjoint);

      // Constraint distribution is part of the discrete transpose: local
      // constrained DoF contributions are mapped to their masters before the
      // caller compresses the distributed accumulation vectors.
      if (previous_previous_temperature_adjoint != nullptr)
        this->get_current_constraints().distribute_local_to_global(
          local_previous_previous_temperature,
          local_dof_indices,
          *previous_previous_temperature_adjoint);

      if (previous_composition_adjoints != nullptr)
        for (unsigned int c = 0; c < n_compositional_fields; ++c)
          {
            this->get_current_constraints().distribute_local_to_global(
              local_previous_composition[c],
              local_dof_indices,
              (*previous_composition_adjoints)[c]);

            if (previous_previous_composition_adjoints != nullptr)
              this->get_current_constraints().distribute_local_to_global(
                local_previous_previous_composition[c],
                local_dof_indices,
                (*previous_previous_composition_adjoints)[c]);
          }
    }
  }
}

    template <int dim>
    StokesAdjointSolver<dim>::StokesAdjointSolver(Simulator<dim> &simulator_object)
      :
      simulator(simulator_object)
    {
      this->initialize_simulator(simulator_object);
    }



    template <int dim>
    void
    StokesAdjointSolver<dim>::solve(
      const LinearAlgebra::BlockVector &right_hand_side,
      LinearAlgebra::BlockVector &adjoint_solution)
    {
      // Under the supported incompressible formulation the linearized mixed
      // Stokes operator has the saddle-point form
      //
      //       [ A  B^T ] [u] = [f],
      //       [ B   0  ] [p]   [g],
      //
      // with symmetric viscous block A and matching divergence/gradient
      // blocks. Its discrete transpose is therefore the same constrained
      // operator. We can reuse Simulator::solve_stokes after replacing the
      // primal RHS by the objective seed and homogenizing all affine boundary
      // data. ScopedSimulatorState restores every overwritten solver field on
      // normal return and exception paths.
      ScopedSimulatorState<dim> simulator_state(simulator);
      const AffineConstraints<double> forward_constraints(simulator.current_constraints);
      const unsigned int velocity_block = this->introspection().block_indices.velocities;
      const unsigned int pressure_block = this->introspection().block_indices.pressure;

      // The exported adjoint vector is owned-only because it is accumulated
      // into kernels. The solver temporary is ghosted because pressure
      // normalization reads locally relevant pressure values.
      adjoint_solution.reinit(this->introspection().index_sets.system_partitioning,
                              this->get_mpi_communicator());
      adjoint_solution = 0.0;
      LinearAlgebra::BlockVector solver_solution(
        this->introspection().index_sets.system_partitioning,
        this->introspection().index_sets.system_relevant_partitioning,
        this->get_mpi_communicator());
      solver_solution = 0.0;

      BoundaryConditions<dim> boundary_conditions;
      // Adjoint perturbations satisfy homogeneous essential conditions. Keep
      // the component relations of the forward AffineConstraints (periodic,
      // hanging-node, and tangential free-slip couplings), but remove their
      // inhomogeneous values before zeroing constrained RHS entries.
      boundary_conditions.build_homogeneous_constraints(forward_constraints,
                                                        simulator.current_constraints);
      simulator.system_rhs = right_hand_side;
      simulator.current_constraints.set_zero(simulator.system_rhs);

      // Advected fields remain at the forward state, preserving eta(T,C) and
      // rho(T,C) at the primal linearization point. Only the Stokes trial
      // blocks are cleared to form a zero adjoint initial guess; changing T or
      // C here would solve a different transpose operator.
      simulator.current_linearization_point.block(velocity_block) = 0.0;
      simulator.current_linearization_point.block(pressure_block) = 0.0;
      simulator.solution.block(velocity_block) = 0.0;
      simulator.solution.block(pressure_block) = 0.0;

      // The adjoint pressure has its own gauge. A forward normalization offset
      // would otherwise contaminate a zero adjoint initial guess.
      simulator.last_pressure_normalization_adjustment = 0.0;
      simulator.solve_stokes(solver_solution);

      // The Simulator solve uses a locally relevant vector because pressure
      // normalization and constraints inspect ghosts. Export only owned
      // entries: kernel assembly will create its own short-lived read view,
      // and retaining ghosts in a vector that is later accumulated is unsafe.
      for (unsigned int block = 0; block < adjoint_solution.n_blocks(); ++block)
        for (const auto index : adjoint_solution.block(block).locally_owned_elements())
          adjoint_solution.block(block)(index) = solver_solution.block(block)(index);
      adjoint_solution.compress(VectorOperation::insert);
    }
  }
